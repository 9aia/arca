1//! arca-vault-age — the local vault, and the default backend
2//! forever (per PL-001 hard constraints).
3//!
4//! One age-x25519-encrypted JSON doc holds every env's records;
5//! `put`/`rm`/`rotate` rewrite the whole file (fine at v1 scale).
6//! Decrypted payload shape (`v` is the format version):
7//!
8//! ```json
9//! {
10//!   "v": 1,
11//!   "envs": {
12//!     "dev": {
13//!       "billing/stripe": {
14//!         "value": "…", "version": 1,
15//!         "created_at": "…", "rotated_at": "…"
16//!       }
17//!     }
18//!   }
19//! }
20//! ```
21//!
22//! Identity is an age keyfile (one `AGE-SECRET-KEY-…` line,
23//! `#` comments allowed) at `~/.config/arca/identity.txt`;
24//! recipients default to that identity's public key plus any extra
25//! `recipients` in the vault's config. [`AgeVault::ensure_identity`]
26//! generates a fresh keyfile (mode `0600`) when missing — that path
27//! backs `arca init`.
28

41/// Current payload format version.
42const FORMAT_VERSION: u32 = 1;
43
44/// The decrypted vault document.
45#[derive(Debug, Default, Serialize, Deserialize)]
46struct Payload {
47    v: u32,
48    #[serde(default)]
49    envs: BTreeMap<String, BTreeMap<String, Entry>>,
50}
51
52/// One stored secret, as persisted (the plaintext lives only inside
53/// the ciphertext — this struct is the *decrypted* in-memory form).
54#[derive(Debug, Clone, Serialize, Deserialize)]
55struct Entry {
56    value: String,
57    #[serde(default = "one")]
58    version: u64,
59    #[serde(with = "time::serde::rfc3339")]
60    created_at: OffsetDateTime,
61    #[serde(default, with = "time::serde::rfc3339::option")]
62    rotated_at: Option<OffsetDateTime>,
63}
64
65fn one() -> u64 {
66    1
67}
68
69impl Entry {
70    fn record(&self) -> SecretRecord {
71        SecretRecord {
72            value: Secret::new(self.value.clone()),
73            version: self.version,
74            created_at: self.created_at,
75            rotated_at: self.rotated_at,
76        }
77    }
78}

81pub struct AgeVault {

83    identity: x25519::Identity,

95    /// The default identity keyfile — `$ARCA_IDENTITY_FILE`, else

97    /// `~/.config/arca/identity.txt`.
98    pub fn default_identity_path() -> PathBuf {
99        if let Ok(p) = std::env::var("ARCA_IDENTITY_FILE") {

101                return PathBuf::from(p);

107    /// Read + validate an age identity keyfile: first non-empty,
108    /// non-`#`-comment line must be an `AGE-SECRET-KEY-…` string.
109    pub fn read_identity(path: &Path) -> Result<x25519::Identity, VaultError> {
110        let text = std::fs::read_to_string(path)
111            .map_err(|e| VaultError::Config(format!("read identity {}: {e}", path.display())))?;
112        let line = text
113            .lines()
114            .map(str::trim)
115            .find(|l| !l.is_empty() && !l.starts_with('#'))
116            .ok_or_else(|| {
117                VaultError::Config(format!("identity {} holds no key", path.display()))
118            })?;
119        x25519::Identity::from_str(line)
120            .map_err(|e| VaultError::Crypto(format!("identity {}: {e}", path.display())))
121    }
122
123    /// Create the identity keyfile (mode `0600`, parents created)
124    /// iff missing; returns `true` when one was generated — the
125    /// `arca init` path. The public recipient goes in a `#` comment
126    /// line so the file stays rage-compatible.
127    #[allow(clippy::unwrap_used)]
128    pub fn ensure_identity(path: &Path) -> Result<bool, VaultError> {
129        if path.exists() {
130            return Ok(false);
131        }
132        use age::secrecy::ExposeSecret;
133        let identity = x25519::Identity::generate();
134        if let Some(parent) = path.parent() {
135            std::fs::create_dir_all(parent)?;
136        }
137        let key = identity.to_string();
138        let body = format!(
139            "# arca identity — created {}\n# public: {}\n{}\n",
140            OffsetDateTime::now_utc(),
141            identity.to_public(),
142            key.expose_secret(),
143        );
144        std::fs::write(path, body)?;
145        // chmod'ing a file this process just created and owns
146        // cannot realistically fail — an expect is honest here
147        // where a `?` would imply a reachable error arm.
148        set_private(path).expect("chmod on a freshly written file");
149        Ok(true)
150    }
151
152    /// Open the vault. `identity` must exist (see
153    /// [`Self::ensure_identity`] for bootstrap); a missing vault file
154    /// is an empty vault. `extra_recipients` are `age1…` strings
155    /// encrypted to alongside the identity's public key.
156    pub fn open(
157        vault_path: &Path,
158        identity_path: &Path,
159        extra_recipients: &[String],
160    ) -> Result<Self, VaultError> {
161        let identity = Self::read_identity(identity_path)?;
162        let mut recipients = vec![identity.to_public()];
163        for r in extra_recipients {
164            recipients.push(
165                x25519::Recipient::from_str(r)
166                    .map_err(|e| VaultError::Config(format!("invalid recipient {r:?}: {e}")))?,
167            );
168        }
169        let payload = match std::fs::read(vault_path) {
170            Ok(bytes) => Self::decrypt(&identity, &bytes)?,
171            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Payload::default(),
172            Err(e) => return Err(e.into()),
173        };
174        Ok(AgeVault {
175            path: vault_path.to_path_buf(),
176            identity,
177            recipients,
178            state: Mutex::new(payload),
179        })
180    }
181
182    /// [`Self::ensure_identity`] + [`Self::open`] — bootstrap when
183    /// the keyfile is missing, then open. Returns the vault and
184    /// whether a fresh identity was generated.
185    pub fn open_or_init(
186        vault_path: &Path,
187        identity_path: &Path,
188        extra_recipients: &[String],
189    ) -> Result<(Self, bool), VaultError> {
190        let created = Self::ensure_identity(identity_path)?;
191        Ok((
192            Self::open(vault_path, identity_path, extra_recipients)?,
193            created,
194        ))
195    }

197    /// The identity's public recipient (`age1…`).
198    pub fn public_recipient(&self) -> String {
199        self.identity.to_public().to_string()
200    }

202    fn decrypt(identity: &x25519::Identity, bytes: &[u8]) -> Result<Payload, VaultError> {
203        let decryptor = age::Decryptor::new_buffered(bytes)
204            .map_err(|e| VaultError::Crypto(format!("read vault: {e}")))?;
205        if decryptor.is_scrypt() {
206            return Err(VaultError::Crypto(
207                "vault is passphrase-encrypted; recipient key required".into(),
208            ));
209        }
210        let mut reader = decryptor
211            .decrypt(std::iter::once(identity as &dyn age::Identity))
212            .map_err(|e| VaultError::Crypto(format!("decrypt vault: {e}")))?;
213        let mut buf = Vec::new();
214        reader.read_to_end(&mut buf)?;
215        let payload: Payload = serde_json::from_slice(&buf)
216            .map_err(|e| VaultError::Crypto(format!("vault payload: {e}")))?;
217        if payload.v > FORMAT_VERSION {
218            return Err(VaultError::Crypto(format!(
219                "vault format v{} is newer than supported v{FORMAT_VERSION}",
220                payload.v
221            )));
222        }
223        Ok(payload)
224    }
225
226    /// Serialize + encrypt a payload to all recipients. In-memory
227    /// encryption cannot realistically fail — `recipients` is never
228    /// empty (the identity's public key is always first) and a
229    /// `Vec` sink never errors — so the `expect`s document
230    /// invariants rather than guard reachable paths.
231    #[allow(clippy::unwrap_used)]
232    fn encrypt_payload(&self, payload: &Payload) -> Vec<u8> {
233        let encryptor = age::Encryptor::with_recipients(
234            self.recipients.iter().map(|r| r as &dyn age::Recipient),
235        )
236        .expect("at least one recipient");
237        let plain = serde_json::to_vec(payload).expect("payload serializes");
238        let mut cipher = Vec::new();
239        {
240            let mut w = encryptor.wrap_output(&mut cipher).expect("in-memory wrap");
241            w.write_all(&plain).expect("in-memory write");
242            w.finish().expect("in-memory finish");
243        }
244        cipher
245    }
246
247    /// Encrypt + atomically rewrite the vault file.
248    #[allow(clippy::unwrap_used)]
249    fn save(&self, payload: &Payload) -> Result<(), VaultError> {
250        let cipher = self.encrypt_payload(payload);
251        if let Some(parent) = self.path.parent() {
252            std::fs::create_dir_all(parent)?;
253        }
254        // tmp + rename so a crash can't leave a truncated vault.
255        let tmp = self.path.with_extension("age.tmp");
256        std::fs::write(&tmp, &cipher)?;
257        set_private(&tmp).expect("chmod on a freshly written file");
258        std::fs::rename(&tmp, &self.path)?;
259        Ok(())
260    }
261
262    /// Lock the payload; poison = unrecoverable backend error.
263    fn state(&self) -> Result<std::sync::MutexGuard<'_, Payload>, VaultError> {
264        self.state
265            .lock()
266            .map_err(|_| VaultError::Backend("vault state lock poisoned".into()))
267    }
268}

270impl Vault for AgeVault {
271    fn get(&self, env: &Env, rref: &Ref) -> Result<Option<SecretRecord>, VaultError> {
272        let p = self.state()?;
273        Ok(p.envs
274            .get(env.as_str())
275            .and_then(|m| m.get(&rref.path()))
276            .map(|e| e.record()))
277    }
278
279    fn put(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
280        let mut p = self.state()?;
281        let map = p.envs.entry(env.as_str().to_string()).or_default();
282        let entry = match map.get_mut(&rref.path()) {
283            Some(e) => {
284                e.value = value.expose().to_string();
285                e.version += 1;
286                e.clone()
287            }
288            None => {
289                let e = Entry {
290                    value: value.expose().to_string(),
291                    version: 1,
292                    created_at: OffsetDateTime::now_utc(),
293                    rotated_at: None,
294                };
295                map.insert(rref.path(), e.clone());
296                e
297            }
298        };
299        self.save(&p)?;
300        Ok(entry.record())
301    }
302
303    fn rm(&self, env: &Env, rref: &Ref) -> Result<bool, VaultError> {
304        let mut p = self.state()?;
305        let map = match p.envs.get_mut(env.as_str()) {
306            Some(m) => m,
307            None => return Ok(false),
308        };
309        if map.remove(&rref.path()).is_none() {
310            return Ok(false);
311        }
312        self.save(&p)?;
313        Ok(true)
314    }
315
316    fn list(&self, env: &Env) -> Result<Vec<Ref>, VaultError> {
317        let p = self.state()?;
318        let mut out = Vec::new();
319        if let Some(map) = p.envs.get(env.as_str()) {
320            for path in map.keys() {
321                out.push(
322                    Ref::from_path(path)
323                        .map_err(|e| VaultError::Backend(format!("stored key {e}")))?,
324                );
325            }
326        }
327        out.sort();
328        Ok(out)
329    }
330
331    fn rotate(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
332        let mut p = self.state()?;
333        let map = p.envs.entry(env.as_str().to_string()).or_default();
334        let e = map
335            .get_mut(&rref.path())
336            .ok_or_else(|| VaultError::NotFound {
337                rref: rref.path(),
338                env: env.as_str().to_string(),
339            })?;
340        e.value = value.expose().to_string();
341        e.version += 1;
342        e.rotated_at = Some(OffsetDateTime::now_utc());
343        let entry = e.clone();
344        self.save(&p)?;
345        Ok(entry.record())
346    }
347}

368/// `chmod 0600` — identity files hold key material; the vault file
369/// is ciphertext but private-by-default is still right.
370#[cfg(unix)]
371fn set_private(path: &Path) -> Result<(), VaultError> {
372    use std::os::unix::fs::PermissionsExt;
373    std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o600))?;
374    Ok(())
375}
376
377#[cfg(not(unix))]
378fn set_private(_path: &Path) -> Result<(), VaultError> {
379    Ok(())
380}

604            "fake-pw".to_string(),
605        ));
606        let mut blob = Vec::new();
607        {
608            let mut w = scrypt.wrap_output(&mut blob).unwrap();
609            w.write_all(b"{}").unwrap();
610            w.finish().unwrap();
611        }
612        let e = AgeVault::decrypt(&id, &blob).unwrap_err();
613        assert!(e.to_string().contains("passphrase"), "{e}");
614        // encrypted to someone else → can't decrypt
615        let other = x25519::Identity::generate();
616        assert!(AgeVault::decrypt(&id, &encrypt_to(&other.to_public(), b"{}")).is_err());
617        // valid envelope, garbage plaintext → payload error
618        let blob = encrypt_to(&id.to_public(), b"not json");
619        assert!(AgeVault::decrypt(&id, &blob).is_err());
620        // a newer format version is refused, honestly
621        let blob = encrypt_to(&id.to_public(), br#"{"v":99,"envs":{}}"#);
622        let e = AgeVault::decrypt(&id, &blob).unwrap_err();
623        assert!(e.to_string().contains("v99"), "{e}");
624        // truncated mid-ciphertext → the stream read fails
625        let good = encrypt_to(&id.to_public(), br#"{"v":1,"envs":{}}"#);
626        assert!(AgeVault::decrypt(&id, &good[..good.len() / 2]).is_err());
627    }

678    #[test]
679    fn save_surfaces_filesystem_failures() {
680        let dir = tempfile::tempdir().unwrap();
681        let id_path = dir.path().join("id.txt");
682        AgeVault::ensure_identity(&id_path).unwrap();
683        let r = Ref::new("arca://x/y").unwrap();
684
685        // parent dir swapped for a file → create_dir_all can't proceed
686        let parent = dir.path().join("swapped");
687        let v_path = parent.join("vault.age");
688        let v = AgeVault::open(&v_path, &id_path, &[]).unwrap();
689        std::fs::write(&parent, "not a dir").unwrap();
690        assert!(v.put(&Env::dev(), &r, Secret::new("fake")).is_err());
691        let _ = std::fs::remove_file(&parent);
692
693        // vault path became a directory → the rename can't land
694        let v_path = dir.path().join("isdir.age");
695        let v = AgeVault::open(&v_path, &id_path, &[]).unwrap();
696        std::fs::create_dir(&v_path).unwrap();
697        assert!(v.put(&Env::dev(), &r, Secret::new("fake")).is_err());
698
699        // read-only vault dir → every mutating verb's save fails
700        #[cfg(unix)]
701        {
702            use std::os::unix::fs::PermissionsExt;
703            let sub = dir.path().join("ro");
704            std::fs::create_dir(&sub).unwrap();
705            let v_path = sub.join("vault.age");
706            let v = AgeVault::open(&v_path, &id_path, &[]).unwrap();
707            v.put(&Env::dev(), &r, Secret::new("fake")).unwrap();
708            std::fs::set_permissions(&sub, std::fs::Permissions::from_mode(0o500)).unwrap();
709            assert!(v.put(&Env::dev(), &r, Secret::new("fake2")).is_err());
710            assert!(v.rm(&Env::dev(), &r).is_err());
711            assert!(v.rotate(&Env::dev(), &r, Secret::new("fake3")).is_err());
712            std::fs::set_permissions(&sub, std::fs::Permissions::from_mode(0o700)).unwrap();
713        }
714    }
