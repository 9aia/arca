1//! arca-vault-age — the local vault, and the default backend
2//! forever (per PL-001 hard constraints).
3//!
4//! One age-x25519-encrypted JSON doc holds every env's records;
5//! `put`/`rm`/`rotate` rewrite the whole file (fine at v1 scale).
6//! Decrypted payload shape (`v` is the format version):
7//!
8//! ```json
9//! {
10//!   "v": 1,
11//!   "envs": {
12//!     "dev": {
13//!       "billing/stripe": {
14//!         "value": "…", "version": 1,
15//!         "created_at": "…", "rotated_at": "…"
16//!       }
17//!     }
18//!   }
19//! }
20//! ```
21//!
22//! The vault takes an already-resolved [`x25519::Identity`] —
23//! where the identity *lives* is the `arca-identity` port's
24//! business (keyfile via `arca-identity-file`, session keyring via
25//! `arca-identity-keyring`; `arca-ops` resolves it). Recipients
26//! default to that identity's public key plus any extra
27//! `recipients` in the vault's config.
28//! [`AgeVault::open_or_init`] is the bootstrap path — it runs
29//! [`arca_identity::ensure`] on the caller's chosen source.
30

48/// The decrypted vault document. **No `Debug`** — the whole point
49/// of the payload is that it holds plaintext; a derived dump would
50/// print every value in the vault.
51#[derive(Default, Serialize, Deserialize)]
52struct Payload {
53    v: u32,
54    #[serde(default)]
55    envs: BTreeMap<String, BTreeMap<String, Entry>>,
56}

58/// One stored secret, as persisted (the plaintext lives only inside
59/// the ciphertext — this struct is the *decrypted* in-memory form).
60/// `Zeroizing` scrubs the value buffer on drop; no `Debug` for the
61/// same reason as [`Payload`].
62#[derive(Clone, Serialize, Deserialize)]
63struct Entry {
64    value: Zeroizing<String>,
65    #[serde(default = "one")]
66    version: u64,
67    #[serde(with = "time::serde::rfc3339")]
68    created_at: OffsetDateTime,
69    #[serde(default, with = "time::serde::rfc3339::option")]
70    rotated_at: Option<OffsetDateTime>,
71}

77/// How long a mutating verb waits for a held vault lock before
78/// failing — vault writes are milliseconds, so this is generous.
79const LOCK_WAIT: std::time::Duration = std::time::Duration::from_millis(2_000);
80/// A lockfile older than this is reclaimed — its holder is
81/// presumed dead (crash between create and remove).
82const LOCK_STALE: std::time::Duration = std::time::Duration::from_secs(120);
83
84/// Cross-process exclusion for the vault's read-modify-write
85/// cycle (MED-3). `flock`/`fcntl` aren't in std and a new
86/// dependency for one syscall is heavier than this: `<vault>.lock`
87/// is created with `create_new` (atomic `O_EXCL`), so exactly one
88/// process holds it. A stale lock from a dead holder is reclaimed
89/// after [`LOCK_STALE`]; `Drop` removes it.
90struct VaultLock {
91    path: PathBuf,
92}
93
94impl VaultLock {
95    fn acquire(vault: &Path) -> Result<Self, VaultError> {
96        let path = vault.with_extension("age.lock");
97        let deadline = std::time::Instant::now() + LOCK_WAIT;
98        loop {
99            match std::fs::OpenOptions::new()
100                .write(true)
101                .create_new(true)
102                .open(&path)
103            {
104                Ok(mut f) => {
105                    // pid + creation timestamp — provenance for a
106                    // human debugging a stuck lock, and the
107                    // staleness signal for the next acquirer
108                    let _ = writeln!(f, "{} {}", std::process::id(), unix_ms());
109                    return Ok(VaultLock { path });
110                }
111                Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
112                    if Self::held_is_stale(&path) {
113                        // best-effort: if another process wins the
114                        // race to re-create, our next loop loses
115                        // create_new anyway
116                        let _ = std::fs::remove_file(&path);
117                        continue;
118                    }
119                    if std::time::Instant::now() >= deadline {
120                        return Err(VaultError::Backend(format!(
121                            "vault is locked by another process ({})",
122                            path.display()
123                        )));
124                    }
125                    std::thread::sleep(std::time::Duration::from_millis(20));
126                }
127                Err(e) => return Err(e.into()),
128            }
129        }
130    }
131
132    /// A held lock is stale when the creation timestamp inside it
133    /// is older than [`LOCK_STALE`] — an unreadable or
134    /// foreign-shaped file is treated as held, never clobbered.
135    fn held_is_stale(path: &Path) -> bool {
136        std::fs::read_to_string(path)
137            .ok()
138            .and_then(|s| s.split_whitespace().nth(1)?.parse::<u64>().ok())
139            .map(|created| unix_ms().saturating_sub(created) > LOCK_STALE.as_millis() as u64)
140            .unwrap_or(false)
141    }
142}
143
144/// Milliseconds since the epoch — the lockfile's creation stamp.
145fn unix_ms() -> u64 {
146    std::time::SystemTime::now()
147        .duration_since(std::time::UNIX_EPOCH)
148        .map(|d| d.as_millis() as u64)
149        .unwrap_or(0)
150}

152impl Drop for VaultLock {
153    fn drop(&mut self) {
154        let _ = std::fs::remove_file(&self.path);
155    }
156}

169/// The age-encrypted local vault.
170pub struct AgeVault {
171    path: PathBuf,
172    identity: x25519::Identity,
173    recipients: Vec<x25519::Recipient>,
174    state: Mutex<Payload>,
175}

177impl AgeVault {
178    /// The default vault file — `$XDG_DATA_HOME/arca/vault.age`,
179    /// else `~/.local/share/arca/vault.age`.
180    pub fn default_vault_path() -> PathBuf {
181        data_home().join("arca/vault.age")
182    }

184    /// Open the vault with an already-resolved identity — where the
185    /// identity came from (keyfile, keyring) is the caller's
186    /// business. A missing vault file is an empty vault.
187    /// `extra_recipients` are `age1…` strings encrypted to alongside
188    /// the identity's public key.
189    pub fn open(
190        vault_path: &Path,
191        identity: x25519::Identity,
192        extra_recipients: &[String],
193    ) -> Result<Self, VaultError> {
194        let mut recipients = vec![identity.to_public()];
195        for r in extra_recipients {
196            recipients.push(
197                x25519::Recipient::from_str(r)
198                    .map_err(|e| VaultError::Config(format!("invalid recipient {r:?}: {e}")))?,
199            );
200        }
201        let payload = match std::fs::read(vault_path) {
202            Ok(bytes) => Self::decrypt(&identity, &bytes)?,
203            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Payload::default(),
204            Err(e) => return Err(e.into()),
205        };
206        // F3: loose vault material is a note, never a failure —
207        // the file may not exist yet (a first-run open is an
208        // empty vault) and missing paths simply don't warn.
209        warn_perms(vault_path, false);
210        if let Some(d) = vault_path.parent() {
211            warn_perms(d, true);
212        }
213        Ok(AgeVault {
214            path: vault_path.to_path_buf(),
215            identity,
216            recipients,
217            state: Mutex::new(payload),
218        })
219    }
220
221    /// [`arca_identity::ensure`] + [`Self::open`] — bootstrap the
222    /// identity on `source` when it holds none, then open. Returns
223    /// the vault and whether a fresh identity was generated.
224    pub fn open_or_init(
225        vault_path: &Path,
226        source: &dyn IdentitySource,
227        extra_recipients: &[String],
228    ) -> Result<(Self, bool), VaultError> {
229        let (identity, created) = arca_identity::ensure(source)
230            .map_err(|e| VaultError::Config(format!("identity from {}: {e}", source.describe())))?;
231        Ok((Self::open(vault_path, identity, extra_recipients)?, created))
232    }

234    /// The identity's public recipient (`age1…`).
235    pub fn public_recipient(&self) -> String {
236        self.identity.to_public().to_string()
237    }

239    fn decrypt(identity: &x25519::Identity, bytes: &[u8]) -> Result<Payload, VaultError> {
240        let decryptor = age::Decryptor::new_buffered(bytes)
241            .map_err(|e| VaultError::Crypto(format!("read vault: {e}")))?;
242        if decryptor.is_scrypt() {
243            return Err(VaultError::Crypto(
244                "vault is passphrase-encrypted; recipient key required".into(),
245            ));
246        }
247        let mut reader = decryptor
248            .decrypt(std::iter::once(identity as &dyn age::Identity))
249            .map_err(|e| VaultError::Crypto(format!("decrypt vault: {e}")))?;
250        // Zeroizing — the decrypted plaintext is scrubbed on drop
251        // rather than left in a freed heap buffer (MED-4).
252        let mut buf = Zeroizing::new(Vec::new());
253        reader.read_to_end(&mut buf)?;
254        let payload: Payload = serde_json::from_slice(&buf)
255            .map_err(|e| VaultError::Crypto(format!("vault payload: {e}")))?;
256        if payload.v > FORMAT_VERSION {
257            return Err(VaultError::Crypto(format!(
258                "vault format v{} is newer than supported v{FORMAT_VERSION}",
259                payload.v
260            )));
261        }
262        Ok(payload)
263    }
264
265    /// Serialize + encrypt a payload to all recipients. In-memory
266    /// encryption cannot realistically fail — `recipients` is never
267    /// empty (the identity's public key is always first) and a
268    /// `Vec` sink never errors — so the `expect`s document
269    /// invariants rather than guard reachable paths.
270    #[allow(clippy::unwrap_used)]
271    fn encrypt_payload(&self, payload: &Payload) -> Vec<u8> {
272        let encryptor = age::Encryptor::with_recipients(
273            self.recipients.iter().map(|r| r as &dyn age::Recipient),
274        )
275        .expect("at least one recipient");
276        let plain = Zeroizing::new(serde_json::to_vec(payload).expect("payload serializes"));
277        let mut cipher = Vec::new();
278        {
279            let mut w = encryptor.wrap_output(&mut cipher).expect("in-memory wrap");
280            w.write_all(&plain).expect("in-memory write");
281            w.finish().expect("in-memory finish");
282        }
283        cipher
284    }
285
286    /// Encrypt + atomically rewrite the vault file — tmp + fsync +
287    /// rename + dir fsync, so a crash can't leave a truncated or
288    /// unflushed vault (MED-3).
289    #[allow(clippy::unwrap_used)]
290    fn save(&self, payload: &Payload) -> Result<(), VaultError> {
291        let cipher = self.encrypt_payload(payload);
292        if let Some(parent) = self.path.parent() {
293            std::fs::create_dir_all(parent)?;
294        }
295        let tmp = self.path.with_extension("age.tmp");
296        {
297            let mut f = open_tmp_exclusive(&tmp)?;
298            f.write_all(&cipher)?;
299            f.sync_all()?;
300        }
301        set_private(&tmp).expect("chmod on a freshly written file");
302        std::fs::rename(&tmp, &self.path)?;
303        sync_dir(self.path.parent());
304        Ok(())
305    }
306
307    /// Lock the payload; poison = unrecoverable backend error.
308    fn state(&self) -> Result<std::sync::MutexGuard<'_, Payload>, VaultError> {
309        self.state
310            .lock()
311            .map_err(|_| VaultError::Backend("vault state lock poisoned".into()))
312    }

314    /// The on-disk payload — a missing file is an empty vault.
315    /// Only [`Self::mutate`] re-reads (under the lock, before
316    /// applying its change): reads keep the open-time cache —
317    /// re-reading on every `get`/`list` would both stall on
318    /// special files and cost a decrypt per verb (MED-3).
319    fn read_disk(&self) -> Result<Payload, VaultError> {
320        match std::fs::read(&self.path) {
321            Ok(bytes) => Self::decrypt(&self.identity, &bytes),
322            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(Payload::default()),
323            Err(e) => Err(e.into()),
324        }
325    }
326
327    /// The mutating-verb spine (MED-3): hold the process mutex,
328    /// take the cross-process [`VaultLock`], **re-read** the vault
329    /// file so another process's completed write isn't clobbered,
330    /// apply `f`, persist when `f` reports a change, and leave
331    /// `state` mirroring the new on-disk payload.
332    fn mutate<T>(
333        &self,
334        f: impl FnOnce(&mut Payload) -> Result<(T, bool), VaultError>,
335    ) -> Result<T, VaultError> {
336        let mut p = self.state()?;
337        // the lockfile lives next to the vault — the dir may not
338        // exist yet on a first write (save creates it too)
339        if let Some(parent) = self.path.parent() {
340            std::fs::create_dir_all(parent)?;
341            // F3: a vault dir others can read/write is worth a
342            // note — it weakens every file-permission we set
343            warn_perms(parent, true);
344        }
345        let _lock = VaultLock::acquire(&self.path)?;
346        *p = self.read_disk()?;
347        let (out, dirty) = f(&mut p)?;
348        if dirty {
349            self.save(&p)?;
350        }
351        Ok(out)
352    }
353}

355/// The tmp file's open options — `create_new` is `O_EXCL`:
356/// `File::create` follows symlinks, so a planted `vault.age.tmp`
357/// symlink would redirect the write onto a victim file (F2). On
358/// unix it's `0600` at create — no group/world-readable window
359/// before the post-write chmod.
360fn tmp_open_options() -> std::fs::OpenOptions {
361    let mut opts = std::fs::OpenOptions::new();
362    opts.write(true).create_new(true);
363    #[cfg(unix)]
364    {
365        use std::os::unix::fs::OpenOptionsExt;
366        opts.mode(0o600);
367    }
368    opts
369}

371/// Open `path` for the tmp write, refusing to follow or clobber
372/// anything already there. The [`VaultLock`] serializes writers,
373/// so an existing tmp is crash debris: a plain file is reclaimed
374/// and retried once (a second `AlreadyExists` means a racer —
375/// the io error surfaces honestly); a symlink or anything exotic
376/// was planted — the save fails rather than touching it (F2).
377fn open_tmp_exclusive(path: &Path) -> Result<std::fs::File, VaultError> {
378    match tmp_open_options().open(path) {
379        Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
380            match std::fs::symlink_metadata(path) {
381                Ok(m) if m.file_type().is_file() => std::fs::remove_file(path)?,
382                _ => {
383                    return Err(VaultError::Backend(format!(
384                        "vault tmp {} exists and isn't a plain file — \
385                         refusing to touch it",
386                        path.display()
387                    )))
388                }
389            }
390            Ok(tmp_open_options().open(path)?)
391        }
392        other => other.map_err(Into::into),
393    }
394}

396/// Best-effort fsync of the vault's directory — the durability
397/// hint for the rename. Failure isn't a vault failure.
398#[cfg(unix)]
399fn sync_dir(parent: Option<&Path>) {
400    if let Some(d) = parent {
401        if let Ok(f) = std::fs::File::open(d) {
402            let _ = f.sync_all();
403        }
404    }
405}

410impl Vault for AgeVault {
411    fn get(&self, env: &Env, rref: &Ref) -> Result<Option<SecretRecord>, VaultError> {
412        let p = self.state()?;
413        Ok(p.envs
414            .get(env.as_str())
415            .and_then(|m| m.get(&rref.path()))
416            .map(|e| e.record()))
417    }
418
419    fn put(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
420        self.mutate(|p| {
421            let map = p.envs.entry(env.as_str().to_string()).or_default();
422            let entry = match map.get_mut(&rref.path()) {
423                Some(e) => {
424                    e.value = Zeroizing::new(value.expose().to_string());
425                    e.version += 1;
426                    e.clone()
427                }
428                None => {
429                    let e = Entry {
430                        value: Zeroizing::new(value.expose().to_string()),
431                        version: 1,
432                        created_at: OffsetDateTime::now_utc(),
433                        rotated_at: None,
434                    };
435                    map.insert(rref.path(), e.clone());
436                    e
437                }
438            };
439            Ok((entry.record(), true))
440        })
441    }
442
443    fn rm(&self, env: &Env, rref: &Ref) -> Result<bool, VaultError> {
444        self.mutate(|p| {
445            let map = match p.envs.get_mut(env.as_str()) {
446                Some(m) => m,
447                None => return Ok((false, false)),
448            };
449            let removed = map.remove(&rref.path()).is_some();
450            Ok((removed, removed))
451        })
452    }
453
454    fn list(&self, env: &Env) -> Result<Vec<Ref>, VaultError> {
455        let p = self.state()?;
456        let mut out = Vec::new();
457        if let Some(map) = p.envs.get(env.as_str()) {
458            for path in map.keys() {
459                out.push(
460                    Ref::from_path(path)
461                        .map_err(|e| VaultError::Backend(format!("stored key {e}")))?,
462                );
463            }
464        }
465        out.sort();
466        Ok(out)
467    }
468
469    fn rotate(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
470        self.mutate(|p| {
471            let map = p.envs.entry(env.as_str().to_string()).or_default();
472            let e = map
473                .get_mut(&rref.path())
474                .ok_or_else(|| VaultError::NotFound {
475                    rref: rref.path(),
476                    env: env.as_str().to_string(),
477                })?;
478            e.value = Zeroizing::new(value.expose().to_string());
479            e.version += 1;
480            e.rotated_at = Some(OffsetDateTime::now_utc());
481            let entry = e.clone();
482            Ok((entry.record(), true))
483        })
484    }
485}

500/// The warning text when `path` is more permissive than vault
501/// material should be — `None` when it's strict enough or doesn't
502/// exist (F3). Files must not exceed `0600`, directories `0700`;
503/// the check is "any bit outside the allowed set", like ssh's
504/// strict-modes rule. Returned (not printed) so tests can assert.
505#[cfg(unix)]
506fn perm_note(path: &Path, is_dir: bool) -> Option<String> {
507    use std::os::unix::fs::PermissionsExt;
508    let mode = std::fs::symlink_metadata(path).ok()?.permissions().mode() & 0o777;
509    let allowed: u32 = if is_dir { 0o700 } else { 0o600 };
510    (mode & !allowed != 0).then(|| {
511        let kind = if is_dir { "directory" } else { "file" };
512        format!(
513            "{kind} {} is group/world-accessible (mode {mode:o}) — \
514             vault material should be {allowed:o}; run \
515             `chmod {allowed:o} {}`",
516            path.display(),
517            path.display()
518        )
519    })
520}
521
522#[cfg(not(unix))]
523fn perm_note(_path: &Path, _is_dir: bool) -> Option<String> {
524    None
525}

527/// F3 stderr note — a loose permission is a warning, never a
528/// vault failure.
529fn warn_perms(path: &Path, is_dir: bool) {
530    if let Some(note) = perm_note(path, is_dir) {
531        arca_notice::note(&note);
532    }
533}

675    fn decrypt_reports_every_failure_mode() {
676        let id = x25519::Identity::generate();
677        // not an age file at all
678        assert!(AgeVault::decrypt(&id, b"garbage").is_err());
679        // passphrase-encrypted → we hold no passphrase
680        let scrypt = age::Encryptor::with_user_passphrase(age::secrecy::SecretString::from(
681            "fake-pw".to_string(),
682        ));
683        let mut blob = Vec::new();
684        {
685            let mut w = scrypt.wrap_output(&mut blob).unwrap();
686            w.write_all(b"{}").unwrap();
687            w.finish().unwrap();
688        }
689        let e = AgeVault::decrypt(&id, &blob).err().unwrap();
690        assert!(e.to_string().contains("passphrase"), "{e}");
691        // encrypted to someone else → can't decrypt
692        let other = x25519::Identity::generate();
693        assert!(AgeVault::decrypt(&id, &encrypt_to(&other.to_public(), b"{}")).is_err());
694        // valid envelope, garbage plaintext → payload error
695        let blob = encrypt_to(&id.to_public(), b"not json");
696        assert!(AgeVault::decrypt(&id, &blob).is_err());
697        // a newer format version is refused, honestly
698        let blob = encrypt_to(&id.to_public(), br#"{"v":99,"envs":{}}"#);
699        let e = AgeVault::decrypt(&id, &blob).err().unwrap();
700        assert!(e.to_string().contains("v99"), "{e}");
701        // truncated mid-ciphertext → the stream read fails
702        let good = encrypt_to(&id.to_public(), br#"{"v":1,"envs":{}}"#);
703        assert!(AgeVault::decrypt(&id, &good[..good.len() / 2]).is_err());
704    }

798    fn writes_serialize_across_vault_handles() {
799        // Two handles on one file — the multi-process shape. The
800        // second handle's put must merge the first's write (the
801        // locked re-read), not clobber it (MED-3).
802        let dir = tempfile::tempdir().unwrap();
803        let v_path = dir.path().join("vault.age");
804        let id = x25519::Identity::generate();
805        let v1 = AgeVault::open(&v_path, id.clone(), &[]).unwrap();

807        let r1 = Ref::new("arca://a/one").unwrap();

810        v2.put(&Env::dev(), &r2, Secret::new("fake-2")).unwrap();
811        // both writes landed — the writer's own view is merged
812        let refs = v2.list(&Env::dev()).unwrap();
813        assert!(refs.contains(&r1) && refs.contains(&r2), "{refs:?}");

822        .unwrap();

824            v1.get(&Env::dev(), &r2).unwrap().unwrap().value.expose(),

826        );
827        // the lockfile is gone once the verb returns
828        assert!(!v_path.with_extension("age.lock").exists());
