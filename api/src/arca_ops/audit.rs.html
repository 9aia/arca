1//! Audit events — the wire stand-in until `arca serve` lands.
2//!
3//! ADR-0001 moved audit ownership to peregrin: no local
4//! `audit.jsonl`. Instead arca emits ndjson lines on **stderr** —
5//! a peregrin/sepia wrapper can capture them, a human sees a
6//! breadcrumb. Keep the struct small and serializable: it's the
7//! wire contract.
8//!
9//! Every verb invocation leaves a trail, uniformly:
10//!
11//! - **ok** — one event per ref materialized or mutated
12//!   (`put`/`get`/`rotate`/`rm`, and per-ref inside `run`/`env`),
13//!   or one op-scoped event for ref-less verbs
14//!   (`init`/`list`/`check`/`ui`).
15//! - **denied** — the op was refused by `[guardrails]` (advisory).
16//! - **error** — the op ran but failed (missing secret, vault
17//!   error, bad ref).
18//!
19//! `manifest`/`schema` run before config load — pure
20//! introspection, no event. A `run` child's non-zero exit is the
21//! child's outcome, not arca's — no `error` event for it.
22
23use std::io::{self, Write};
24
25use serde::Serialize;
26use time::OffsetDateTime;
27
28use arca_core::Op;

30/// What the policy decision point said — recorded on serve-path
31/// audit events so the stream shows the verdict behind the
32/// outcome, not just the outcome. `None` on every other path:
33/// standalone CLI ops have no PDP, and the field stays absent.
34#[derive(Debug, Clone, Serialize)]
35pub struct PolicyVerdict {
36    /// The verdict — `false` is what a `denied` outcome carries.
37    pub allow: bool,
38    /// The PDP's reason, when it gave one (a deny's "why", an
39    /// allow-on-error's "the PDP was unreachable").
40    #[serde(skip_serializing_if = "Option::is_none")]
41    pub reason: Option<String>,
42    /// Conditions the PDP attached (`allow-with-conditions` —
43    /// e.g. redact, session-bound). Advisory: serve surfaces them
44    /// in the response and records them here; enforcing them is
45    /// the requester's job.
46    #[serde(skip_serializing_if = "Option::is_none")]
47    pub obligations: Option<Vec<String>>,
48}
49
50/// How an audited op ended.
51#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
52#[serde(rename_all = "lowercase")]
53pub enum Outcome {
54    /// The op did what was asked (a ref materialized/mutated, or a
55    /// ref-less verb completed).
56    Ok,
57    /// Refused by `[guardrails]` — the op never touched the vault.
58    Denied,
59    /// The op ran and failed (missing secret, vault i/o, bad ref).
60    Error,
61}
62
63/// One materialization, mutation, or op outcome record.
64#[derive(Debug, Clone, Serialize)]
65pub struct AuditEvent {
66    /// RFC 3339 timestamp.
67    #[serde(with = "time::serde::rfc3339")]
68    pub ts: OffsetDateTime,
69    /// Self-claimed actor (`--actor`/`ARCA_ACTOR`) — attribution,
70    /// not identity.
71    pub actor: String,
72    /// The verb.
73    pub op: Op,
74    /// The resolved ref path (no env pin — `env` carries that).
75    /// Absent for op-scoped events (`list`, denials before parse).
76    #[serde(skip_serializing_if = "Option::is_none")]
77    pub rref: Option<String>,
78    /// The env the op ran under.
79    pub env: String,
80    /// How it ended.
81    pub outcome: Outcome,
82    /// The PDP verdict behind this event — serve-path only.
83    #[serde(skip_serializing_if = "Option::is_none")]
84    pub policy: Option<PolicyVerdict>,
85}
86
87impl AuditEvent {
88    /// A ref-scoped success — one materialization or mutation.
89    pub fn materialized(actor: &str, op: Op, rref: &str, env: &str) -> Self {
90        Self::new(actor, op, Some(rref), env, Outcome::Ok)
91    }
92
93    /// An op-scoped event — verbs with no single ref, denials, and
94    /// failures. `rref` carries the raw requested ref when known
95    /// (e.g. a denied `get arca://x/y`).
96    pub fn op(actor: &str, op: Op, rref: Option<&str>, env: &str, outcome: Outcome) -> Self {
97        Self::new(actor, op, rref, env, outcome)
98    }
99
100    fn new(actor: &str, op: Op, rref: Option<&str>, env: &str, outcome: Outcome) -> Self {
101        AuditEvent {
102            ts: OffsetDateTime::now_utc(),
103            actor: actor.to_string(),
104            op,
105            rref: rref.map(str::to_string),
106            env: env.to_string(),
107            outcome,
108            policy: None,
109        }
110    }
111
112    /// Attach the PDP verdict that gated this event (serve path).
113    pub fn with_policy(mut self, verdict: Option<PolicyVerdict>) -> Self {
114        self.policy = verdict;
115        self
116    }
117}

119/// Write `event` as one ndjson line to stderr.
120pub fn emit(event: &AuditEvent) {
121    // stderr is the audit wire — a broken pipe must never take the
122    // broker down, so failures are deliberately swallowed.
123    let _ = emit_to(&mut io::stderr(), event);
124}
125
126/// Write `event` as one ndjson line to `w` — the testable core.
127///
128/// Serializing this POD event cannot fail in practice — every
129/// field is a string/enum — so the `expect` documents an invariant
130/// rather than guards a reachable panic.
131#[allow(clippy::unwrap_used)]
132pub fn emit_to(w: &mut impl Write, event: &AuditEvent) -> io::Result<()> {
133    let line = serde_json::to_string(event).expect("audit event serializes");
134    writeln!(w, "{line}")
135}

142    #[test]
143    fn event_is_ndjson() {
144        let ev = AuditEvent::materialized("devin", Op::Run, "billing/stripe", "prod");
145        let mut buf = Vec::new();
146        emit_to(&mut buf, &ev).unwrap();
147        let line = String::from_utf8(buf).unwrap();
148        assert!(line.ends_with('\n'));
149        let v: serde_json::Value = serde_json::from_str(&line).unwrap();
150        assert_eq!(v["actor"], "devin");
151        assert_eq!(v["op"], "run");
152        assert_eq!(v["rref"], "billing/stripe");
153        assert_eq!(v["env"], "prod");
154        assert_eq!(v["outcome"], "ok");
155        assert!(v["ts"].is_string());
156        assert!(!line.contains("value"), "audit never carries values");
157    }

159    #[test]
160    fn op_scoped_and_outcome_events() {
161        let mut buf = Vec::new();
162        emit_to(
163            &mut buf,
164            &AuditEvent::op("devin", Op::List, None, "dev", Outcome::Ok),
165        )
166        .unwrap();
167        emit_to(
168            &mut buf,
169            &AuditEvent::op(
170                "devin",
171                Op::Get,
172                Some("billing/stripe"),
173                "dev",
174                Outcome::Denied,
175            ),
176        )
177        .unwrap();
178        let text = String::from_utf8(buf).unwrap();
179        let lines: Vec<&str> = text.lines().collect();
180        let v: serde_json::Value = serde_json::from_str(lines[0]).unwrap();
181        assert!(v.get("rref").is_none(), "op-scoped events omit rref");
182        assert_eq!(v["outcome"], "ok");
183        let v: serde_json::Value = serde_json::from_str(lines[1]).unwrap();
184        assert_eq!(v["outcome"], "denied");
185        assert_eq!(v["rref"], "billing/stripe");
186    }

188    #[test]
189    fn policy_verdict_serializes_only_when_attached() {
190        let mut buf = Vec::new();
191        emit_to(
192            &mut buf,
193            &AuditEvent::op("devin", Op::Get, Some("a/b"), "prod", Outcome::Denied).with_policy(
194                Some(PolicyVerdict {
195                    allow: false,
196                    reason: Some("pdp: no prod reads".into()),
197                    obligations: Some(vec!["session-bound".into()]),
198                }),
199            ),
200        )
201        .unwrap();
202        let v: serde_json::Value = serde_json::from_str(&String::from_utf8(buf).unwrap()).unwrap();
203        assert_eq!(v["policy"]["allow"], false);
204        assert_eq!(v["policy"]["reason"], "pdp: no prod reads");
205        assert_eq!(v["policy"]["obligations"][0], "session-bound");
206        // unattached → the field stays absent (skip_serializing_if)
207        let mut buf = Vec::new();
208        emit_to(
209            &mut buf,
210            &AuditEvent::materialized("devin", Op::Get, "a/b", "dev"),
211        )
212        .unwrap();
213        let v: serde_json::Value = serde_json::from_str(&String::from_utf8(buf).unwrap()).unwrap();
214        assert!(v.get("policy").is_none());
215    }

217    #[test]
218    fn a_failing_writer_propagates_the_io_error() {
219        struct Dead;
220        impl Write for Dead {
221            fn write(&mut self, _: &[u8]) -> io::Result<usize> {
222                Err(io::Error::new(io::ErrorKind::BrokenPipe, "dead"))
223            }
224            fn flush(&mut self) -> io::Result<()> {
225                Ok(())
226            }
227        }
228        let mut dead = Dead;
229        // a flush on the dead writer is a no-op — only writes fail
230        dead.flush().unwrap();
231        let ev = AuditEvent::op("devin", Op::List, None, "dev", Outcome::Ok);
232        assert!(emit_to(&mut dead, &ev).is_err());
233        // and `emit` swallows the same failure — stderr can be a
234        // broken pipe, arca must not crash
235        emit(&ev);
236    }
