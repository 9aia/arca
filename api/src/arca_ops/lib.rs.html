1//! arca-ops — the I/O shell every front-end calls.
2//!
3//! `arca-core` is pure domain; this crate is everything that touches
4//! the outside world: config loading + env/vault resolution
5//! (`--env`/`ARCA_ENV` → `[envs]` routing → a concrete
6//! [`arca_vault::Vault`]), the env-pin override on refs, the verb
7//! operations the CLI calls, and the audit ndjson emission on
8//! stderr. `arca` (CLI) is a thin clap shell over this; `arca-tui`
9//! calls the same functions.

125impl Ops {
126    /// Load config, resolve env + actor.
127    ///
128    /// Config search: explicit path → `ARCA_CONFIG` → `./arca.toml`
129    /// → `~/.config/arca/arca.toml`. No file on the default search
130    /// means defaults (local vault, `dev`) — arca works without a
131    /// config; an explicit `--config`/`ARCA_CONFIG` that fails is an
132    /// error, as is a file that fails to parse.
133    pub fn load(opts: LoadOpts) -> Result<Self, OpsError> {
134        let src = TomlSource {
135            path: opts.config_path.clone(),
136        };
137        let located = src.locate();
138        let config = match &located {
139            Some(_) => src.load().map_err(OpsError::Config)?,
140            None if opts.config_path.is_some() => {
141                return Err(OpsError::Config(
142                    src.load().err().unwrap_or_else(|| "unreadable".into()),
143                ));
144            }
145            None => Config::default(),
146        };
147        let env = resolve_env(opts.env.as_deref())?;
148        let actor = resolve_actor(opts.actor.as_deref())?;
149        Ok(Ops {
150            config,
151            config_path: located,
152            env,
153            actor,
154        })
155    }

177    /// Apply the env-pin rule: `arca://prod/x` resolves under `prod`
178    /// regardless of context; anything else uses the selected env.
179    /// Returns `(env, ref-with-pin-stripped)`.
180    pub fn resolve(&self, rref: &Ref) -> (Env, Ref) {
181        match rref.env_pin(|s| self.config.is_known_env(s)) {
182            Some((env, rest)) => (env, rest),
183            None => (self.env.clone(), rref.clone()),
184        }
185    }

187    /// Open the vault an env routes to. `local` (the implicit

190        self.open_vault_named(self.config.vault_for(env))

193    /// Open a vault by `[vaults]` name (`local` resolves to its
194    /// implicit defaults when unconfigured).
195    pub fn open_vault_named(&self, name: &str) -> Result<Box<dyn Vault>, OpsError> {
196        let cfg = self.config.vault_config(name);
197        let kind = cfg.map(|c| c.kind.as_str()).unwrap_or("age");
198        match kind {
199            "age" => {
200                let vault_path = cfg
201                    .and_then(|c| c.path.as_ref())
202                    .map(PathBuf::from)
203                    .unwrap_or_else(AgeVault::default_vault_path);
204                let identity_path = cfg
205                    .and_then(|c| c.identity.as_ref())
206                    .map(PathBuf::from)
207                    .unwrap_or_else(AgeVault::default_identity_path);
208                let recipients = cfg.map(|c| c.recipients.as_slice()).unwrap_or(&[]);
209                let v = AgeVault::open(&vault_path, &identity_path, recipients)?;
210                Ok(Box::new(v))
211            }
212            "cf" => {
213                // `kind` was read off `cfg` — "cf" here means the
214                // vault is configured, so this expect documents the
215                // invariant rather than guarding a reachable path.
216                #[allow(clippy::unwrap_used)]
217                let c = cfg.expect("kind cf implies a configured vault");
218                let url = c.url.as_deref().ok_or_else(|| {
219                    OpsError::Config(format!("[vaults.{name}] kind cf needs a `url`"))
220                })?;
221                let token_env = c.token_env.as_deref().unwrap_or("ARCA_CF_TOKEN");
222                let identity_path = c
223                    .identity
224                    .as_ref()
225                    .map(PathBuf::from)
226                    .unwrap_or_else(AgeVault::default_identity_path);
227                let v = CfVault::new(url, token_env, &identity_path, &c.recipients)?;
228                Ok(Box::new(v))
229            }
230            other => Err(OpsError::Config(format!(
231                "vault {name:?} has unknown kind {other:?} (have: age, cf)"
232            ))),
233        }
234    }

236    /// Open a `[deploy.<name>]` sink — write-only push targets,
237    /// never sources.
238    pub fn open_deploy_named(&self, name: &str) -> Result<Box<dyn Deploy>, OpsError> {
239        let cfg = self
240            .config
241            .deploy_config(name)
242            .ok_or_else(|| OpsError::Config(format!("no [deploy.{name}] table in config")))?;
243        match cfg.kind.as_str() {
244            "cf-secrets-store" => {
245                let need = |field: &str, v: &Option<String>| {
246                    v.clone()
247                        .ok_or_else(|| OpsError::Config(format!("[deploy.{name}] needs `{field}`")))
248                };
249                let token_env = cfg
250                    .token_env
251                    .as_deref()
252                    .unwrap_or(arca_deploy_cf::DEFAULT_TOKEN_ENV);
253                let d = CfSecretsStore::new(
254                    name,
255                    &need("account_id", &cfg.account_id)?,
256                    &need("store_id", &cfg.store_id)?,
257                    token_env,
258                    cfg.base_url.as_deref(),
259                )?;
260                Ok(Box::new(d))
261            }
262            other => Err(OpsError::Config(format!(
263                "deploy target {name:?} has unknown kind {other:?} (have: cf-secrets-store)"
264            ))),
265        }
266    }

268    /// The `(vault file, identity file)` pair an env's vault would
269    /// use — for `init`/`check` reporting without opening anything.
270    pub fn vault_paths(&self, env: &Env) -> Result<(PathBuf, PathBuf, Vec<String>), OpsError> {
271        let name = self.config.vault_for(env);
272        let cfg = self.config.vault_config(name);
273        let kind = cfg.map(|c| c.kind.as_str()).unwrap_or("age");
274        match kind {
275            "age" => {}
276            "cf" => {
277                return Err(OpsError::Config(format!(
278                    "vault {name:?} is a remote cf vault — `arca init` only \
279                     bootstraps local age vaults"
280                )));
281            }
282            other => {
283                return Err(OpsError::Config(format!(
284                    "vault {name:?} has unknown kind {other:?} (have: age, cf)"
285                )));
286            }
287        }
288        Ok((
289            cfg.and_then(|c| c.path.as_ref())
290                .map(PathBuf::from)
291                .unwrap_or_else(AgeVault::default_vault_path),
292            cfg.and_then(|c| c.identity.as_ref())
293                .map(PathBuf::from)
294                .unwrap_or_else(AgeVault::default_identity_path),
295            cfg.map(|c| c.recipients.clone()).unwrap_or_default(),
296        ))
297    }

299    /// `arca init` — ensure the identity exists (generate 0600
300    /// keyfile if missing), then verify the vault opens.
301    pub fn init(&self) -> Result<InitInfo, OpsError> {
302        let (vault_path, identity_path, recipients) = self.vault_paths(&self.env)?;
303        let created = AgeVault::ensure_identity(&identity_path)?;
304        let vault = AgeVault::open(&vault_path, &identity_path, &recipients)?;
305        Ok(InitInfo {
306            vault_path,
307            identity_path,
308            identity_created: created,
309            recipient: vault.public_recipient(),
310        })
311    }

350    /// Push a stored value to a deploy target — the vault is the
351    /// source, the `[deploy.NAME]` table the sink. The plaintext
352    /// value crosses here (and only here): the sink is an external
353    /// store that serves runtimes directly, outside arca's
354    /// ciphertext boundary. Emits an `op = "deploy"` audit event.
355    ///
356    /// `target` selects the sink; absent, a lone configured target
357    /// is used — ambiguity and absence are both errors.
358    pub fn deploy(&self, rref: &Ref, target: Option<&str>) -> Result<DeployReport, OpsError> {
359        let name = match target {
360            Some(t) => t.to_string(),
361            None => match self.config.deploy.len() {
362                1 => self
363                    .config
364                    .deploy
365                    .keys()
366                    .next()
367                    .cloned()
368                    .unwrap_or_default(),
369                0 => return Err(OpsError::Config("no [deploy.*] targets configured".into())),
370                _ => {
371                    let names: Vec<&str> = self.config.deploy.keys().map(String::as_str).collect();
372                    return Err(OpsError::Config(format!(
373                        "multiple deploy targets — pass --target ({})",
374                        names.join(", ")
375                    )));
376                }
377            },
378        };
379        let (env, r) = self.resolve(rref);
380        let rec = self
381            .open_vault(&env)?
382            .get(&env, &r)?
383            .ok_or_else(|| OpsError::NotStored {
384                var: "(deploy)".into(),
385                rref: rref.to_string(),
386                env: env.as_str().to_string(),
387            })?;
388        let sink = self.open_deploy_named(&name)?;
389        let report = sink.push(&env, &r, &rec.value)?;
390        self.audit(Op::Deploy, &r, &env);
391        Ok(DeployReport {
392            target: name,
393            env,
394            rref: r,
395            secret_name: report.name,
396            created: report.created,
397        })
398    }

400    /// Read a secret in an explicit env — the audited read for
401    /// callers that already know the env (the TUI's reveal, which
402    /// browses envs other than the selected one). Same event
403    /// [`get`](Self::get) emits: one materialization per value out.
404    pub fn get_in(&self, env: &Env, rref: &Ref) -> Result<Option<SecretRecord>, OpsError> {
405        let rec = self.open_vault(env)?.get(env, rref)?;
406        if rec.is_some() {
407            self.audit(Op::Get, rref, env);
408        }
409        Ok(rec)

419    /// Refs plus their records (metadata) in the selected env —
420    /// backs `list --json`. Values come along inside
421    /// [`SecretRecord`]s but are never printed.
422    pub fn list_records(
423        &self,
424        env: Option<&Env>,
425    ) -> Result<(Env, Vec<(Ref, SecretRecord)>), OpsError> {
426        let (env, refs) = self.list(env)?;
427        let vault = self.open_vault(&env)?;
428        let mut out = Vec::new();
429        for r in refs {
430            if let Some(rec) = vault.get(&env, &r)? {
431                out.push((r, rec));
432            }
433        }
434        Ok((env, out))
435    }

437    /// Resolve every declared + ad-hoc `VAR = arca://ref` pair into
438    /// a `VAR → Secret` map for injection. Emits one
439    /// materialization event per resolved ref. Missing refs and bad
440    /// var names fail the whole operation — partial injection is
441    /// worse than none.
442    pub fn resolve_for_injection(
443        &self,
444        op: Op,
445        extra: &[(String, String)],
446    ) -> Result<BTreeMap<String, Secret>, OpsError> {
447        // declared surface first, then ad-hoc overrides/additions
448        let mut wanted: BTreeMap<String, Ref> = self.config.secrets.clone();
449        for (var, raw) in extra {
450            if !is_valid_env_var(var) {
451                return Err(OpsError::Msg(format!("invalid env var name {var:?}")));
452            }
453            wanted.insert(var.clone(), raw.parse::<Ref>()?);
454        }
455        let mut vaults: BTreeMap<String, Box<dyn Vault>> = BTreeMap::new();
456        let mut out = BTreeMap::new();
457        for (var, rref) in wanted {
458            if !is_valid_env_var(&var) {
459                return Err(OpsError::Msg(format!(
460                    "[secrets] key {var:?} is not a valid env var name"
461                )));
462            }
463            let (env, r) = self.resolve(&rref);
464            let vault = match vaults.entry(env.as_str().to_string()) {
465                std::collections::btree_map::Entry::Occupied(e) => e.into_mut(),
466                std::collections::btree_map::Entry::Vacant(e) => e.insert(self.open_vault(&env)?),
467            };
468            let rec = vault.get(&env, &r)?.ok_or_else(|| OpsError::NotStored {
469                var: var.clone(),
470                rref: rref.to_string(),
471                env: env.as_str().to_string(),
472            })?;
473            self.audit(op, &r, &env);
474            out.insert(var, rec.value);
475        }
476        Ok(out)
477    }

479    /// `arca check` — validate the config, the declared surface,
480    /// and vault readability. Opens vaults read-only (no identity
481    /// generation, no writes); declared secrets that don't resolve
482    /// and undecryptable vaults are findings.
483    pub fn check(&self) -> CheckReport {
484        let mut problems = Vec::new();
485        for var in self.config.secrets.keys() {
486            if !is_valid_env_var(var) {
487                problems.push(format!("[secrets] key {var:?} is not a valid env var name"));
488            }
489        }
490        for (env, route) in &self.config.envs {
491            let name = route.vault();
492            if name != arca_core::config::LOCAL_VAULT && !self.config.vaults.contains_key(name) {
493                problems.push(format!(
494                    "[envs] {env} routes to unconfigured vault {name:?}"
495                ));
496            }
497        }
498        for (name, v) in &self.config.vaults {
499            if !matches!(v.kind.as_str(), "age" | "cf") {
500                problems.push(format!("[vaults.{name}] kind {:?} — have: age, cf", v.kind));
501            }
502            if v.kind == "cf" && v.url.is_none() {
503                problems.push(format!("[vaults.{name}] kind cf needs a `url`"));
504            }
505        }
506        for (name, d) in &self.config.deploy {
507            if d.kind != "cf-secrets-store" {
508                problems.push(format!(
509                    "[deploy.{name}] kind {:?} — have: cf-secrets-store",
510                    d.kind
511                ));
512            }
513            if d.account_id.is_none() || d.store_id.is_none() {
514                problems.push(format!("[deploy.{name}] needs `account_id` and `store_id`"));
515            }
516        }
517
518        // Every vault worth reporting on: configured names plus
519        // every route's target (the implicit `local` included).
520        let mut vault_names: Vec<String> = self.config.vaults.keys().cloned().collect();
521        for route in self.config.envs.values() {
522            let name = route.vault().to_string();
523            if !vault_names.contains(&name) {
524                vault_names.push(name);
525            }
526        }
527        let selected_vault = self.config.vault_for(&self.env).to_string();
528        if !vault_names.contains(&selected_vault) {
529            vault_names.push(selected_vault.clone());
530        }
531        vault_names.sort();
532        vault_names.dedup();
533
534        let mut vaults = Vec::new();
535        let mut opened: BTreeMap<String, Box<dyn Vault>> = BTreeMap::new();
536        let mut failures: BTreeMap<String, String> = BTreeMap::new();
537        for name in &vault_names {
538            let cfg = self.config.vault_config(name);
539            let kind = cfg.map(|c| c.kind.clone()).unwrap_or_else(|| "age".into());
540            let path = cfg.and_then(|c| c.path.as_ref()).map(PathBuf::from);
541            match self.open_vault_named(name).and_then(|v| {
542                // Probe: construction is cheap (no I/O); a `list`
543                // proves a remote vault is actually reachable.
544                match v.list(&self.env) {
545                    Ok(_) => Ok(v),
546                    Err(e) => Err(e.into()),
547                }
548            }) {
549                Ok(v) => {
550                    opened.insert(name.clone(), v);
551                    vaults.push(VaultCheck {
552                        name: name.clone(),
553                        kind,
554                        path,
555                        ok: true,
556                        error: None,
557                    });
558                }
559                Err(e) => {
560                    problems.push(format!("vault {name:?} is not readable: {e}"));
561                    failures.insert(name.clone(), e.to_string());
562                    vaults.push(VaultCheck {
563                        name: name.clone(),
564                        kind,
565                        path,
566                        ok: false,
567                        error: Some(e.to_string()),
568                    });
569                }
570            }
571        }
572
573        // Declared surface: does each VAR's ref resolve in its env?
574        let mut secrets = Vec::new();
575        for (var, rref) in &self.config.secrets {
576            let (env, r) = self.resolve(rref);
577            let vault_name = self.config.vault_for(&env);
578            let resolves = match opened.get(vault_name) {
579                Some(v) => v.get(&env, &r).map(|rec| rec.is_some()).unwrap_or(false),
580                None => {
581                    // The vault probe above already tried this name
582                    // and it failed — a second open would fail
583                    // identically, so report the recorded error.
584                    let err = failures
585                        .get(vault_name)
586                        .map(String::as_str)
587                        .unwrap_or("unknown error");
588                    problems.push(format!(
589                        "{var} → {rref}: env {env} vault {vault_name:?} not readable: {err}"
590                    ));
591                    false
592                }
593            };
594            if !resolves && is_valid_env_var(var) {
595                problems.push(format!("{var} → {rref} is not stored in env {env}"));
596            }
597            secrets.push(SecretCheck {
598                var: var.clone(),
599                rref: r.to_string(),
600                env: env.as_str().to_string(),
601                resolves,
602            });
603        }
604
605        CheckReport {
606            config_path: self.config_path.clone(),
607            env: self.env.clone(),
608            vault: selected_vault,
609            vaults,
610            secrets,
611            guardrails: self.config.guardrails.clone(),
612            problems,
613        }
614    }

639/// `--env` → `ARCA_ENV` → `dev`.
640fn resolve_env(flag: Option<&str>) -> Result<Env, OpsError> {
641    let name = flag
642        .map(str::to_string)
643        .or_else(|| std::env::var("ARCA_ENV").ok())
644        .unwrap_or_else(|| "dev".into());
645    Ok(Env::new(&name)?)
646}
647
648/// `--actor`/`ARCA_ACTOR` (must validate) → `$USER`/`$LOGNAME`
649/// (best-effort) → `anon`.
650fn resolve_actor(flag: Option<&str>) -> Result<Actor, OpsError> {
651    let claimed = flag
652        .map(str::to_string)
653        .or_else(|| std::env::var("ARCA_ACTOR").ok());
654    if let Some(a) = claimed {
655        return Ok(Actor::new(&a)?);
656    }
657    for var in ["USER", "LOGNAME"] {
658        if let Ok(u) = std::env::var(var) {
659            if let Ok(a) = Actor::new(&u) {
660                return Ok(a);
661            }
662        }
663    }
664    Ok(Actor::new("anon")?)
665}

1267    #[test]
1268    fn cf_backend_errors_propagate_through_verbs() {
1269        use arca_testkit::httpmock::{HttpMock, Response};
1270        // List says the ref exists; single-get says otherwise or
1271        // just fails — the verbs surface backend errors, no panic.
1272        let mock = HttpMock::new(|req| {
1273            if req.path == "/v1/dev" {
1274                Response::json(200, r#"{"refs":["gone/one","x/y"]}"#)
1275            } else if req.path.ends_with("/gone/one") {
1276                Response::json(404, r#"{"error":"not found"}"#)
1277            } else {
1278                Response::json(500, r#"{"error":"boom"}"#)
1279            }
1280        });
1281        let home = tempfile::tempdir().unwrap();
1282        let id_path = write_identity(home.path());
1283        let ops = ops_on_cf(mock.url(), "ARCA_TEST_OPS_TOKEN", &id_path);
1284        temp_env::with_var("ARCA_TEST_OPS_TOKEN", Some("fake-token"), || {
1285            let r = Ref::new("arca://x/y").unwrap();
1286            // backend 500s propagate as vault errors on every verb
1287            assert!(ops.get(&r).is_err());
1288            assert!(ops.put(&r, Secret::new("fake")).is_err());
1289            assert!(ops.rm(&r).is_err());
1290            assert!(ops.rotate(&r, Secret::new("fake")).is_err());
1291            // list_records: gone/one → 404 → skipped silently;
1292            // x/y → 500 → the error propagates
1293            assert!(ops.list_records(None).is_err());
1294            // injection hits the same failing get
1295            let mut config = Config::default();
1296            let mut v = vault_cfg("cf", Some(mock.url()));
1297            v.token_env = Some("ARCA_TEST_OPS_TOKEN".into());
1298            v.identity = Some(id_path.to_string_lossy().into_owned());
1299            config.vaults.insert("cloud".into(), v);
1300            config.envs.insert(Env::dev(), "cloud".into());
1301            config.secrets.insert("V".into(), r.clone());
1302            let ops = mk_ops(config);
1303            assert!(ops.resolve_for_injection(Op::Run, &[]).is_err());
1304        });
1305    }
