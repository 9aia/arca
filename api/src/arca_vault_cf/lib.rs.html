1//! arca-vault-cf — the Cloudflare Worker + D1 vault impl.
2//!
3//! A [`Vault`] over HTTPS: a self-hosted Worker (source in
4//! `worker/`) fronts a D1 table of opaque rows —
5//! `(env, ref) → {ciphertext, version, created_at, rotated_at}`.
6//! **The Worker never sees plaintext.** Each value is age-encrypted
7//! client-side to the configured recipients *before* it leaves the
8//! process; the store holds and returns base64 ciphertext only.
9//! This is the same crypto boundary as `arca-vault-age` (ADR-0001)
10//! — a remote store is just a different persistence medium — but
11//! with *per-secret* ciphertext so rows map 1:1 to D1.
12//!
13//! ## Wire protocol (`arca-vault-cf/1`)
14//!
15//! Bearer auth on every request: `Authorization: Bearer <token>`,
16//! the token read from the env var named by `token_env` in config —
17//! never inline.
18//!
19//! | Verb | Path | Body → Response |
20//! | --- | --- | --- |
21//! | `GET` | `/v1/{env}` | `{"refs": ["a/b", …]}` sorted |
22//! | `GET` | `/v1/{env}/{ref}` | `{"ciphertext": "<b64>", "version": n, "created_at": rfc3339, "rotated_at": rfc3339\|null}` |
23//! | `PUT` | `/v1/{env}/{ref}` | `{"ciphertext": "<b64>", "rotate": bool}` → record metadata; `rotate:true` on a missing ref → 404 |
24//! | `DELETE` | `/v1/{env}/{ref}` | `{"removed": bool}` |
25//!
26//! `{ref}` is the bare ref path (`billing/stripe`) — both sides
27//! split on `/`. Errors are `{"error": "…"}` with a non-2xx status.
28//! Bookkeeping is **server-side**: the Worker bumps `version`,
29//! preserves `created_at`, and stamps `rotated_at` on rotate, so
30//! concurrent writers can't race client-computed versions.
31//!
32//! HTTP uses `ureq` (blocking, no async runtime) per AGENTS.md.
33

47/// One row of the wire protocol — `ciphertext` present only on the
48/// single-record GET.

52    ciphertext: Option<String>,
53    version: u64,
54    #[serde(with = "time::serde::rfc3339")]
55    created_at: OffsetDateTime,
56    #[serde(default, with = "time::serde::rfc3339::option")]

58}

75/// A [`Vault`] backed by the Worker + D1 service.
76pub struct CfVault {
77    agent: Agent,
78    base: String,
79    auth: String,
80    identity: x25519::Identity,
81    recipients: Vec<x25519::Recipient>,
82}

84impl CfVault {
85    /// Build a vault from config values: `url` is the Worker base
86    /// (no trailing slash needed), `token_env` names the env var
87    /// holding the bearer token, `identity_path` the age keyfile
88    /// used to decrypt, `extra_recipients` extra `age1…` keys
89    /// encrypted to alongside the identity's public key.
90    ///
91    /// Construction does **no** network I/O — the first verb does.
92    pub fn new(
93        url: &str,
94        token_env: &str,
95        identity_path: &Path,
96        extra_recipients: &[String],
97    ) -> Result<Self, VaultError> {
98        let token = std::env::var(token_env).map_err(|_| {
99            VaultError::Config(format!(
100                "env var {token_env} is unset — cf vault tokens come via env, never config"
101            ))
102        })?;
103        if token.is_empty() {
104            return Err(VaultError::Config(format!(
105                "env var {token_env} is empty — cf vault tokens come via env, never config"
106            )));
107        }
108        let identity = read_identity(identity_path)?;
109        Self::with_identity(url, &token, identity, extra_recipients)
110    }
111
112    /// Build from an already-parsed identity + token — the test and
113    /// embedding path; `new` is the config path.
114    pub fn with_identity(
115        url: &str,
116        token: &str,
117        identity: x25519::Identity,
118        extra_recipients: &[String],
119    ) -> Result<Self, VaultError> {
120        let mut recipients = vec![identity.to_public()];
121        for r in extra_recipients {
122            recipients.push(
123                x25519::Recipient::from_str(r)
124                    .map_err(|e| VaultError::Config(format!("invalid recipient {r:?}: {e}")))?,
125            );
126        }
127        let agent: Agent = Agent::config_builder()
128            .timeout_global(Some(Duration::from_secs(15)))
129            .build()
130            .into();
131        Ok(CfVault {
132            agent,
133            base: url.trim_end_matches('/').to_string(),
134            auth: format!("Bearer {token}"),
135            identity,
136            recipients,
137        })
138    }
139
140    /// `{base}/v1/{env}/{ref}` — ref segments are URL-safe by the
141    /// `Ref` grammar (`[A-Za-z0-9._-]`), no escaping needed.
142    fn item_url(&self, env: &Env, rref: &Ref) -> String {
143        format!("{}/v1/{}/{}", self.base, env.as_str(), rref.path())
144    }
145
146    /// One HTTP round-trip. `http_status_as_error(false)` so error
147    /// bodies are readable; transport failure is `VaultError::Io`.
148    /// (ureq is typestate: GET/DELETE and body-carrying verbs are
149    /// different builder types, so each arm is self-contained.)
150    fn request(
151        &self,
152        method: &str,
153        url: &str,
154        body: Option<String>,
155    ) -> Result<(u16, String), VaultError> {
156        let io = |e: ureq::Error| VaultError::Io(format!("{method} {url}: {e}"));
157        let has_body = body.is_some();
158        let mut res = match (method, body) {
159            ("GET", None) => self
160                .agent
161                .get(url)
162                .header("Authorization", &self.auth)
163                .config()
164                .http_status_as_error(false)
165                .build()
166                .call()
167                .map_err(io),
168            ("DELETE", None) => self
169                .agent
170                .delete(url)
171                .header("Authorization", &self.auth)
172                .config()
173                .http_status_as_error(false)
174                .build()
175                .call()
176                .map_err(io),
177            ("PUT", Some(b)) => self
178                .agent
179                .put(url)
180                .header("Authorization", &self.auth)
181                .config()
182                .http_status_as_error(false)
183                .build()
184                .content_type("application/json")
185                .send(b)
186                .map_err(io),
187            _ => {
188                return Err(VaultError::Backend(format!(
189                    "bad request shape {method} body={has_body}"
190                )))
191            }
192        }?;
193        let status = res.status().as_u16();
194        let text = res
195            .body_mut()
196            .read_to_string()
197            .map_err(|e| VaultError::Io(format!("{method} {url}: read body: {e}")))?;
198        Ok((status, text))
199    }
200
201    /// Extract `{"error": "…"}` from a non-2xx body.
202    fn error_msg(status: u16, body: &str) -> VaultError {
203        if status == 401 || status == 403 {
204            return VaultError::Backend(format!(
205                "HTTP {status} — unauthorized; check the token env var"
206            ));
207        }
208        let msg = serde_json::from_str::<ErrorBody>(body)
209            .ok()
210            .and_then(|e| e.error)
211            .unwrap_or_else(|| body.chars().take(120).collect());
212        VaultError::Backend(format!("HTTP {status}: {msg}"))
213    }
214
215    fn parse_row(body: &str) -> Result<Row, VaultError> {
216        serde_json::from_str(body)
217            .map_err(|e| VaultError::Backend(format!("worker returned bad row json: {e}")))
218    }
219}

221impl Vault for CfVault {
222    fn get(&self, env: &Env, rref: &Ref) -> Result<Option<SecretRecord>, VaultError> {
223        let (status, body) = self.request("GET", &self.item_url(env, rref), None)?;
224        match status {
225            200 => {
226                let row = Self::parse_row(&body)?;
227                let b64 = row
228                    .ciphertext
229                    .ok_or_else(|| VaultError::Backend("worker row missing ciphertext".into()))?;
230                let blob = base64::engine::general_purpose::STANDARD
231                    .decode(&b64)
232                    .map_err(|e| VaultError::Backend(format!("bad b64 ciphertext: {e}")))?;
233                let plain = open(&self.identity, &blob)?;
234                Ok(Some(SecretRecord {
235                    value: Secret::new(plain),
236                    version: row.version,
237                    created_at: row.created_at,
238                    rotated_at: row.rotated_at,
239                }))
240            }
241            404 => Ok(None),
242            other => Err(Self::error_msg(other, &body)),
243        }
244    }
245
246    fn put(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
247        self.write(env, rref, value, false)
248    }
249
250    fn rm(&self, env: &Env, rref: &Ref) -> Result<bool, VaultError> {
251        let (status, body) = self.request("DELETE", &self.item_url(env, rref), None)?;
252        match status {
253            200 => {
254                let out: RmBody = serde_json::from_str(&body).map_err(|e| {
255                    VaultError::Backend(format!("worker returned bad rm json: {e}"))
256                })?;
257                Ok(out.removed)
258            }
259            other => Err(Self::error_msg(other, &body)),
260        }
261    }
262
263    fn list(&self, env: &Env) -> Result<Vec<Ref>, VaultError> {
264        let url = format!("{}/v1/{}", self.base, env.as_str());
265        let (status, body) = self.request("GET", &url, None)?;
266        match status {
267            200 => {
268                let out: ListBody = serde_json::from_str(&body).map_err(|e| {
269                    VaultError::Backend(format!("worker returned bad list json: {e}"))
270                })?;
271                let mut refs = Vec::new();
272                for path in out.refs {
273                    refs.push(
274                        Ref::from_path(&path)
275                            .map_err(|e| VaultError::Backend(format!("stored key {e}")))?,
276                    );
277                }
278                refs.sort();
279                Ok(refs)
280            }
281            other => Err(Self::error_msg(other, &body)),
282        }
283    }
284
285    fn rotate(&self, env: &Env, rref: &Ref, value: Secret) -> Result<SecretRecord, VaultError> {
286        self.write(env, rref, value, true)
287    }
288}

290impl CfVault {
291    /// PUT shared by `put` (`rotate:false`) and `rotate`
292    /// (`rotate:true`); the server does the version bookkeeping.
293    fn write(
294        &self,
295        env: &Env,
296        rref: &Ref,
297        value: Secret,
298        rotate: bool,
299    ) -> Result<SecretRecord, VaultError> {
300        let blob = seal(&self.recipients, value.expose());
301        let b64 = base64::engine::general_purpose::STANDARD.encode(blob);
302        let body = serde_json::json!({ "ciphertext": b64, "rotate": rotate }).to_string();
303        let (status, text) = self.request("PUT", &self.item_url(env, rref), Some(body))?;
304        match status {
305            200 => {
306                let row = Self::parse_row(&text)?;
307                Ok(SecretRecord {
308                    value,
309                    version: row.version,
310                    created_at: row.created_at,
311                    rotated_at: row.rotated_at,
312                })
313            }
314            404 => Err(VaultError::NotFound {
315                rref: rref.path(),
316                env: env.as_str().to_string(),
317            }),
318            other => Err(Self::error_msg(other, &text)),
319        }
320    }
321}

323/// Age-encrypt `plaintext` to `recipients` — per-secret ciphertext,
324/// unlike `arca-vault-age`'s whole-doc encryption (deliberate small
325/// duplication rather than a shared crypto crate). In-memory
326/// encryption cannot realistically fail — callers always pass ≥1
327/// recipient and a `Vec` sink never errors — so the `expect`s
328/// document invariants rather than guard reachable paths.
329#[allow(clippy::unwrap_used)]
330fn seal(recipients: &[x25519::Recipient], plaintext: &str) -> Vec<u8> {
331    let encryptor =
332        age::Encryptor::with_recipients(recipients.iter().map(|r| r as &dyn age::Recipient))
333            .expect("seal is only called with ≥1 recipient");
334    let mut cipher = Vec::new();
335    {
336        let mut w = encryptor.wrap_output(&mut cipher).expect("in-memory wrap");
337        w.write_all(plaintext.as_bytes()).expect("in-memory write");
338        w.finish().expect("in-memory finish");
339    }
340    cipher
341}

343/// Decrypt one per-secret ciphertext blob with `identity`.
344fn open(identity: &x25519::Identity, cipher: &[u8]) -> Result<String, VaultError> {
345    let decryptor = age::Decryptor::new_buffered(cipher)
346        .map_err(|e| VaultError::Crypto(format!("read ciphertext: {e}")))?;
347    if decryptor.is_scrypt() {
348        return Err(VaultError::Crypto(
349            "passphrase-encrypted ciphertext; recipient key required".into(),
350        ));
351    }
352    let mut reader = decryptor
353        .decrypt(std::iter::once(identity as &dyn age::Identity))
354        .map_err(|e| VaultError::Crypto(format!("decrypt: {e}")))?;
355    let mut buf = String::new();
356    reader.read_to_string(&mut buf)?;
357    Ok(buf)
358}

360/// Read + validate an age identity keyfile: first non-empty,
361/// non-`#`-comment line must be an `AGE-SECRET-KEY-…` string (same
362/// format as `arca-vault-age` — deliberately duplicated, ~15 lines).
363fn read_identity(path: &Path) -> Result<x25519::Identity, VaultError> {
364    let text = std::fs::read_to_string(path)
365        .map_err(|e| VaultError::Config(format!("read identity {}: {e}", path.display())))?;
366    let line = text
367        .lines()
368        .map(str::trim)
369        .find(|l| !l.is_empty() && !l.starts_with('#'))
370        .ok_or_else(|| VaultError::Config(format!("identity {} holds no key", path.display())))?;
371    x25519::Identity::from_str(line)
372        .map_err(|e| VaultError::Crypto(format!("identity {}: {e}", path.display())))
373}

380    #[test]
381    fn seal_open_roundtrip_hides_value() {
382        let id = x25519::Identity::generate();
383        let blob = seal(&[id.to_public()], "fake-roundtrip");
384        assert!(!String::from_utf8_lossy(&blob).contains("fake-roundtrip"));
385        assert_eq!(open(&id, &blob).unwrap(), "fake-roundtrip");
386        // a different identity can't open it
387        let other = x25519::Identity::generate();
388        assert!(open(&other, &blob).is_err());
389    }

391    #[test]
392    fn open_rejects_scrypt_and_tampered_blobs() {
393        let id = x25519::Identity::generate();
394        // passphrase-encrypted → we hold no passphrase
395        let scrypt = age::Encryptor::with_user_passphrase(age::secrecy::SecretString::from(
396            "fake-pw".to_string(),
397        ));
398        let mut blob = Vec::new();
399        {
400            let mut w = scrypt.wrap_output(&mut blob).unwrap();
401            w.write_all(b"x").unwrap();
402            w.finish().unwrap();
403        }
404        let e = open(&id, &blob).unwrap_err();
405        assert!(e.to_string().contains("passphrase"), "{e}");
406        // not an age stream at all
407        assert!(open(&id, b"garbage").is_err());
408        // valid envelope but corrupted/truncated body → read fails
409        let good = seal(&[id.to_public()], "fake-body");
410        let cut = &good[..good.len() / 2];
411        assert!(open(&id, cut).is_err());
412    }
