48/// A revision is path-safe — rejects path-injection payloads before
49/// the value reaches worktree paths. Deliberately not hex-only: the
50/// shape comes from the VCS, not the domain.
51pub fn is_valid_rev(rev: &str) -> bool {
52    (7..=64).contains(&rev.len())
53        && !rev.contains("..")
54        && !rev.contains('/')
55        && rev
56            .bytes()
57            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.'))
58}

60/// Bounded dedup set — `insert` returning false means "already
61/// handled". Oldest ids are evicted past `cap`; memory-only.
62pub struct Seen {
63    ids: std::collections::HashSet<String>,
64    order: VecDeque<String>,
65    cap: usize,
66}
67
68impl Seen {
69    pub fn new(cap: usize) -> Self {
70        Seen {
71            ids: Default::default(),
72            order: VecDeque::new(),
73            cap,
74        }
75    }
76
77    /// true = fresh, false = replay/duplicate.
78    pub fn insert(&mut self, id: String) -> bool {
79        if !self.ids.insert(id.clone()) {
80            return false;
81        }
82        self.order.push_back(id);
83        while self.order.len() > self.cap {
84            if let Some(old) = self.order.pop_front() {
85                self.ids.remove(&old);
86            }
87        }
88        true
89    }
90}
91
92impl Default for Seen {
93    fn default() -> Self {
94        Self::new(10_000)
95    }
96}

98/// A `[verbs.<name>]` config-declared verb (P-007). Both kinds are
99/// read-only by construction: an alias bakes argv for a read-only
100/// built-in; a view-emit binds a [`ViewSpec`] to an emit mode.
101/// Mutations stay builtin — the state machine, id alloc, locking
102/// and commit semantics aren't config data.
103#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Serialize)]
104#[serde(untagged)]
105pub enum ConfigVerb {
106    /// `target = "<builtin>"` + baked `args` — `myco <name>` runs
107    /// the builtin with `args` then any extra tokens appended.
108    Alias(AliasVerb),
109    /// A flattened [`ViewSpec`] + `emit` — `myco <name>` evaluates
110    /// the spec and emits in the declared shape.
111    ViewEmit(ViewEmitVerb),
112}
113
114/// The `alias` config verb — name → builtin + baked args.
115#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Serialize)]
116#[serde(deny_unknown_fields)]
117pub struct AliasVerb {
118    /// The built-in verb it invokes (`view`, `status`, `todo.list`, …).
119    pub target: String,
120    /// Baked argv tokens — positionals and `--flag`/`--opt value`
121    /// pairs for the target verb (global `--json`/`--dir`/`--root`/
122    /// `--no-commit` also understood). Extra CLI tokens append.
123    #[serde(default)]
124    pub args: Vec<String>,
125}

127/// The `view-emit` config verb — a [`ViewSpec`] (every field
128/// inline, like `[views.<name>]`) + an emit mode.
129#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Serialize)]
130#[serde(deny_unknown_fields)]
131pub struct ViewEmitVerb {
132    /// The view's spec — same fields as `[views.<name>]`.
133    #[serde(flatten)]
134    pub spec: ViewSpec,
135    /// The output shape.
136    pub emit: EmitMode,
137}

139/// How a `view-emit` verb renders its evaluated spec.
140#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Serialize)]
141#[serde(rename_all = "lowercase")]
142pub enum EmitMode {
143    /// The rendered table/board text.
144    Table,
145    /// The full `{view, spec, groups, interrupted}` payload.
146    Json,
147    /// The matching item ids.
148    Ids,
149    /// The single top item — `{"item": …}` like `next`.
150    First,
151}

153/// A repo's ledger relationship in the org index — the `ledger`
154/// key in `repos.toml` (legacy registry) or `[index] ledger` in
155/// `myco.toml` (P-026). Absent → [`LedgerMode::Local`], the
156/// conventional reading.
157#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize)]
158#[serde(rename_all = "lowercase")]
159pub enum LedgerMode {
160    /// The repo's own ledger — `worktree`, else `<path>/project`.
161    #[default]
162    Local,
163    /// The org/global ledger — the one fallback target
164    /// ledger-less repos resolve to. At most one repo should
165    /// carry it.
166    Org,
167    /// Explicitly ledger-less — an org member with no `project/`
168    /// worktree; verbs run inside it fall back to the `org`
169    /// entry's ledger, and index views contribute no items.
170    None,
171}
172
173impl LedgerMode {
174    /// `local` — the serde-skipped default.
175    pub fn is_local(&self) -> bool {
176        matches!(self, LedgerMode::Local)
177    }
178}

191/// The `myco.toml`-shaped config value the config loader
192/// (`myco_ops::config`) produces. Fields grow per phase; unknown keys are rejected so a typo
193/// fails loud instead of silently doing nothing.
194#[derive(Debug, Clone, Default, Deserialize, Serialize)]
195#[serde(deny_unknown_fields)]
196pub struct Config {
197    /// Path to the project worktree, relative to the repo root.
198    pub project_dir: Option<String>,
199    /// The named view bare `myco view` resolves when no name is
200    /// given — a repo pins its landing view, a machine `~/myco.toml`
201    /// pins a global one (scalar: repo layer wins).
202    pub default_view: Option<String>,
203    /// Named declarative views — merged across config layers
204    /// (P-006): repo `myco.toml` presets are shared, user-config
205    /// presets personal; the *user* layer wins a name collision.
206    #[serde(default)]
207    pub views: std::collections::BTreeMap<String, ViewSpec>,
208    /// Config-declared verbs (P-007) — merged like `views`, but
209    /// the *repo* layer wins collisions (repo verbs are shared
210    /// surface, personal verbs don't override them). Builtins win
211    /// over both at resolution.
212    #[serde(default)]
213    pub verbs: std::collections::BTreeMap<String, ConfigVerb>,
214    /// `[[kinds]]` — user item kinds layered over the prelude
215    /// (P-008). A prelude name *shadows* that spec (prefix stays
216    /// locked); new names extend the table. Merged like verbs:
217    /// the repo layer wins a name collision — a repo's kind
218    /// vocabulary is shared surface.
219    #[serde(default)]
220    pub kinds: Vec<KindSpec>,
221    /// `[sync]` — ledger-branch sync policy. `None` = defaults.
222    /// Repo layer wins the table like the scalars.
223    pub sync: Option<SyncConfig>,
224    /// `[index]` — org-index designation (`ledger = "org"|"none"`).
225    /// The index universe reads the repo-layer `myco.toml`
226    /// directly, so a user-layer table can't leak a global
227    /// designation into every repo.
228    pub index: Option<IndexConfig>,
229    /// `[[cycles]]` — named iteration windows (sprints, releases,
230    /// quarters). Scheduling is config, not ledger memory: item
231    /// lines just carry `cycle:<name>` and `check` warns when a
232    /// name isn't declared here (P-012). Rollover is manual —
233    /// nothing auto-advances; declare the next window when you
234    /// start it. Merged like `kinds`: repo entries shadow user
235    /// entries by name.
236    #[serde(default)]
237    pub cycles: Vec<CycleConfig>,
238}

240/// A `[[cycles]]` entry — one named window between two calendar
241/// dates. Dates are `YYYY-MM-DD` — no times, no timezones; the
242/// iteration boundary is a day. `end` is the last day of the
243/// window (inclusive).
244#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Serialize)]
245#[serde(deny_unknown_fields)]
246pub struct CycleConfig {
247    /// The cycle's name — the value `cycle:<name>` tokens carry.
248    /// Same charset as the other group names
249    /// (`[A-Za-z0-9_.-]`, non-empty) so it round-trips through
250    /// the line grammar.
251    pub name: String,
252    /// First day of the window.
253    #[serde(with = "crate::date::req")]
254    pub start: time::Date,
255    /// Last day of the window (inclusive) — must be after
256    /// `start`.
257    #[serde(with = "crate::date::req")]
258    pub end: time::Date,
259}
260
261/// The cycle-name charset — identical to the line grammar's
262/// group-name rule so any declared name parses back cleanly.
263fn valid_cycle_name(name: &str) -> bool {
264    !name.is_empty()
265        && name
266            .chars()
267            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '.' | '-'))
268}

270/// The `[sync]` config table — how `myco sync` and the mutating
271/// verbs' divergence gate treat the ledger branch's remote.
272#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Serialize)]
273#[serde(deny_unknown_fields)]
274pub struct SyncConfig {
275    /// Fetch `origin/<branch>` before the behind-check every
276    /// mutating verb runs — a stale remote-tracking ref can only
277    /// *under*-report divergence. `false` opts out (offline-first
278    /// repos); fetch failure never blocks a mutation either way.
279    /// Default `true`.
280    #[serde(default = "default_auto_fetch")]
281    pub auto_fetch: bool,
282}
283
284fn default_auto_fetch() -> bool {
285    true
286}

288impl Config {
289    /// The resolved kind table — prelude + this config's
290    /// `[[kinds]]`. Missing `kinds` = the prelude alone, so
291    /// zero-config is today's five kinds.
292    pub fn kind_table(&self) -> Result<KindTable, String> {
293        KindTable::new(self.kinds.clone())
294    }
295
296    /// The validated `[[cycles]]` table — unique names, the
297    /// group-name charset, `end > start`. Config errors are
298    /// hard (config is code, not data), unlike the `check`
299    /// warnings an undeclared `cycle:` token earns.
300    pub fn validated_cycles(&self) -> Result<Vec<CycleConfig>, String> {
301        let mut out = Vec::with_capacity(self.cycles.len());
302        for c in &self.cycles {
303            if !valid_cycle_name(&c.name) {
304                return Err(format!(
305                    "cycle {:?}: name must match [A-Za-z0-9_.-]+",
306                    c.name
307                ));
308            }
309            if c.end <= c.start {
310                return Err(format!(
311                    "cycle {:?}: end {} must be after start {}",
312                    c.name,
313                    crate::date::render(c.end),
314                    crate::date::render(c.start)
315                ));
316            }
317            if out.iter().any(|e: &CycleConfig| e.name == c.name) {
318                return Err(format!("cycle {:?}: duplicate name", c.name));
319            }
320            out.push(c.clone());
321        }
322        Ok(out)
323    }
324
325    /// The cycle covering `d`, if any — the window
326    /// `[start, end]` inclusive on both ends.
327    pub fn cycle_on(&self, d: time::Date) -> Option<&CycleConfig> {
328        self.cycles.iter().find(|c| c.start <= d && d <= c.end)
329    }
330}

337    #[test]
338    fn cycles_validate_names_windows_and_duplicates() {
339        let d = |s: &str| date::parse(s).unwrap();
340        let cfg = |name: &str, s: &str, e: &str| CycleConfig {
341            name: name.into(),
342            start: d(s),
343            end: d(e),
344        };
345        let mut c = Config::default();
346        assert_eq!(c.validated_cycles().unwrap(), Vec::new());
347        c.cycles = vec![cfg("sprint-12", "2026-01-05", "2026-01-18")];
348        assert_eq!(c.validated_cycles().unwrap().len(), 1);
349        // bad charset, inverted window, duplicate name — all hard errors
350        for bad in [
351            vec![cfg("bad name!", "2026-01-05", "2026-01-18")],
352            vec![cfg("s", "2026-01-18", "2026-01-05")],
353            vec![cfg("s", "2026-01-05", "2026-01-05")],
354            vec![
355                cfg("s", "2026-01-05", "2026-01-18"),
356                cfg("s", "2026-02-01", "2026-02-14"),
357            ],
358        ] {
359            c.cycles = bad;
360            assert!(c.validated_cycles().is_err(), "{:?}", c.cycles);
361        }
362        // cycle_on finds the covering window
363        c.cycles = vec![cfg("s", "2026-01-05", "2026-01-18")];
364        assert_eq!(
365            c.cycle_on(d("2026-01-10")).map(|x| x.name.as_str()),
366            Some("s")
367        );
368        assert!(c.cycle_on(d("2026-01-19")).is_none());
369    }
