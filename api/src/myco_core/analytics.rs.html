1//! Analytics emits — roadmap/timeline, stats, and burnup over the
2//! filtered view universe (P-012). Pure domain: callers hand in
3//! filter-applied [`Scoped`] items (the same universe `eval`
4//! consumes); [`myco-cli`] renders the results. The design
5//! principle holds — **fields are data, scheduling is config,
6//! views are emits**: these functions only read `Item` fields and
7//! `[[cycles]]` config, never ledger state.
8//!
9//! - [`timeline`] — epic → milestone → cycle lanes, items placed
10//!   by `due:`. An item lands in exactly one lane: its most
11//!   specific scheduling field wins (`cycle` > `milestone` >
12//!   `epic`); items with none land in `ungrouped`.
13//! - [`stats`] — count-by-dimension tallies (status, priority,
14//!   owner, epic, milestone, cycle, label).
15//! - [`burnup`] — cumulative `done:`-stamped completions vs total
16//!   scope per cycle (or one series when nothing is cycled).
17//!   Items archived before the stamp existed count toward scope
18//!   but never the curve — their completion date is unknowable.
19

27/// Which scheduling dimension a timeline lane groups by.
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29pub enum LaneKind {
30    /// `epic:<name>` — the broadest lane.
31    Epic,
32    /// `ms:<name>` — narrower than epic.
33    Milestone,
34    /// `cycle:<name>` — the iteration window.
35    Cycle,
36    /// No scheduling field — always trails.
37    Ungrouped,
38}
39
40impl LaneKind {
41    /// The lane-kind token — `epic|milestone|cycle|ungrouped`.
42    pub fn name(self) -> &'static str {
43        match self {
44            LaneKind::Epic => "epic",
45            LaneKind::Milestone => "milestone",
46            LaneKind::Cycle => "cycle",
47            LaneKind::Ungrouped => "ungrouped",
48        }
49    }
50
51    /// Sort rank — epic lanes first, ungrouped last.
52    fn rank(self) -> usize {
53        match self {
54            LaneKind::Epic => 0,
55            LaneKind::Milestone => 1,
56            LaneKind::Cycle => 2,
57            LaneKind::Ungrouped => 3,
58        }
59    }
60}

62/// One roadmap lane — every item carrying the lane's scheduling
63/// value (as its most specific field) lands here, `due:`-sorted.
64#[derive(Debug, PartialEq, Eq)]
65pub struct Lane<'a> {
66    /// The dimension this lane groups by.
67    pub kind: LaneKind,
68    /// The epic/milestone/cycle name, or [`UNGROUPED`].
69    pub name: String,
70    /// Lane items — `due:` ascending, undated last, id tiebreak.
71    pub items: Vec<Scoped<'a>>,
72}
73
74/// The lane an item belongs to — most specific scheduling field
75/// wins: `cycle` > `milestone` > `epic` > ungrouped.
76fn lane_of(item: &Item) -> (LaneKind, &str) {
77    if let Some(c) = &item.cycle {
78        (LaneKind::Cycle, c.as_str())
79    } else if let Some(m) = &item.milestone {
80        (LaneKind::Milestone, m.as_str())
81    } else if let Some(e) = &item.epic {
82        (LaneKind::Epic, e.as_str())
83    } else {
84        (LaneKind::Ungrouped, UNGROUPED)
85    }
86}
87
88fn due_then_id(a: &Scoped, b: &Scoped) -> std::cmp::Ordering {
89    a.item
90        .due
91        .cmp(&b.item.due)
92        .then_with(|| a.item.id.cmp(&b.item.id))
93}

95/// Roadmap lanes over the (already filtered) universe. Lane
96/// order: epic lanes alphabetical, then milestone, then cycle,
97/// `ungrouped` last — the epic → milestone → cycle hierarchy the
98/// proposal fixes.
99pub fn timeline<'a>(mut items: Vec<Scoped<'a>>) -> Vec<Lane<'a>> {
100    let mut lanes: BTreeMap<(usize, String), Vec<Scoped>> = BTreeMap::new();
101    for s in items.drain(..) {
102        let (kind, name) = lane_of(s.item);
103        lanes
104            .entry((kind.rank(), name.to_string()))
105            .or_default()
106            .push(s);
107    }
108    lanes
109        .into_iter()
110        .map(|((rank, name), mut items)| {
111            items.sort_by(due_then_id);
112            Lane {
113                kind: match rank {
114                    0 => LaneKind::Epic,
115                    1 => LaneKind::Milestone,
116                    2 => LaneKind::Cycle,
117                    _ => LaneKind::Ungrouped,
118                },
119                name,
120                items,
121            }
122        })
123        .collect()
124}

126/// One count-by-name bucket.
127#[derive(Debug, Clone, PartialEq, Eq)]
128pub struct Tally {
129    /// Bucket name — a status/priority token, owner, group name,
130    /// label, or [`UNGROUPED`].
131    pub name: String,
132    /// Items in the bucket.
133    pub count: usize,
134}

136/// The stats emit — every dimension counted over the same
137/// universe; empty dimensions yield empty vecs (renderers skip
138/// them).
139#[derive(Debug, Default, PartialEq, Eq)]
140pub struct Stats {
141    /// By lifecycle status — canonical [`Status::ALL`] order.
142    pub status: Vec<Tally>,
143    /// By priority — crit-first.
144    pub priority: Vec<Tally>,
145    /// By owner — alphabetical, unclaimed last.
146    pub owner: Vec<Tally>,
147    /// By epic — alphabetical, ungrouped last.
148    pub epic: Vec<Tally>,
149    /// By milestone — alphabetical, ungrouped last.
150    pub milestone: Vec<Tally>,
151    /// By cycle — alphabetical, ungrouped last.
152    pub cycle: Vec<Tally>,
153    /// By label — every observed `#label`, alphabetical.
154    pub label: Vec<Tally>,
155}
156
157fn tally_named<'a>(
158    items: impl Iterator<Item = &'a Item>,
159    pick: impl Fn(&'a Item) -> Option<&'a str>,
160) -> Vec<Tally> {
161    let mut named: BTreeMap<&str, usize> = BTreeMap::new();
162    let mut ungrouped = 0usize;
163    for i in items {
164        match pick(i) {
165            Some(v) => *named.entry(v).or_default() += 1,
166            None => ungrouped += 1,
167        }
168    }
169    let mut out: Vec<Tally> = named
170        .into_iter()
171        .map(|(name, count)| Tally {
172            name: name.to_string(),
173            count,
174        })
175        .collect();
176    if ungrouped > 0 {
177        out.push(Tally {
178            name: UNGROUPED.to_string(),
179            count: ungrouped,
180        });
181    }
182    out
183}

185/// Count-by-dimension tallies over the filtered universe.
186/// Children count too when the caller fed them — stats are
187/// counts, not rows.
188pub fn stats<'a>(items: &[Scoped<'a>]) -> Stats {
189    let mut status_counts = vec![0usize; Status::ALL.len()];
190    let mut pri_counts = vec![0usize; Priority::ALL.len()];
191    let mut labels: BTreeMap<&str, usize> = BTreeMap::new();
192    for s in items {
193        status_counts[s.item.status as usize] += 1;
194        pri_counts[s.item.priority as usize] += 1;
195        for l in &s.item.labels {
196            *labels.entry(l.as_str()).or_default() += 1;
197        }
198    }
199    Stats {
200        status: Status::ALL
201            .iter()
202            .enumerate()
203            .filter(|(i, _)| status_counts[*i] > 0)
204            .map(|(_, s)| Tally {
205                name: s.token().to_string(),
206                count: status_counts[*s as usize],
207            })
208            .collect(),
209        priority: Priority::ALL
210            .iter()
211            .rev()
212            .filter(|p| pri_counts[**p as usize] > 0)
213            .map(|p| Tally {
214                name: p.name().to_string(),
215                count: pri_counts[*p as usize],
216            })
217            .collect(),
218        owner: tally_named(items.iter().map(|s| s.item), |i| {
219            i.owner.as_ref().map(|o| o.as_str())
220        }),
221        epic: tally_named(items.iter().map(|s| s.item), |i| i.epic.as_deref()),
222        milestone: tally_named(items.iter().map(|s| s.item), |i| i.milestone.as_deref()),
223        cycle: tally_named(items.iter().map(|s| s.item), |i| i.cycle.as_deref()),
224        label: labels
225            .into_iter()
226            .map(|(name, count)| Tally {
227                name: name.to_string(),
228                count,
229            })
230            .collect(),
231    }
232}

234/// One day on a burnup curve.
235#[derive(Debug, Clone, Copy, PartialEq, Eq)]
236pub struct BurnupPoint {
237    /// The day.
238    pub date: Date,
239    /// Items whose `done:` stamp is on or before `date`.
240    pub done: u64,
241    /// The series' total scope — constant across the window.
242    pub scope: u64,
243}
244
245/// One burnup series — a cycle's window, or the whole ledger
246/// (`cycle: None`) when items aren't cycled.
247#[derive(Debug, PartialEq, Eq)]
248pub struct BurnupSeries<'a> {
249    /// The `[[cycles]]` name this series tracks; `None` for the
250    /// uncycled remainder (always last when present).
251    pub cycle: Option<&'a str>,
252    /// One point per day in the window, inclusive.
253    pub points: Vec<BurnupPoint>,
254}
255
256/// The burnup window — the `[[cycles]]` window covering `today`
257/// when one exists (the in-flight iteration), else the trailing
258/// 8 weeks. Rollover is manual: outside every declared window the
259/// emit falls back rather than guessing.
260pub fn burnup_window(cycles: &[CycleConfig], today: Date) -> (Date, Date) {
261    if let Some(c) = cycles.iter().find(|c| c.start <= today && today <= c.end) {
262        (c.start, c.end)
263    } else {
264        (today - Duration::days(56), today)
265    }
266}
267
268/// Burnup over `[start, end]` inclusive — one series per observed
269/// `cycle:` value plus one per declared-but-empty cycle, and a
270/// `None` series for uncycled items (last). Scope counts every
271/// item in the partition; `done` counts `done:`-stamped items
272/// whose stamp falls on or before each day — unstamped archive
273/// predates the field and never moves the curve.
274pub fn burnup<'a>(
275    items: &[Scoped<'a>],
276    cycles: &'a [CycleConfig],
277    window: (Date, Date),
278) -> Vec<BurnupSeries<'a>> {
279    let (start, end) = window;
280    if end < start {
281        return Vec::new();
282    }
283    // partition: declared cycles first (sorted), then observed
284    // undeclared names, then the uncycled remainder
285    let mut by_cycle: BTreeMap<Option<&str>, Vec<&Item>> = BTreeMap::new();
286    for s in items {
287        by_cycle
288            .entry(s.item.cycle.as_deref())
289            .or_default()
290            .push(s.item);
291    }
292    let mut series_cycles: Vec<Option<&str>> = Vec::new();
293    for c in cycles {
294        let name = c.name.as_str();
295        if !series_cycles.contains(&Some(name)) {
296            series_cycles.push(Some(name));
297        }
298    }
299    for key in by_cycle.keys() {
300        if key.is_some() && !series_cycles.contains(key) {
301            series_cycles.push(*key);
302        }
303    }
304    if by_cycle.contains_key(&None) {
305        series_cycles.push(None);
306    }
307
308    let days = (end - start).whole_days() as usize + 1;
309    series_cycles
310        .into_iter()
311        .map(|cycle| {
312            let members: &[&Item] = by_cycle.get(&cycle).map(Vec::as_slice).unwrap_or(&[]);
313            let scope = members.len() as u64;
314            let stamps: Vec<Date> = members.iter().filter_map(|i| i.done).collect();
315            let points = (0..days)
316                .map(|d| {
317                    let date = start + Duration::days(d as i64);
318                    BurnupPoint {
319                        date,
320                        done: stamps.iter().filter(|s| **s <= date).count() as u64,
321                        scope,
322                    }
323                })
324                .collect();
325            BurnupSeries { cycle, points }
326        })
327        .collect()
328}

348    #[test]
349    fn timeline_lanes_order_and_specificity() {
350        let items = vec![
351            t(1).epic("e1").due(d("2026-01-05")),
352            t(2).epic("e1").milestone("m1").due(d("2026-01-02")),
353            t(3).cycle("s1")
354                .milestone("m1")
355                .epic("e1")
356                .due(d("2026-01-09")),
357            t(4),
358            t(5).epic("e2"),
359        ];
360        let lanes = timeline(scoped(&items));
361        let names: Vec<(&str, &str)> = lanes
362            .iter()
363            .map(|l| (l.kind.name(), l.name.as_str()))
364            .collect();
365        // epic lanes first (e1 has only t1 — t2/t3 promoted),
366        // then milestone m1 (t2), then cycle s1 (t3), ungrouped last
367        assert_eq!(
368            names,
369            [
370                ("epic", "e1"),
371                ("epic", "e2"),
372                ("milestone", "m1"),
373                ("cycle", "s1"),
374                ("ungrouped", UNGROUPED),
375            ]
376        );
377        // within e1: t1 only; due ordering inside lanes
378        assert_eq!(lanes[0].items[0].item.id.to_string(), "T-001");
379        assert_eq!(lanes[4].items[0].item.id.to_string(), "T-004");
380    }

411    #[test]
412    fn burnup_counts_stamps_and_scope() {
413        let mut done1 = t(1);
414        done1.status = Status::Done;
415        done1.done = Some(d("2026-01-03"));
416        let mut done2 = t(2);
417        done2.status = Status::Done;
418        done2.done = Some(d("2026-01-05"));
419        let mut old = t(3); // archived pre-stamp: scope, never curve
420        old.status = Status::Done;
421        let live = t(4);
422        let items = vec![done1, done2, old, live];
423        let cycles = vec![CycleConfig {
424            name: "s1".into(),
425            start: d("2026-01-01"),
426            end: d("2026-01-07"),
427        }];
428        let series = burnup(&scoped(&items), &cycles, (d("2026-01-01"), d("2026-01-07")));
429        // one declared-cycle series (empty: no items carry cycle s1)
430        // plus the uncycled remainder
431        assert_eq!(series.len(), 2);
432        let uncycled = &series[1];
433        assert_eq!(uncycled.cycle, None);
434        assert_eq!(uncycled.points.len(), 7);
435        assert_eq!(uncycled.points[0].scope, 4);
436        assert_eq!(uncycled.points[0].done, 0);
437        assert_eq!(uncycled.points[2].done, 1); // 01-03 stamp
438        assert_eq!(uncycled.points[4].done, 2); // 01-05 stamp
439        assert_eq!(uncycled.points[6].done, 2); // old never counts
440    }

443    fn burnup_window_prefers_covering_cycle() {
444        let cycles = vec![CycleConfig {
