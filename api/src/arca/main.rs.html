1//! arca — secrets broker for agents. Sealed, not spilled.
2//!
3//! Code and config carry `arca://` references, never values; `arca
4//! run` materializes declared secrets as env vars in the child only.
5//! Every materialization — and every denial/failure — emits an
6//! audit ndjson line on stderr.
7//!
8//! Values never enter via CLI args — `put`/`rotate` read stdin or a
9//! hidden `/dev/tty` prompt (args land in shell history and
10//! `/proc`). `get` is the escape hatch that prints a value to
11//! stdout; `[guardrails]` can advisory-deny it for agents.
12

21/// arca — secrets broker for agents: `arca://` refs, redaction,
22/// env-injection instead of plaintext.
23///
24/// Refs name logical secrets (`arca://billing/stripe`); `--env`
25/// picks the resolution context. `[secrets]` in arca.toml bounds
26/// what `run`/`env` may inject. `manifest`/`schema` describe the
27/// surface for machines; every verb honors `--json`.
28#[derive(Parser)]
29#[command(name = "arca", version, about, long_about)]
30struct Cli {
31    /// Emit JSON — stable machine shapes on every verb; errors are
32    /// `{"error": "…"}` on stderr with a non-zero exit.
33    #[arg(long, global = true)]
34    json: bool,
35
36    /// Env context — `dev`/`staging`/`prod`/custom. Overrides
37    /// `ARCA_ENV`; default `dev`.
38    #[arg(long, global = true, value_name = "ENV")]
39    env: Option<String>,
40
41    /// Config file, overriding the search
42    /// ($ARCA_CONFIG → ./arca.toml → ~/.config/arca/arca.toml).
43    #[arg(long, global = true, value_name = "PATH")]
44    config: Option<String>,
45
46    /// Actor attribution for audit events (default $ARCA_ACTOR,
47    /// else $USER).
48    #[arg(long, global = true, value_name = "ACTOR")]
49    actor: Option<String>,
50
51    #[command(subcommand)]
52    command: Cmd,
53}
54
55#[derive(Subcommand)]
56enum Cmd {
57    /// Bootstrap: generate the age identity (0600) if missing and
58    /// verify the vault opens.
59    Init,
60    /// Store a secret. The value comes from stdin (`printf … | arca
61    /// put arca://ns/name`) or a hidden terminal prompt — never
62    /// args, which land in shell history and `/proc`.
63    Put {
64        /// `arca://` ref, e.g. arca://billing/stripe.
65        rref: String,
66    },
67    /// Print a secret's value to stdout — the escape hatch. Prefer
68    /// `arca run` so values never enter agent context; guardrails
69    /// can deny this op.
70    Get {
71        /// `arca://` ref.
72        rref: String,
73    },
74    /// List the refs stored in the selected env — names only, never
75    /// values.
76    List,
77    /// Remove a secret from the selected env.
78    Rm {
79        /// `arca://` ref.
80        rref: String,
81    },
82    /// Rotate a secret to a new value (stdin or hidden prompt, like
83    /// `put`); bumps version + rotated_at.
84    Rotate {
85        /// `arca://` ref.
86        rref: String,
87    },
88    /// Run a command with declared secrets injected as env vars.
89    ///
90    /// Injection surface = `[secrets]` in arca.toml plus `-s
91    /// VAR=arca://ref` ad-hoc pairs; nothing else is injectable.
92    /// Two exec modes after `--`:
93    ///
94    /// `arca run -- 'pnpm test | tee log'` — one string → `bash -c`
95    ///
96    /// `arca run -- pnpm test --filter x` — argv → direct exec
97    ///
98    /// argv mode needs no shell quoting — prefer it when the
99    /// command has arguments. Exit code is the child's.
100    Run {
101        /// Extra `VAR=arca://ref` pairs to inject.
102        #[arg(short = 's', long = "secret", value_name = "VAR=ARCA_REF")]
103        secrets: Vec<String>,
104        /// Command to run (after `--`). One arg → `bash -c`;
105        /// several → `argv[0]` exec'd directly with the rest.
106        #[arg(last = true, required = true)]
107        cmd: Vec<String>,
108    },
109    /// Print `export VAR='…'` lines for `eval "$(arca env)"` — the
110    /// human path (`run` is the agent path). Values print to
111    /// stdout: `--json` emits a `VAR → value` map.
112    Env {
113        /// Extra `VAR=arca://ref` pairs to resolve.
114        #[arg(short = 's', long = "secret", value_name = "VAR=ARCA_REF")]
115        secrets: Vec<String>,
116    },
117    /// Validate config + the declared surface: `[envs]` routes
118    /// point at declared vaults, vaults decrypt, every `[secrets]`
119    /// ref resolves in its routed env. Exit 1 on findings.
120    Check,
121    /// Push a stored secret to a write-only deploy target
122    /// (`[deploy.NAME]` in arca.toml, e.g. a CF Secrets Store).
123    /// The vault is the source of truth; the sink never gives the
124    /// value back. With one configured target, `--target` may be
125    /// omitted.
126    Deploy {
127        /// `arca://` ref to push.
128        rref: String,
129        /// Deploy target name (`[deploy.NAME]`); required when
130        /// several are configured.
131        #[arg(long, value_name = "NAME")]
132        target: Option<String>,
133    },
134    /// Interactive vault browser (requires the `tui` feature).
135    Ui,
136    /// Capabilities, JSON shapes, env vars, audit contract — agent
137    /// self-discovery (always JSON).
138    Manifest,
139    /// JSON Schema for `arca.toml`.
140    Schema,
141    /// Print the CLI reference as markdown (docs pipeline —
142    /// `bun run docs:cli` pipes it to `docs/reference/cli.md`).
143    #[command(hide = true, name = "gen-cli-docs")]
144    GenCliDocs,
145}

147fn main() -> ExitCode {
148    sigpipe::reset();
149    let cli = Cli::parse();
150    match &cli.command {
151        // Pure introspection — before config load, no audit event.
152        Cmd::Manifest => json_out(&manifest()),
153        Cmd::GenCliDocs => out(&clap_markdown::help_markdown::<Cli>()),
154        Cmd::Schema => out(&schema_json()),
155        cmd => {
156            let op = cmd_op(cmd);
157            let rref = cmd_rref(cmd);
158            let ops = match load(&cli) {
159                Ok(ops) => ops,
160                Err(e) => {
161                    audit_fallback(&cli, op, rref, Outcome::Error);
162                    return fail(&cli, &e.to_string());
163                }
164            };
165            // Advisory guardrails — accident prevention, not
166            // policy; enforced before the op touches the vault.
167            if let Some(g) = ops.config().guardrails.as_ref() {
168                if !g.allows(op) {
169                    let rref = norm_rref(&ops, rref);
170                    ops.emit(op, rref.as_deref(), Outcome::Denied);
171                    // guardrails only exist when a config file
172                    // loaded — config_path is always Some here
173                    #[allow(clippy::unwrap_used)]
174                    let at = ops
175                        .config_path()
176                        .expect("guardrails imply a loaded config file")
177                        .display()
178                        .to_string();
179                    return fail(
180                        &cli,
181                        &format!(
182                            "op '{op}' denied by guardrails in {at} \
183                             (advisory — remove/relax the rule to allow)"
184                        ),
185                    );
186                }
187            }
188            match run_cmd(cmd, &cli, &ops) {
189                Ok(code) => code,
190                Err(e) => {
191                    let rref = norm_rref(&ops, rref);
192                    ops.emit(op, rref.as_deref(), Outcome::Error);
193                    fail(&cli, &e.to_string())
194                }
195            }
196        }
197    }
198}

200/// The [`Op`] a subcommand maps to — what guardrails and audit
201/// record.
202fn cmd_op(cmd: &Cmd) -> Op {
203    match cmd {
204        Cmd::Init => Op::Init,
205        Cmd::Put { .. } => Op::Put,
206        Cmd::Get { .. } => Op::Get,
207        Cmd::List => Op::List,
208        Cmd::Rm { .. } => Op::Rm,
209        Cmd::Rotate { .. } => Op::Rotate,
210        Cmd::Run { .. } => Op::Run,
211        Cmd::Env { .. } => Op::Env,
212        Cmd::Check => Op::Check,
213        Cmd::Deploy { .. } => Op::Deploy,
214        Cmd::Ui => Op::Ui,
215        Cmd::Manifest => Op::Manifest,
216        Cmd::Schema => Op::Schema,
217        // Handled before ops load — this arm is only for
218        // exhaustiveness, never reached.
219        Cmd::GenCliDocs => Op::Schema,
220    }
221}

223/// The raw ref arg for verbs that take one — for denied/error
224/// audit events, before (or regardless of) parse.
225fn cmd_rref(cmd: &Cmd) -> Option<&str> {
226    match cmd {
227        Cmd::Put { rref }
228        | Cmd::Get { rref }
229        | Cmd::Rm { rref }
230        | Cmd::Rotate { rref }
231        | Cmd::Deploy { rref, .. } => Some(rref),
232        _ => None,
233    }
234}

236/// Normalize a requested ref to the resolved-path form success
237/// events carry (`env pin` stripped, no scheme) so the audit
238/// stream correlates regardless of outcome. Falls back to the raw
239/// arg when it doesn't parse — the honest record of a bad request.
240fn norm_rref(ops: &Ops, raw: Option<&str>) -> Option<String> {
241    raw.map(|s| {
242        Ref::new(s)
243            .map(|r| ops.resolve(&r).1.path().to_string())
244            .unwrap_or_else(|_| s.to_string())
245    })
246}

248/// Audit an op that never got an `Ops` (config/env/actor
249/// resolution failed) — best-effort actor + env from flags and
250/// process env.
251fn audit_fallback(cli: &Cli, op: Op, rref: Option<&str>, outcome: Outcome) {
252    let actor = cli
253        .actor
254        .clone()
255        .or_else(|| std::env::var("ARCA_ACTOR").ok())
256        .or_else(|| std::env::var("USER").ok())
257        .unwrap_or_else(|| "anon".into());
258    let env = cli
259        .env
260        .clone()
261        .or_else(|| std::env::var("ARCA_ENV").ok())
262        .unwrap_or_else(|| "dev".into());
263    let rref = rref.map(|s| {
264        Ref::new(s)
265            .map(|r| r.path().to_string())
266            .unwrap_or_else(|_| s.to_string())
267    });
268    arca_ops::audit::emit(&AuditEvent::op(&actor, op, rref.as_deref(), &env, outcome));
269}

271fn load(cli: &Cli) -> Result<Ops, OpsError> {
272    Ops::load(LoadOpts {
273        config_path: cli.config.clone(),
274        env: cli.env.clone(),
275        actor: cli.actor.clone(),
276    })

279fn run_cmd(cmd: &Cmd, cli: &Cli, ops: &Ops) -> Result<ExitCode, OpsError> {
280    match cmd {
281        Cmd::Init => {
282            let info = ops.init()?;
283            ops.emit(Op::Init, None, Outcome::Ok);
284            if cli.json {
285                return Ok(json_out(&serde_json::json!({
286                    "identity": info.identity_path,
287                    "created": info.identity_created,
288                    "recipient": info.recipient,
289                    "vault": info.vault_path,
290                })));
291            }
292            eprintln!(
293                "identity: {}{}",
294                info.identity_path.display(),
295                if info.identity_created {
296                    " (generated)"
297                } else {
298                    ""
299                }
300            );
301            eprintln!("recipient: {}", info.recipient);
302            eprintln!("vault:     {}", info.vault_path.display());
303            Ok(ExitCode::SUCCESS)
304        }
305        Cmd::Put { rref } => {
306            let parsed = parse_ref(rref)?;
307            let secret = read_secret(&format!("value for {rref}: "))?;
308            let rec = ops.put(&parsed, secret)?;
309            if cli.json {
310                let mut v = record_json(ops, &parsed, &rec);
311                v["ok"] = true.into();
312                return Ok(json_out(&v));
313            }
314            eprintln!("stored {rref} (v{})", rec.version);
315            Ok(ExitCode::SUCCESS)
316        }
317        Cmd::Get { rref } => {
318            let parsed = parse_ref(rref)?;
319            match ops.get(&parsed)? {
320                Some(rec) => {
321                    if cli.json {
322                        let mut v = record_json(ops, &parsed, &rec);
323                        v["value"] = rec.value.expose().into();
324                        return Ok(json_out(&v));
325                    }
326                    if std::io::stdout().is_terminal() {
327                        eprintln!("note: secret value printed to terminal");
328                    }
329                    println!("{}", rec.value.expose());
330                    Ok(ExitCode::SUCCESS)
331                }
332                None => Err(OpsError::Msg(format!(
333                    "{rref} is not stored in env {}",
334                    ops.resolve(&parsed).0
335                ))),
336            }
337        }
338        Cmd::List => {
339            let (env, refs) = ops.list(None)?;
340            ops.emit(Op::List, None, Outcome::Ok);
341            if cli.json {
342                let (_, records) = ops.list_records(None)?;
343                let arr: Vec<serde_json::Value> = records
344                    .iter()
345                    .map(|(r, rec)| {
346                        serde_json::json!({
347                            "ref": r.to_string(),
348                            "env": env.as_str(),
349                            "version": rec.version,
350                            "created_at": rec.created_at.to_string(),
351                            "rotated_at": rec.rotated_at.map(|t| t.to_string()),
352                        })
353                    })
354                    .collect();
355                return Ok(json_out(&serde_json::Value::Array(arr)));
356            }
357            for r in refs {
358                println!("{r}");
359            }
360            Ok(ExitCode::SUCCESS)
361        }
362        Cmd::Rm { rref } => {
363            let parsed = parse_ref(rref)?;
364            let (env, _) = ops.resolve(&parsed);
365            if !ops.rm(&parsed)? {
366                return Err(OpsError::Msg(format!("{rref} is not stored in env {env}")));
367            }
368            if cli.json {
369                return Ok(json_out(&serde_json::json!({
370                    "ok": true,
371                    "removed": true,
372                    "ref": parsed.to_string(),
373                    "env": env.as_str(),
374                })));
375            }
376            eprintln!("removed {rref}");
377            Ok(ExitCode::SUCCESS)
378        }
379        Cmd::Rotate { rref } => {
380            let parsed = parse_ref(rref)?;
381            let secret = read_secret(&format!("new value for {rref}: "))?;
382            let rec = ops.rotate(&parsed, secret)?;
383            if cli.json {
384                let mut v = record_json(ops, &parsed, &rec);
385                v["ok"] = true.into();
386                return Ok(json_out(&v));
387            }
388            eprintln!("rotated {rref} (v{})", rec.version);
389            Ok(ExitCode::SUCCESS)
390        }
391        Cmd::Run { secrets, cmd } => {
392            let vars = ops.resolve_for_injection(Op::Run, &parse_pairs(secrets)?)?;
393            let exposed: Vec<(String, String)> = vars
394                .iter()
395                .map(|(k, v)| (k.clone(), v.expose().to_string()))
396                .collect();
397            let cwd = std::env::current_dir()?;
398            // One arg → `bash -c` (shell syntax); several → exec
399            // argv[0] directly, no shell quoting.
400            let status = if cmd.len() == 1 {
401                arca_shell::run_with_env(&cwd, &cmd[0], &exposed)?
402            } else {
403                arca_shell::exec_with_env(&cwd, cmd, &exposed)?
404            };
405            Ok(ExitCode::from(status.code().unwrap_or(1) as u8))
406        }
407        Cmd::Env { secrets } => {
408            let vars = ops.resolve_for_injection(Op::Env, &parse_pairs(secrets)?)?;
409            if cli.json {
410                let map: BTreeMap<&String, &str> =
411                    vars.iter().map(|(k, v)| (k, v.expose())).collect();
412                return Ok(json_out(&serde_json::json!(map)));
413            }
414            let stdout = std::io::stdout();
415            let mut w = stdout.lock();
416            for (k, v) in &vars {
417                writeln!(w, "export {k}={}", sh_escape(v.expose()))?;
418            }
419            Ok(ExitCode::SUCCESS)
420        }
421        Cmd::Check => {
422            let report = ops.check();
423            ops.emit(Op::Check, None, Outcome::Ok);
424            let ok = report.problems.is_empty();
425            if cli.json {
426                let secrets: serde_json::Map<String, serde_json::Value> = report
427                    .secrets
428                    .iter()
429                    .map(|s| {
430                        (
431                            s.var.clone(),
432                            serde_json::json!({
433                                "ref": s.rref,
434                                "env": s.env,
435                                "resolves": s.resolves,
436                            }),
437                        )
438                    })
439                    .collect();
440                let vaults: Vec<serde_json::Value> = report
441                    .vaults
442                    .iter()
443                    .map(|v| {
444                        serde_json::json!({
445                            "name": v.name,
446                            "kind": v.kind,
447                            "path": v.path,
448                            "ok": v.ok,
449                            "error": v.error,
450                        })
451                    })
452                    .collect();
453                let guardrails = report.guardrails.as_ref().map(|g| {
454                    serde_json::json!({
455                        "default": match g.default {
456                            arca_core::GuardDefault::Allow => "allow",
457                            arca_core::GuardDefault::Deny => "deny",
458                        },
459                        "allow": g.allow.iter().map(|o| o.as_str()).collect::<Vec<_>>(),
460                        "deny": g.deny.iter().map(|o| o.as_str()).collect::<Vec<_>>(),
461                    })
462                });
463                let code = json_out(&serde_json::json!({
464                    "config": report.config_path,
465                    "env": report.env.as_str(),
466                    "vault": report.vault,
467                    "vaults": vaults,
468                    "secrets": secrets,
469                    "guardrails": guardrails,
470                    "errors": report.problems,
471                }));
472                debug_assert_eq!(code, ExitCode::SUCCESS);
473            } else {
474                match &report.config_path {
475                    Some(p) => eprintln!("config: {}", p.display()),
476                    None => eprintln!("config: (none found — defaults)"),
477                }
478                eprintln!("env: {} → vault {:?}", report.env, report.vault,);
479                for v in &report.vaults {
480                    match &v.error {
481                        None => eprintln!("vault {:?} ({}): readable", v.name, v.kind),
482                        Some(e) => eprintln!("vault {:?} ({}): {e}", v.name, v.kind),
483                    }
484                }
485                for s in &report.secrets {
486                    eprintln!(
487                        "secret {} → {} (env {}): {}",
488                        s.var,
489                        s.rref,
490                        s.env,
491                        if s.resolves { "resolves" } else { "NOT STORED" }
492                    );
493                }
494                if let Some(g) = &report.guardrails {
495                    let deny: Vec<_> = g.deny.iter().map(|o| o.as_str()).collect();
496                    eprintln!("guardrails: configured (deny: {})", deny.join(", "));
497                }
498                if report.problems.is_empty() {
499                    eprintln!("ok");
500                } else {
501                    for p in &report.problems {
502                        eprintln!("problem: {p}");
503                    }
504                }
505            }
506            Ok(if ok {
507                ExitCode::SUCCESS
508            } else {
509                ExitCode::FAILURE
510            })
511        }
512        Cmd::Deploy { rref, target } => {
513            let parsed = parse_ref(rref)?;
514            let report = ops.deploy(&parsed, target.as_deref())?;
515            if cli.json {
516                return Ok(json_out(&serde_json::json!({
517                    "ok": true,
518                    "ref": report.rref.to_string(),
519                    "env": report.env.as_str(),
520                    "target": report.target,
521                    "secret_name": report.secret_name,
522                    "created": report.created,
523                })));
524            }
525            eprintln!(
526                "deployed {} → {:?} as {} ({})",
527                report.rref,
528                report.target,
529                report.secret_name,
530                if report.created { "created" } else { "updated" }
531            );
532            Ok(ExitCode::SUCCESS)
533        }
534        Cmd::Ui => {
535            #[cfg(feature = "tui")]
536            {
537                arca_tui::run(ops)?;
538                ops.emit(Op::Ui, None, Outcome::Ok);
539                Ok(ExitCode::SUCCESS)
540            }
541            #[cfg(not(feature = "tui"))]
542            {
543                eprintln!("`arca ui` requires the `tui` feature (on by default)");
544                Ok(ExitCode::FAILURE)
545            }
546        }
547        Cmd::Manifest | Cmd::Schema | Cmd::GenCliDocs => {
548            unreachable!("handled before ops load")
549        }
550    }
551}

553fn parse_ref(s: &str) -> Result<Ref, OpsError> {
554    Ok(s.parse::<Ref>()?)
555}

557/// `-s VAR=arca://ref` pairs.
558fn parse_pairs(raw: &[String]) -> Result<Vec<(String, String)>, OpsError> {
559    raw.iter()
560        .map(|s| {
561            s.split_once('=')
562                .map(|(k, v)| (k.to_string(), v.to_string()))
563                .ok_or_else(|| OpsError::Msg(format!("-s wants VAR=arca://ref, got {s:?}")))
564        })
565        .collect()
566}

568/// Secrets arrive on stdin (the scripted path) or a hidden
569/// `/dev/tty` prompt (the human path) — never args. One trailing
570/// line terminator is stripped; the rest is the value verbatim
571/// (newlines allowed on the pipe path).
572fn read_secret(prompt: &str) -> Result<Secret, OpsError> {
573    let stdin = std::io::stdin();
574    let raw = if stdin.is_terminal() {
575        rpassword::prompt_password(prompt)
576            .map_err(|e| OpsError::Msg(format!("read secret: {e}")))?
577    } else {
578        let mut buf = String::new();
579        stdin
580            .lock()
581            .read_to_string(&mut buf)
582            .map_err(|e| OpsError::Msg(format!("read stdin: {e}")))?;
583        buf
584    };
585    let v = raw
586        .strip_suffix("\r\n")
587        .or_else(|| raw.strip_suffix('\n'))
588        .unwrap_or(&raw);
589    if v.is_empty() {
590        return Err(OpsError::Msg("empty secret".into()));
591    }
592    Ok(Secret::new(v))
593}

595/// Shell single-quote escape for `arca env` output — `'` becomes
596/// `'\''`, so any value is safe inside the quotes.
597fn sh_escape(v: &str) -> String {
598    format!("'{}'", v.replace('\'', "'\\''"))
599}

601/// The canonical record shape: resolved ref + env plus record
602/// metadata. Callers add `value`/`ok` as needed — never by
603/// default.
604fn record_json(ops: &Ops, parsed: &Ref, rec: &arca_vault::SecretRecord) -> serde_json::Value {
605    let (env, r) = ops.resolve(parsed);
606    serde_json::json!({
607        "ref": r.to_string(),
608        "env": env.as_str(),
609        "version": rec.version,
610        "created_at": rec.created_at.to_string(),
611        "rotated_at": rec.rotated_at.map(|t| t.to_string()),
612    })
613}

615/// The machine-readable capability doc — verbs, JSON shapes, env
616/// vars, the audit contract. Keep honest: this is what the binary
617/// actually does, not what it plans to.
618fn manifest() -> serde_json::Value {
619    serde_json::json!({
620        "name": "arca",
621        "version": env!("CARGO_PKG_VERSION"),
622        "description": "secrets broker for agents — arca:// refs in code/config, env-var injection at run time, ndjson audit on stderr",

708fn out(s: &str) -> ExitCode {
709    println!("{s}");
710    ExitCode::SUCCESS
711}

713/// Serializing a `serde_json::Value` cannot fail — the expect
714/// documents that invariant rather than guarding a real path.
715#[allow(clippy::unwrap_used)]
716fn json_out(v: &serde_json::Value) -> ExitCode {
717    out(&serde_json::to_string_pretty(v).expect("Value serializes"))
718}
719
720/// The `arca schema` payload — generated schema serializes
721/// infallibly, same invariant as [`json_out`].
722#[allow(clippy::unwrap_used)]
723fn schema_json() -> String {
724    serde_json::to_string_pretty(&schemars::schema_for!(arca_core::Config))
725        .expect("generated schema serializes")
726}
727
728/// Error path honoring `--json`: `{"error": …}` on stderr for
729/// machines, `arca: …` for humans.
730fn fail(cli: &Cli, msg: &str) -> ExitCode {
731    if cli.json {
732        eprintln!("{}", serde_json::json!({ "error": msg }));
733    } else {
734        eprintln!("arca: {msg}");
735    }
736    ExitCode::FAILURE
737}
