1//! Sealed packs — tamias ADR-0004 / org P-027 / PL-017 task 3.
2//!
3//! Tamias carries age-encrypted payloads **opaquely**: the engine,
4//! store, cache, registry, and logs digest ciphertext and never see
5//! plaintext. This module is the argv-verb side — the verbs a
6//! `type: secrets` (or any `sealed:`-carrying) pack's driver answers:
7//!
8//! - `seal` — author-side: plaintext files → `sealed/` payload
9//!   layout (`sealed/recipients` + `sealed/<name>.age`), plus the
10//!   `sealed:` manifest block on stdout — or merged straight into a
11//!   `tamias.yaml` with `--manifest <path>` (surgical text merge:
12//!   comments and unrelated keys survive; the result is validated
13//!   as a manifest before it lands).
14//! - `open <pack> <name>` / `open cuora://<pack>/<name>` — decrypt
15//!   one logical secret to stdout or `--out` (0600). The escape
16//!   hatch, like `cuora get`. `<pack>` may pin a version —
17//!   `cuora://<pack>@<version>/<name>` or `<pack@version> <name>` —
18//!   the tamias `name@version` label spelling, which is how
19//!   multi-install ambiguity gets resolved.
20//! - `materialize <pack> --out <dir>` — decrypt every declared
21//!   secret into a fresh dir (0600 files).
22//! - `recipients <pack>` — list the pack's age recipients (public
23//!   keys) and mark which the local keyfile can open.
24//! - `packs` — list installed sealed packs in the tamias store.
25//! - `install`/`deploy`/`uninstall` — lifecycle no-ops: a sealed
26//!   pack needs no driver action at install time (tamias dispatches
27//!   them because `capabilities.types` claims `secrets`).
28//!
29//! All pack/store/identity/age machinery lives in `cuora-sealed` —
30//! the same crate `cuora-ops` uses for the engine's
31//! `cuora://<pack>/<secret>` ref fallback, so `cuora get`/`run`/
32//! `env` and these verbs share one resolution + decrypt path.
33//!
34//! **Secrets safety:** plaintext exists only inside decrypt buffers
35//! (`Zeroizing`) between read and write; nothing here logs values —
36//! reports carry names, paths, and counts only. Every successful
37//! open/materialize emits a `cuora` audit event (`Op::Get`) so the
38//! materialization trail survives outside the vault path too.
39//!
40//! **Identity:** opening needs an age private key — the same
41//! keyfile `cuora init` manages (`~/.config/cuora/identity.txt`,
42//! mode 0600, `$CUORA_IDENTITY_FILE` override). `--keyfile` points
43//! at another file in the same format. There is no passphrase
44//! prompt: sealed packs are recipient-key only.
45//!
46//! **Store resolution:** `<pack>` resolves against the tamias
47//! install store — `$TAMIAS_HOME/store`, `$XDG_DATA_HOME/tamias/
48//! store`, `~/.local/share/tamias/store`, `./.tamias/store`
49//! (`--store` overrides). `--dir <payload>` (what `tamias driver
50//! exec`/install dispatch passes) skips the store entirely.

64/// Flags every sealed verb understands.
65#[derive(Default)]
66struct Flags {
67    /// `--out <path>` (`--out=…` too).
68    out: Option<PathBuf>,
69    /// `--in <dir>` — seal's plaintext source dir.
70    input_dir: Option<PathBuf>,
71    /// `--dir <payload>` — payload dir directly (driver dispatch).
72    dir: Option<PathBuf>,
73    /// `--store <dir>` — tamias store root override.
74    store: Option<PathBuf>,
75    /// `--keyfile <path>` — age identity keyfile override.
76    keyfile: Option<PathBuf>,
77    /// `--name <label>` — informational (dispatch passes the pkg name).
78    name: Option<String>,
79    /// `--recipient <age1…>` — repeatable public recipients.
80    recipients: Vec<String>,
81    /// `--recipients <file>` — repeatable recipients files.
82    recipient_files: Vec<PathBuf>,
83    /// `--manifest <path>` — merge the printed `sealed:` block into
84    /// this tamias.yaml instead of leaving it paste-by-hand.
85    manifest: Option<PathBuf>,
86}

88/// Parse `--flag value` / `--flag=value`; unknown `--x` is an error,
89/// everything else is positional.
90fn parse_flags(args: &[String], known: &[&str]) -> Result<(Flags, Vec<String>), String> {
91    let mut f = Flags::default();
92    let mut pos = Vec::new();
93    let mut i = 0;
94    while i < args.len() {
95        let a = &args[i];
96        let (flag, inline) = match a.split_once('=') {
97            Some((k, v)) if k.starts_with("--") => (k, Some(v.to_string())),
98            _ if a.starts_with("--") => (a.as_str(), None),
99            _ => {
100                pos.push(a.clone());
101                i += 1;
102                continue;
103            }
104        };
105        if !known.contains(&flag) {
106            return Err(format!("unknown flag {flag}"));
107        }
108        let value = match inline {
109            Some(v) => v,
110            None => {
111                i += 1;
112                args.get(i)
113                    .cloned()
114                    .ok_or_else(|| format!("{flag} needs a value"))?
115            }
116        };
117        match flag {
118            "--out" => f.out = Some(PathBuf::from(value)),
119            "--in" => f.input_dir = Some(PathBuf::from(value)),
120            "--dir" => f.dir = Some(PathBuf::from(value)),
121            "--store" => f.store = Some(PathBuf::from(value)),
122            "--keyfile" => f.keyfile = Some(PathBuf::from(value)),
123            "--name" => f.name = Some(value),
124            "--recipient" => f.recipients.push(value),
125            "--recipients" => f.recipient_files.push(PathBuf::from(value)),
126            "--manifest" => f.manifest = Some(PathBuf::from(value)),
127            _ => unreachable!("flag checked above"),
128        }
129        i += 1;
130    }
131    Ok((f, pos))
132}

134/// `--dir` mode: the payload dir is handed to us (driver dispatch).
135/// Otherwise resolve `<pack>` (+ optional secret) from the store.
136fn resolve_target(
137    f: &Flags,
138    pack_spec: Option<&str>,
139    positional_secret: Option<&str>,
140) -> Result<(Target, Option<String>), String> {
141    let target = cuora_sealed::resolve_target(

146    )?;
147    Ok((target, positional_secret.map(str::to_string)))

152/// One audit event per materialized secret — `Op::Get` against a
153/// `cuora://<pack>/<name>` ref, env `sealed` (packs aren't
154/// env-routed; the tag keeps the stream honest).
155fn audit_open(target: &Target, name: &str, outcome: Outcome) {
156    let rref = format!("cuora://{}/{name}", target.pack);
157    audit::emit(&AuditEvent::op(
158        &cuora_sealed::actor(),
159        Op::Get,
160        Some(&rref),
161        "sealed",
162        outcome,
163    ));
164    audit::flush();
165}

181/// `seal` — plaintext files → `sealed/` payload + the manifest block.
182fn seal(args: &[String]) -> i32 {
183    match seal_inner(args) {
184        Ok(code) => code,
185        Err(e) => {
186            eprintln!("cuora seal: {e}");
187            1
188        }
189    }
190}
191
192fn seal_inner(args: &[String]) -> Result<i32, String> {
193    let (f, pos) = parse_flags(args, SEAL_FLAGS)?;
194    let out = f.out.clone().ok_or("seal needs --out <payload-dir>")?;
195
196    // Collect (logical name, plaintext path) pairs.
197    let mut secrets: Vec<(String, PathBuf)> = Vec::new();
198    if let Some(dir) = &f.input_dir {
199        let read = std::fs::read_dir(dir).map_err(|e| format!("read {}: {e}", dir.display()))?;
200        for entry in read {
201            let entry = entry.map_err(|e| format!("read {}: {e}", dir.display()))?;
202            let path = entry.path();
203            let ft = entry
204                .file_type()
205                .map_err(|e| format!("stat {}: {e}", path.display()))?;
206            if ft.is_dir() {
207                return Err(format!(
208                    "{} is a directory — seal takes files; nested layouts use NAME=PATH pairs",
209                    path.display()
210                ));
211            }
212            if !ft.is_file() {
213                continue; // symlinks/fifos — not sealable payload
214            }
215            let fname = entry.file_name().to_string_lossy().into_owned();
216            // Logical name = file stem (`token.txt` → `token`);
217            // a bare `token` keeps its name.
218            let stem = Path::new(&fname)
219                .file_stem()
220                .map(|s| s.to_string_lossy().into_owned())
221                .unwrap_or(fname);
222            secrets.push((stem, path));
223        }
224        secrets.sort_by(|a, b| a.0.cmp(&b.0));
225    }
226    for p in &pos {
227        let Some((name, path)) = p.split_once('=') else {
228            return Err(format!(
229                "seal operands are NAME=PATH pairs (got '{p}') — or use --in <dir>"
230            ));
231        };
232        secrets.push((name.to_string(), PathBuf::from(path)));
233    }
234    if secrets.is_empty() {
235        return Err("seal needs at least one secret — --in <dir> or NAME=PATH pairs".into());
236    }
237    let mut seen = std::collections::BTreeSet::new();
238    for (name, path) in &secrets {
239        if !is_valid_name(name) {
240            return Err(format!("invalid sealed secret name '{name}'"));
241        }
242        if !seen.insert(name.as_str()) {
243            return Err(format!("duplicate sealed secret name '{name}'"));
244        }
245        if !path.is_file() {
246            return Err(format!("{} is not a file", path.display()));
247        }
248    }
249
250    // Recipients: --recipient args, --recipients files, --keyfile pub.
251    let mut recipients: Vec<String> = f.recipients.clone();
252    for rf in &f.recipient_files {
253        let text =
254            std::fs::read_to_string(rf).map_err(|e| format!("read {}: {e}", rf.display()))?;
255        recipients.extend(cuora_sealed::parse_recipient_lines(&text)?);
256    }
257    if let Some(kf) = &f.keyfile {
258        let id = cuora_sealed::load_identity(Some(kf))?;
259        recipients.push(id.to_public().to_string());
260    }
261    recipients.sort();
262    recipients.dedup();
263    if recipients.is_empty() {
264        return Err(
265            "seal needs at least one recipient — --recipient age1…, --recipients <file>, or --keyfile"
266                .into(),
267        );
268    }
269    let recips: Vec<x25519::Recipient> = recipients
270        .iter()
271        .map(|r| {
272            x25519::Recipient::from_str(r).map_err(|e| format!("invalid recipient {r:?}: {e}"))
273        })
274        .collect::<Result<_, _>>()?;
275
276    // Encrypt + lay out `sealed/` inside --out.
277    let sealed_dir = out.join("sealed");
278    std::fs::create_dir_all(&sealed_dir)
279        .map_err(|e| format!("create {}: {e}", sealed_dir.display()))?;
280    let mut files = BTreeMap::new();
281    for (name, path) in &secrets {
282        let plain = std::fs::read(path).map_err(|e| format!("read {}: {e}", path.display()))?;
283        let cipher = cuora_sealed::encrypt(&plain, &recips)?;
284        let rel = format!("sealed/{name}.age");
285        let dest = out.join(&rel);
286        if let Some(parent) = dest.parent() {
287            std::fs::create_dir_all(parent)
288                .map_err(|e| format!("create {}: {e}", parent.display()))?;
289        }
290        std::fs::write(&dest, &cipher).map_err(|e| format!("write {}: {e}", dest.display()))?;
291        files.insert(name.clone(), rel);
292    }
293    let recipients_body = format!("{}\n", recipients.join("\n"));
294    std::fs::write(sealed_dir.join("recipients"), recipients_body)
295        .map_err(|e| format!("write recipients: {e}"))?;
296
297    // The manifest block — stdout is the machine-usable half.
298    let mut block =
299        String::from("sealed:\n  format: age/v1\n  recipients: sealed/recipients\n  files:\n");
300    for (name, rel) in &files {
301        block.push_str(&format!("    {name}: {rel}\n"));
302    }
303    print!("{block}");
304    if let Some(manifest) = &f.manifest {
305        merge_sealed_block(manifest, &block)?;
306        eprintln!("merged the sealed: block into {}", manifest.display());
307    } else if out.join("tamias.yaml").exists() {
308        eprintln!(
309            "hint: {} exists — re-run with --manifest to merge the block",
310            out.join("tamias.yaml").display()
311        );
312    }
313    eprintln!(
314        "sealed {} secret(s) → {} ({} recipient(s)) — merge the block above into tamias.yaml, then `tamias pack`",
315        secrets.len(),
316        out.display(),
317        recipients.len()
318    );
319    Ok(0)
320}

324/// Merge the generated `sealed:` block into an existing
325/// `tamias.yaml`, preserving everything outside the block — comments,
326/// key order, quoting, unrelated sections all survive byte-for-byte.
327///
328/// This is a *surgical text* merge, not a YAML round-trip: serde_yaml
329/// would drop comments and reflow the file, which is exactly what an
330/// author hand-merging wants to avoid. The trade-off is that block
331/// detection is line-oriented — a top-level `sealed:` key and the
332/// indented lines belonging to it. Anything weirder (a `sealed:` key
333/// hidden inside a literal block scalar is impossible at column 0;
334/// a column-0 comment *inside* the sealed mapping truncates the
335/// replaced region) either merges correctly or fails the post-merge
336/// [`Manifest::load_at`] validation with the file left untouched.
337///
338/// The merge is validated before it lands: the result is written to
339/// a sibling temp file, parsed as a tamias manifest, and only then
340/// renamed over the original (permissions carried over).
341fn merge_sealed_block(path: &Path, block: &str) -> Result<(), String> {
342    let text =
343        std::fs::read_to_string(path).map_err(|e| format!("read {}: {e}", path.display()))?;
344    // The target must already be a valid package manifest — merging
345    // into a project/deps-only tamias.yaml is out of scope.
346    Manifest::load_at(path).map_err(|e| {
347        format!(
348            "{} is not a valid tamias package manifest ({e}) — --manifest merges into a pack's tamias.yaml",
349            path.display()
350        )
351    })?;
352    let merged = merge_sealed_text(&text, block);
353    commit_manifest(path, &merged)
354}
355
356/// The pure half of [`merge_sealed_block`]: splice `block` (a
357/// `sealed:\n…` YAML fragment) into `text` at the top-level `sealed:`
358/// key, replacing its old block, or appending at EOF when absent.
359fn merge_sealed_text(text: &str, block: &str) -> String {
360    let lines: Vec<&str> = text.lines().collect();
361    // Locate the top-level `sealed:` key — column 0, `key:` followed
362    // by end-of-line, whitespace, or a comment.
363    let key_at = lines.iter().position(|l| {
364        l.strip_prefix("sealed:").is_some_and(|rest| {
365            rest.is_empty()
366                || rest.starts_with(' ')
367                || rest.starts_with('\t')
368                || rest.starts_with('#')
369        })
370    });
371    let block_lines: Vec<&str> = block.lines().collect();
372    let mut out: Vec<&str> = Vec::with_capacity(lines.len() + block_lines.len() + 2);
373    match key_at {
374        Some(i) => {
375            out.extend_from_slice(&lines[..i]);
376            out.extend(block_lines.iter().copied());
377            // The old block runs while lines stay inside the mapping:
378            // blank or indented. A column-0 line (next key, comment,
379            // `---`/`...`) ends it — a column-0 comment *inside* the
380            // mapping truncates the splice and the post-merge
381            // validation reports it rather than eating the comment.
382            let mut j = i + 1;
383            while j < lines.len() {
384                let l = lines[j];
385                if l.trim().is_empty() || l.starts_with(' ') || l.starts_with('\t') {
386                    j += 1;
387                } else {
388                    break;
389                }
390            }
391            out.extend_from_slice(&lines[j..]);
392        }
393        None => {
394            out.extend_from_slice(&lines);
395            // Separate the appended block from a non-empty tail.
396            if out.last().is_some_and(|l| !l.trim().is_empty()) {
397                out.push("");
398            }
399            out.extend(block_lines.iter().copied());
400        }
401    }
402    let mut s = out.join("\n");
403    s.push('\n');
404    s
405}

407/// Validate `merged` as a tamias manifest, then rename it over
408/// `path`. The temp file keeps the `.yaml` extension so
409/// [`Manifest::load_at`] picks the YAML parser, and inherits the
410/// original's permissions before the rename.
411fn commit_manifest(path: &Path, merged: &str) -> Result<(), String> {
412    let file_name = path
413        .file_name()
414        .and_then(|n| n.to_str())
415        .ok_or_else(|| format!("{} has no usable file name", path.display()))?;
416    let tmp = path.with_file_name(format!("{file_name}.cuora-merge.yaml"));
417    let write = |p: &Path| -> Result<(), String> {
418        std::fs::write(p, merged).map_err(|e| format!("write {}: {e}", p.display()))
419    };
420    write(&tmp)?;
421    if let Ok(meta) = std::fs::metadata(path) {
422        let _ = std::fs::set_permissions(&tmp, meta.permissions());
423    }
424    let result = Manifest::load_at(&tmp).map_err(|e| {
425        format!(
426            "merged manifest failed validation ({e}) — {} left untouched",
427            path.display()
428        )
429    });
430    match result {
431        Ok(_) => {
432            std::fs::rename(&tmp, path)
433                .map_err(|e| format!("rename {} → {}: {e}", tmp.display(), path.display()))?;
434            Ok(())
435        }
436        Err(e) => {
437            let _ = std::fs::remove_file(&tmp);
438            Err(e)
439        }
440    }
441}

450/// `open` — decrypt one logical secret. `open <pack> <name>`,
451/// `open cuora://<pack>/<name>`, or dispatch form
452/// `open --dir <payload> --name <pkg> <name>`.
453fn open(args: &[String]) -> i32 {
454    match open_inner(args) {
455        Ok(code) => code,
456        Err(e) => {
457            eprintln!("cuora open: {e}");
458            1
459        }
460    }
461}
462
463fn open_inner(args: &[String]) -> Result<i32, String> {
464    let (f, pos) = parse_flags(args, OPEN_FLAGS)?;
465    // Forms: `cuora://p/s` alone; `<pack> <name>`; `--dir` + `<name>`.
466    let (pack_spec, secret) = if f.dir.is_some() {
467        // Dispatch form — the payload dir is explicit, so exactly
468        // one positional: the secret name.
469        match pos.as_slice() {
470            [name] => (String::new(), name.clone()),
471            _ => return Err("open with --dir takes exactly one operand: the secret name".into()),
472        }
473    } else if pos.len() == 1 {
474        if pos[0].starts_with("cuora://") || pos[0].starts_with("arca://") {
475            cuora_sealed::parse_pack_ref(&pos[0])?
476        } else {
477            return Err(format!(
478                "open needs <pack> <name> or cuora://<pack>/<name> (got '{}')",
479                pos[0]
480            ));
481        }
482    } else if pos.len() == 2 {
483        (pos[0].clone(), pos[1].clone())
484    } else {
485        return Err(
486            "open needs <pack> <name>, cuora://<pack>/<name>, or --dir <payload> <name>".into(),
487        );
488    };
489    let pack_spec = if pack_spec.is_empty() {
490        None
491    } else {
492        Some(pack_spec.as_str())
493    };
494    let (target, secret) = resolve_target(&f, pack_spec, Some(&secret))?;
495    let secret = secret.ok_or("open needs a secret name")?;
496
497    let rel = target
498        .sealed
499        .files
500        .get(&secret)
501        .ok_or_else(|| format!("'{}' declares no sealed secret '{secret}'", target.label))?;
502    let cipher_path = target.dir.join(rel);
503    let cipher =
504        std::fs::read(&cipher_path).map_err(|e| format!("read {}: {e}", cipher_path.display()))?;
505    let identity = cuora_sealed::load_identity(f.keyfile.as_deref())?;
506    let plain = cuora_sealed::decrypt(&identity, &cipher)?;
507
508    match &f.out {
509        Some(out) => {
510            cuora_sealed::write_secret_file(out, &plain)?;
511            eprintln!("opened {secret} → {}", out.display());
512        }
513        None => {
514            let mut stdout = std::io::stdout().lock();
515            stdout
516                .write_all(&plain)
517                .map_err(|e| format!("write stdout: {e}"))?;
518            if !plain.ends_with(b"\n") {
519                let _ = stdout.write_all(b"\n");
520            }
521        }
522    }
523    audit_open(&target, &secret, Outcome::Ok);
524    Ok(0)
525}

527/// `materialize` — decrypt every declared secret into `--out`.
528fn materialize(args: &[String]) -> i32 {
529    match materialize_inner(args) {
530        Ok(code) => code,
531        Err(e) => {
532            eprintln!("cuora materialize: {e}");
533            1
534        }
535    }
536}
537
538fn materialize_inner(args: &[String]) -> Result<i32, String> {
539    let (f, pos) = parse_flags(args, MATERIALIZE_FLAGS)?;
540    let out = f.out.clone().ok_or("materialize needs --out <dir>")?;
541    let pack_spec = match pos.as_slice() {
542        [] if f.dir.is_some() => None,
543        [p] if f.dir.is_none() => Some(p.as_str()),
544        _ => return Err(
545            "materialize takes one pack spec (<pack>, @scope/name@ver, or --dir with no operands)"
546                .into(),
547        ),
548    };
549    // Refuse to write into a non-empty dir — materialization is
550    // all-or-nothing over a fresh target.
551    if out.exists() {
552        let mut it = std::fs::read_dir(&out).map_err(|e| format!("read {}: {e}", out.display()))?;
553        if it.next().is_some() {
554            return Err(format!(
555                "{} exists and is not empty — materialize wants a fresh dir",
556                out.display()
557            ));
558        }
559    }
560    let (target, _) = resolve_target(&f, pack_spec, None)?;
561    let identity = cuora_sealed::load_identity(f.keyfile.as_deref())?;
562
563    let mut wrote = 0usize;
564    for (name, rel) in &target.sealed.files {
565        let cipher_path = target.dir.join(rel);
566        let cipher = std::fs::read(&cipher_path)
567            .map_err(|e| format!("read {}: {e}", cipher_path.display()))?;
568        let plain = cuora_sealed::decrypt(&identity, &cipher)?;
569        // Logical names are manifest-validated ([A-Za-z0-9._-]) —
570        // flat files, no traversal possible.
571        cuora_sealed::write_secret_file(&out.join(name), &plain)?;
572        audit_open(&target, name, Outcome::Ok);
573        wrote += 1;
574    }
575    eprintln!("materialized {wrote} secret(s) → {}", out.display());
576    Ok(0)
577}

579/// `recipients` — the pack's public recipient set + local match.
580fn recipients(args: &[String]) -> i32 {
581    match recipients_inner(args) {
582        Ok(code) => code,
583        Err(e) => {
584            eprintln!("cuora recipients: {e}");
585            1
586        }
587    }
588}
589
590fn recipients_inner(args: &[String]) -> Result<i32, String> {
591    let (f, pos) = parse_flags(args, RECIPIENTS_FLAGS)?;
592    let pack_spec = match pos.as_slice() {
593        [] if f.dir.is_some() => None,
594        [p] if f.dir.is_none() => Some(p.as_str()),
595        _ => return Err("recipients takes one pack spec (or --dir with no operands)".into()),
596    };
597    let (target, _) = resolve_target(&f, pack_spec, None)?;
598    let recipients_path = target.dir.join(&target.sealed.recipients);
599    let text = std::fs::read_to_string(&recipients_path)
600        .map_err(|e| format!("read {}: {e}", recipients_path.display()))?;
601    let list = cuora_sealed::parse_recipient_lines(&text)?;
602
603    // Local match is best-effort: no keyfile → still list publics.
604    let local = match cuora_sealed::load_identity(f.keyfile.as_deref()) {
605        Ok(id) => Some(id.to_public().to_string()),
606        Err(_) => None,
607    };
608    println!("pack:      {}", target.label);
609    println!("recipients file: {}", target.sealed.recipients);
610    for r in &list {
611        let mark = if local.as_deref() == Some(r.as_str()) {
612            "  ← local keyfile"
613        } else {
614            ""
615        };
616        println!("{r}{mark}");
617    }
618    if list.is_empty() {
619        eprintln!("warning: recipients file holds no keys");
620    }
621    if local.is_none() {
622        eprintln!("note: no local identity — cannot test which recipients this host opens");
623    }
624    Ok(0)
625}
626
627/// `packs` — installed sealed packs in the store (names only).
628fn packs(args: &[String]) -> i32 {
629    match packs_inner(args) {
630        Ok(code) => code,
631        Err(e) => {
632            eprintln!("cuora packs: {e}");
633            1
634        }
635    }
636}
637
638fn packs_inner(args: &[String]) -> Result<i32, String> {
639    let (f, pos) = parse_flags(args, PACKS_FLAGS)?;
640    if !pos.is_empty() {
641        return Err("packs takes no operands".into());
642    }
643    let root = cuora_sealed::store_root(f.store.as_deref());
644    let mut found = 0usize;
645    for e in cuora_sealed::scan_store(&root) {
646        let Ok(m) = e.manifest() else { continue };
647        let Some(sealed) = &m.sealed else { continue };
648        let n = sealed.files.len();
649        println!(
650            "{}\t{}\t{} secret(s)\t{}",
651            e.label(),
652            m.name,
653            n,
654            e.dir().display()
655        );
656        found += 1;
657    }
658    if found == 0 {
659        eprintln!("no sealed packs installed under {}", root.display());
660    }
661    Ok(0)
662}

664// ------------------------------------------------------------- lifecycle
665
666/// `install`/`deploy`/`uninstall` — dispatched by tamias for
667/// `type: secrets` packs (the driver's `capabilities.types` claims
668/// the kind). Sealed packs need no lifecycle action: ciphertext
669/// installs verbatim and opens consumer-side. Anything *not* sealed
670/// fails loudly — this driver only owns the sealed contract.
671fn lifecycle(verb: &str, args: &[String]) -> i32 {
672    // Tamias appends `--dir <payload> --name <pkg> [--config …]`;
673    // accept and ignore everything but --dir/--name.
674    let (f, _) = match parse_flags(args, &["--dir", "--name", "--config"]) {
675        Ok(v) => v,
676        Err(e) => {
677            eprintln!("cuora {verb}: {e}");
678            return 1;
679        }
680    };
681    let Some(dir) = f.dir else {
682        eprintln!("cuora {verb}: --dir <payload> is required (tamias dispatch provides it)");
683        return 1;
684    };
685    match Manifest::load(&dir) {
686        Ok(m) if m.sealed.is_some() => {
687            eprintln!(
688                "cuora {verb}: {} is a sealed pack — ciphertext only, nothing to {}",
689                f.name.as_deref().unwrap_or("pack"),
690                verb
691            );
692            0
693        }
694        Ok(_) => {
695            eprintln!(
696                "cuora {verb}: {} carries no sealed: block — the cuora driver only handles sealed packs",
697                dir.display()
698            );
699            1
700        }
701        Err(e) => {
702            eprintln!("cuora {verb}: {}", e);
703            1
704        }
705    }
706}
707
708/// Entry point from `main` — `args` is everything after the verb.
709pub fn run(verb: &str, args: &[String]) -> i32 {
710    match verb {
711        "seal" => seal(args),
712        "open" => open(args),
713        "materialize" => materialize(args),
714        "recipients" => recipients(args),
715        "packs" => packs(args),
716        "install" | "deploy" | "uninstall" => lifecycle(verb, args),
717        _ => {
718            eprintln!("unknown sealed verb '{verb}'");
719            2
720        }
721    }
722}

876    fn manifest_merge_text_append_and_replace() {
877        // Append: no sealed key — block lands at EOF, comments kept.
878        let src = "# pack header\nname: x\nversion: \"1.0.0\"\ntype: secrets\n";
879        let m = merge_sealed_text(src, BLOCK);
880        assert!(m.starts_with("# pack header\nname: x"));
881        assert!(m.contains("\ntype: secrets\n\nsealed:\n"));
882        assert!(m.trim_end().ends_with("token: sealed/token.age"));
883
884        // Replace: old block swapped, position + neighbors preserved.
885        let src = "name: x\nversion: \"1.0.0\"\n# keep me\ntype: secrets\n\nsealed:\n  format: age/v1\n  recipients: sealed/recipients\n  files:\n    old: sealed/old.age\n\nbins:\n  - name: b\n";
886        let m = merge_sealed_text(src, BLOCK);
887        assert!(m.contains("# keep me"));
888        assert!(m.contains("bins:\n  - name: b\n"));
889        assert!(!m.contains("old: sealed/old.age"));
890        assert!(m.contains("token: sealed/token.age"));
891        assert!(m.find("sealed:").unwrap() < m.find("bins:").unwrap());
892
893        // Inline `sealed: {}` is a one-line block — replaced whole.
894        let m = merge_sealed_text("name: x\nversion: \"1.0.0\"\nsealed: {}\n", BLOCK);
895        assert!(!m.contains("sealed: {}"));
896        assert!(m.contains("files:\n    token: sealed/token.age"));
897
898        // `sealed:` with a trailing comment still counts as the key.
899        let m = merge_sealed_text(
900            "name: x\nversion: \"1.0.0\"\nsealed: # ciphertext only\n  files: {}\n",
901            BLOCK,
902        );
903        assert!(m.contains("format: age/v1"));
904        assert!(!m.contains("ciphertext only"));
905    }
