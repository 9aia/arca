1//! The analytics emits — `format = "timeline"|"stats"|"burnup"`
2//! (P-012). Same pipeline as every view: the spec's
3//! `filter → group → sort → limit` resolves first; these formats
4//! replace the *group render* with a purpose-built emit over the
5//! filtered universe. Structured `--json` for pavo/agents,
6//! deterministic ASCII on tty.
7//!
8//! Universe rules differ per emit:
9//! - `timeline` — live top-level items (a roadmap is forward
10//!   work; archived history has no lanes).
11//! - `stats`/`burnup` — the whole ledger incl. DONE.md and
12//!   `archive/` (the archive *is* their data), children flattened
13//!   in — they're counts, not rows.
14//!
15//! `source = "index"` works identically: snapshots carry items
16//! and the archived-id set, so repo/index parity holds for all
17//! three emits.
18

27/// Does `spec.format` route to an analytics emit instead of the
28/// group renderers?
29pub(crate) fn is_emit(spec: &ViewSpec) -> bool {
30    matches!(
31        spec.format,
32        ViewFormat::Timeline | ViewFormat::Stats | ViewFormat::Burnup
33    )
34}

36/// The filtered, format-appropriate universe as [`Scoped`] items.
37/// `scan`/`views` are the repo/index universe owners — exactly
38/// one is populated per `spec.source`.
39pub(crate) fn emit_items<'a>(
40    spec: &ViewSpec,
41    scan: Option<&'a Scan>,
42    views: &'a [org::RepoView],
43    kinds: &KindTable,
44) -> Vec<Scoped<'a>> {
45    // stats/burnup read the archive; timeline is forward work only
46    let archived_ok = matches!(spec.format, ViewFormat::Stats | ViewFormat::Burnup);
47    let mut out = Vec::new();
48    match spec.source {
49        myco_core::ViewSource::Repo => {
50            if let Some(s) = scan {
51                for top in &s.items {
52                    if archived_ok {
53                        for i in top.subtree() {
54                            if spec.filter.matches(i, kinds) {
55                                out.push(Scoped::new(i));
56                            }
57                        }
58                    } else if !s.archived.contains(&top.id) && spec.filter.matches(top, kinds) {
59                        out.push(Scoped::new(top));
60                    }
61                }
62            }
63        }
64        myco_core::ViewSource::Index => {
65            for v in views {
66                if let Some(snap) = &v.snap {
67                    for top in &snap.items {
68                        if archived_ok {
69                            for i in top.subtree() {
70                                if spec.filter.matches(i, kinds) {
71                                    out.push(Scoped::tagged(snap.name.as_str(), i));
72                                }
73                            }
74                        } else if !snap.archived.contains(&top.id)
75                            && spec.filter.matches(top, kinds)
76                        {
77                            out.push(Scoped::tagged(snap.name.as_str(), top));
78                        }
79                    }
80                }
81            }
82        }
83    }
84    out
85}

87/// The `--json` payload for an analytics emit —
88/// `{view, spec, format, …}` where `…` is `lanes`, `stats`, or
89/// `{window, series}` per format, plus `warnings`.
90pub(crate) fn emit_json(
91    name: Option<&str>,
92    spec: &ViewSpec,
93    items: &[Scoped],
94    cycles: &[CycleConfig],
95    warnings: &[String],
96) -> serde_json::Value {
97    let mut payload = json!({
98        "view": name,
99        "spec": spec,
100        "warnings": warnings,
101    });
102    match spec.format {
103        ViewFormat::Timeline => {
104            payload["format"] = json!("timeline");
105            payload["lanes"] = json!(analytics::timeline(items.to_vec())
106                .iter()
107                .map(|l| json!({
108                    "kind": l.kind.name(),
109                    "name": l.name,
110                    "items": l.items,
111                }))
112                .collect::<Vec<_>>());
113        }
114        ViewFormat::Stats => {
115            payload["format"] = json!("stats");
116            payload["stats"] = stats_json(&analytics::stats(items));
117        }
118        ViewFormat::Burnup => {
119            let today = myco_ops::items::today();
120            let window = analytics::burnup_window(cycles, today);
121            payload["format"] = json!("burnup");
122            payload["window"] = json!({
123                "start": date::render(window.0),
124                "end": date::render(window.1),
125            });
126            payload["series"] = json!(analytics::burnup(items, cycles, window)
127                .iter()
128                .map(|s| json!({
129                    "cycle": s.cycle,
130                    "points": s.points.iter().map(|p| json!({
131                        "date": date::render(p.date),
132                        "done": p.done,
133                        "scope": p.scope,
134                    })).collect::<Vec<_>>(),
135                }))
136                .collect::<Vec<_>>());
137        }
138        _ => unreachable!("emit_json only serves analytics formats"),
139    }
140    payload
141}

143/// `Stats` → `{status, priority, owner, epic, milestone, cycle,
144/// label}` — each `[{name, count}]`.
145fn stats_json(s: &Stats) -> serde_json::Value {
146    let tallies = |v: &[analytics::Tally]| {
147        json!(v
148            .iter()
149            .map(|t| json!({"name": t.name, "count": t.count}))
150            .collect::<Vec<_>>())
151    };
152    json!({
153        "status": tallies(&s.status),
154        "priority": tallies(&s.priority),
155        "owner": tallies(&s.owner),
156        "epic": tallies(&s.epic),
157        "milestone": tallies(&s.milestone),
158        "cycle": tallies(&s.cycle),
159        "label": tallies(&s.label),
160    })
161}

163/// The tty render — deterministic ASCII; `--json` callers never
164/// reach here. Resolution `warnings` already printed to stderr
165/// by the caller (`view::warn`).
166pub(crate) fn render(spec: &ViewSpec, items: &[Scoped], cycles: &[CycleConfig]) -> String {
167    match spec.format {
168        ViewFormat::Timeline => render_timeline(&analytics::timeline(items.to_vec())),
169        ViewFormat::Stats => render_stats(&analytics::stats(items)),
170        ViewFormat::Burnup => {
171            let window = analytics::burnup_window(cycles, myco_ops::items::today());
172            render_burnup(&analytics::burnup(items, cycles, window))
173        }
174        _ => String::new(),
175    }
176}

178/// `kind:name (n)` lane headers + compact item lines — the same
179/// `items::compact` rendering `list` uses, so a roadmap reads
180/// like the ledger it projects.
181fn render_timeline(lanes: &[Lane]) -> String {
182    let mut out = String::new();
183    for l in lanes {
184        if l.items.is_empty() {
185            continue;
186        }
187        out.push_str(&format!(
188            "{}:{} ({})\n",
189            l.kind.name(),
190            l.name,
191            l.items.len()
192        ));
193        for s in &l.items {
194            out.push_str("  ");
195            if let Some(r) = s.repo {
196                out.push_str(r);
197                out.push(':');
198            }
199            out.push_str(&crate::items::compact(s.item));
200            out.push('\n');
201        }
202    }
203    out.trim_end_matches('\n').to_string()
204}

206/// ASCII bars — `name` left, count, then `#` scaled so the
207/// largest bucket fills 40 columns. Dimensions print only when
208/// non-empty; ungrouped trails inside each.
209fn render_stats(s: &Stats) -> String {
210    fn dim(out: &mut String, title: &str, tallies: &[analytics::Tally]) {
211        if tallies.is_empty() {
212            return;
213        }
214        out.push_str(&format!("{title}:\n"));
215        let max = tallies.iter().map(|t| t.count).max().unwrap_or(1).max(1);
216        let width = tallies.iter().map(|t| t.name.len()).max().unwrap_or(0);
217        for t in tallies {
218            let bar = "#".repeat(t.count * 40 / max);
219            out.push_str(&format!(
220                "  {:<width$} {:>4}  {}\n",
221                t.name,
222                t.count,
223                bar,
224                width = width
225            ));
226        }
227    }
228    let mut out = String::new();
229    dim(&mut out, "status", &s.status);
230    dim(&mut out, "priority", &s.priority);
231    dim(&mut out, "owner", &s.owner);
232    dim(&mut out, "epic", &s.epic);
233    dim(&mut out, "milestone", &s.milestone);
234    dim(&mut out, "cycle", &s.cycle);
235    dim(&mut out, "label", &s.label);
236    if !out.is_empty() {
237        out.push('\n');
238    }
239    out.trim_end_matches('\n').to_string()
240}

242/// Per-series: a header (`cycle <name> — <start> → <end>`), a
243/// daily sparkline over an ASCII ramp (position = done/scope
244/// fraction), and the closing `done/scope`. Unstamped archive
245/// counts toward scope but never lifts the line — the curve
246/// only moves on `done:` stamps.
247fn render_burnup(series: &[BurnupSeries]) -> String {
248    const RAMP: &[u8; 10] = b" .:-=+*#%@";
249    let mut out = String::new();
250    for s in series {
251        let label = s.cycle.unwrap_or("uncycled");
252        let (first, last) = match (s.points.first(), s.points.last()) {
253            (Some(f), Some(l)) => (f.date, l.date),
254            _ => continue,
255        };
256        let final_pt = s
257            .points
258            .last()
259            .copied()
260            .unwrap_or(myco_core::analytics::BurnupPoint {
261                date: last,
262                done: 0,
263                scope: 0,
264            });
265        out.push_str(&format!(
266            "{label} — {} → {}: {}/{} done\n",
267            date::render(first),
268            date::render(last),
269            final_pt.done,
270            final_pt.scope
271        ));
272        let line: String = s
273            .points
274            .iter()
275            .map(|p| {
276                let idx = p
277                    .done
278                    .checked_mul(9)
279                    .and_then(|n| n.checked_div(p.scope))
280                    .unwrap_or(0) as usize;
281                RAMP[idx.min(9)] as char
282            })
283            .collect();
284        out.push_str(&format!("  {line}\n"));
285    }
286    out.trim_end_matches('\n').to_string()
287}
