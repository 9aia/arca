1//! Protocol fakes on top of [`HttpMock`] — in-memory reference
2//! implementations of the two remote surfaces arca talks to:
3//!
4//! - [`vault_wire`] — the `arca-vault-cf/1` wire protocol, the same
5//!   semantics `arca-vault-cf/worker/index.js` implements against
6//!   D1. Stores base64 ciphertext rows only — never plaintext.
7//! - [`cf_secrets_store`] — the CF API v4 Secrets Store envelope
8//!   (`{success, errors, result}`) that `arca-deploy-cf` pushes to.
9//!   A real Secrets Store *is* a plaintext sink, so this fake
10//!   records what it receives (fake values only, by rule).
11
12use std::collections::BTreeMap;
13use std::sync::{Arc, Mutex};
14
15use serde_json::json;
16use time::format_description::well_known::Rfc3339;
17use time::OffsetDateTime;
18
19use crate::httpmock::{HttpMock, Request, Response};
20
21/// One stored vault row (ciphertext is opaque to the store).
22#[derive(Clone)]
23struct VaultRow {
24    ciphertext: String,
25    version: u64,
26    created_at: String,
27    rotated_at: Option<String>,
28}
29
30fn now() -> String {
31    // rfc3339 formatting of a valid datetime cannot fail
32    OffsetDateTime::now_utc()
33        .format(&Rfc3339)
34        .expect("rfc3339 formatting is infallible")
35}
36
37fn row_json(s: &VaultRow) -> String {
38    json!({
39        "version": s.version,
40        "created_at": s.created_at,
41        "rotated_at": s.rotated_at,
42    })
43    .to_string()
44}

46/// The `arca-vault-cf/1` wire protocol, in memory. Bearer `token`
47/// required on every request; per-env keyspace; server-side
48/// version/created_at/rotated_at bookkeeping (the client's rotate
49/// is a `PUT` with `"rotate": true`, 404 on a missing ref).
50pub fn vault_wire(token: &'static str) -> HttpMock {
51    let store: Arc<Mutex<BTreeMap<(String, String), VaultRow>>> =
52        Arc::new(Mutex::new(BTreeMap::new()));
53    HttpMock::new(move |req: &Request| {
54        if req.header("authorization") != Some(format!("Bearer {token}").as_str()) {
55            return Response::json(401, r#"{"error":"unauthorized"}"#);
56        }
57        let path = req.path.split('?').next().unwrap_or("");
58        let rest = match path.strip_prefix("/v1/") {
59            Some(r) => r,
60            None => return Response::json(404, r#"{"error":"no such route"}"#),
61        };
62        // /v1/{env} lists; /v1/{env}/{ref…} addresses one record.
63        let (env, rref) = match rest.split_once('/') {
64            Some((env, rref)) => (env.to_string(), Some(rref.to_string())),
65            None => (rest.to_string(), None),
66        };
67        if env.is_empty() {
68            return Response::json(404, r#"{"error":"no such route"}"#);
69        }
70        let mut store = store.lock().unwrap();
71        match (req.method.as_str(), rref) {
72            ("GET", None) => {
73                let refs: Vec<String> = store
74                    .keys()
75                    .filter(|(e, _)| e == &env)
76                    .map(|(_, r)| r.clone())
77                    .collect();
78                Response::json(200, json!({ "refs": refs }).to_string())
79            }
80            ("GET", Some(rref)) => match store.get(&(env, rref)) {
81                Some(s) => Response::json(
82                    200,
83                    json!({
84                        "ciphertext": s.ciphertext,
85                        "version": s.version,
86                        "created_at": s.created_at,
87                        "rotated_at": s.rotated_at,
88                    })
89                    .to_string(),
90                ),
91                None => Response::json(404, r#"{"error":"not found"}"#),
92            },
93            ("PUT", Some(rref)) => {
94                let body: serde_json::Value =
95                    serde_json::from_str(&req.text()).unwrap_or(json!({}));
96                let ciphertext = match body["ciphertext"].as_str() {
97                    Some(c) if !c.is_empty() => c.to_string(),
98                    _ => return Response::json(400, r#"{"error":"missing ciphertext"}"#),
99                };
100                let rotate = body["rotate"].as_bool().unwrap_or(false);
101                let key = (env, rref);
102                match (store.get(&key), rotate) {
103                    (None, true) => Response::json(404, r#"{"error":"not found"}"#),
104                    (None, false) => {
105                        let s = VaultRow {
106                            ciphertext,
107                            version: 1,
108                            created_at: now(),
109                            rotated_at: None,
110                        };
111                        let out = row_json(&s);
112                        store.insert(key, s);
113                        Response::json(200, out)
114                    }
115                    (Some(old), rot) => {
116                        let s = VaultRow {
117                            ciphertext,
118                            version: old.version + 1,
119                            created_at: old.created_at.clone(),
120                            rotated_at: if rot {
121                                Some(now())
122                            } else {
123                                old.rotated_at.clone()
124                            },
125                        };
126                        let out = row_json(&s);
127                        store.insert(key, s);
128                        Response::json(200, out)
129                    }
130                }
131            }
132            ("DELETE", Some(rref)) => {
133                let removed = store.remove(&(env, rref)).is_some();
134                Response::json(200, json!({ "removed": removed }).to_string())
135            }
136            _ => Response::json(405, r#"{"error":"method not allowed"}"#),
137        }
138    })
139}

141/// A fake CF API v4 Secrets Store. Routes under
142/// `/accounts/{account}/secrets_store/stores/{store}/secrets`:
143/// `GET ?search=` lists (name-substring), `POST` creates from an
144/// array body, `PATCH /{id}` updates, `DELETE /{id}` removes —
145/// all inside the `{success, errors, result}` envelope.
146///
147/// Returns the running mock plus the request log for assertions.
148pub fn cf_secrets_store(
149    account: &'static str,
150    store_id: &'static str,
151    token: &'static str,
152) -> (HttpMock, Arc<Mutex<Vec<Request>>>) {
153    let seen: Arc<Mutex<Vec<Request>>> = Arc::new(Mutex::new(Vec::new()));
154    let store: Arc<Mutex<BTreeMap<String, serde_json::Value>>> =
155        Arc::new(Mutex::new(BTreeMap::new()));
156    let next_id = Arc::new(Mutex::new(0u64));
157    let mock = {
158        let seen = seen.clone();
159        HttpMock::new(move |req: &Request| {
160            seen.lock().unwrap().push(req.clone());
161            if req.header("authorization") != Some(format!("Bearer {token}").as_str()) {
162                return Response::json(
163                    403,
164                    r#"{"success":false,"errors":[{"code":1000,"message":"bad token"}],"result":null}"#,
165                );
166            }
167            let ok = |result: serde_json::Value| {
168                Response::json(
169                    200,
170                    json!({"success":true,"errors":[],"result":result}).to_string(),
171                )
172            };
173            let base = format!("/accounts/{account}/secrets_store/stores/{store_id}/secrets");
174            let (path, query) = match req.path.split_once('?') {
175                Some((p, q)) => (p, q),
176                None => (req.path.as_str(), ""),
177            };
178            let suffix = path.strip_prefix(&base);
179            let mut store = store.lock().unwrap();
180            match (req.method.as_str(), suffix) {
181                ("GET", Some("")) => {
182                    let want = query
183                        .split('&')
184                        .find_map(|kv| kv.strip_prefix("search="))
185                        .unwrap_or("");
186                    let result: Vec<serde_json::Value> = store
187                        .values()
188                        .filter(|s| {
189                            want.is_empty() || s["name"].as_str().unwrap_or("").contains(want)
190                        })
191                        .map(|s| json!({"id": s["id"], "name": s["name"], "status": "active"}))
192                        .collect();
193                    ok(json!(result))
194                }
195                ("POST", Some("")) => {
196                    let body: serde_json::Value =
197                        serde_json::from_str(&req.text()).unwrap_or(json!([]));
198                    let mut results = Vec::new();
199                    for item in body.as_array().cloned().unwrap_or_default() {
200                        let name = item["name"].as_str().unwrap_or("").to_string();
201                        let mut id = next_id.lock().unwrap();
202                        *id += 1;
203                        let id = format!("id{:04}", *id);
204                        store.insert(
205                            id.clone(),
206                            json!({"id": id, "name": name, "value": item["value"]}),
207                        );
208                        results.push(json!({"id": id, "name": name}));
209                    }
210                    ok(json!(results))
211                }
212                ("PATCH", Some(suf)) | ("DELETE", Some(suf)) => {
213                    let id = suf.trim_start_matches('/');
214                    match store.get(&id.to_string()) {
215                        None => Response::json(
216                            404,
217                            r#"{"success":false,"errors":[{"code":1007,"message":"not found"}],"result":null}"#,
218                        ),
219                        Some(_) if req.method == "DELETE" => {
220                            store.remove(&id.to_string());
221                            ok(json!({"id": id}))
222                        }
223                        Some(old) => {
224                            let body: serde_json::Value =
225                                serde_json::from_str(&req.text()).unwrap_or_default();
226                            let mut new = old.clone();
227                            new["value"] = body["value"].clone();
228                            store.insert(id.to_string(), new.clone());
229                            ok(json!({"id": id, "name": new["name"]}))
230                        }
231                    }
232                }
233                _ => Response::json(404, r#"{"success":false,"errors":[],"result":null}"#),
234            }
235        })
236    };
237    (mock, seen)
238}

240#[cfg(test)]
241mod tests {
242    use super::*;
243    use std::io::{Read, Write};
244
245    /// Fire one raw HTTP request at `mock` and read the raw reply.
246    fn exchange(mock: &HttpMock, request: &str) -> String {
247        let addr = mock.url().trim_start_matches("http://");
248        let mut s = std::net::TcpStream::connect(addr).unwrap();
249        s.write_all(request.as_bytes()).unwrap();
250        s.shutdown(std::net::Shutdown::Write).unwrap();
251        let mut out = String::new();
252        s.read_to_string(&mut out).unwrap();
253        out
254    }
255
256    const AUTH: &str = "Authorization: Bearer tok\r\n";
257
258    #[test]
259    fn vault_wire_rejects_bad_routes_and_shapes() {
260        let mock = vault_wire("tok");
261        // no /v1/ prefix → 404
262        let out = exchange(&mock, &format!("GET /bogus HTTP/1.1\r\n{AUTH}\r\n"));
263        assert!(out.contains("404"), "{out}");
264        // /v1/ with an empty env → 404
265        let out = exchange(&mock, &format!("GET /v1/ HTTP/1.1\r\n{AUTH}\r\n"));
266        assert!(out.contains("404"), "{out}");
267        // PUT without a ciphertext field → 400
268        let out = exchange(
269            &mock,
270            &format!("PUT /v1/dev/x/y HTTP/1.1\r\n{AUTH}Content-Length: 2\r\n\r\n{{}}"),
271        );
272        assert!(out.contains("400"), "{out}");
273        // a method the protocol doesn't know → 405
274        let out = exchange(&mock, &format!("PATCH /v1/dev/x HTTP/1.1\r\n{AUTH}\r\n"));
275        assert!(out.contains("405"), "{out}");
276        // sanity: a real list still works
277        let out = exchange(&mock, &format!("GET /v1/dev HTTP/1.1\r\n{AUTH}\r\n"));
278        assert!(out.contains("200") && out.contains("\"refs\""), "{out}");
279    }
280
281    #[test]
282    fn cf_store_handles_missing_ids_and_odd_methods() {
283        let (mock, _) = cf_secrets_store("acc", "sto", "tok");
284        let base = "/accounts/acc/secrets_store/stores/sto/secrets";
285        // DELETE of an id that was never created → 404 envelope
286        let out = exchange(
287            &mock,
288            &format!("DELETE {base}/id9999 HTTP/1.1\r\n{AUTH}\r\n"),
289        );
290        assert!(out.contains("404"), "{out}");
291        // a method+suffix no arm claims → the 404 fallthrough
292        let out = exchange(&mock, &format!("GET {base}/id0001 HTTP/1.1\r\n{AUTH}\r\n"));
293        assert!(out.contains("404"), "{out}");
294        let out = exchange(&mock, &format!("PUT {base} HTTP/1.1\r\n{AUTH}\r\n"));
295        assert!(out.contains("404"), "{out}");
296    }
