1//! cuora-sealed — the sealed-pack contract, shared by the driver
2//! verbs and the engine's `cuora://<pack>/<secret>` ref fallback.
3//!
4//! Tamias carries age-encrypted payloads **opaquely**; this crate
5//! is the decrypt/seal side both front-ends use:
6//!
7//! - pack resolution against the tamias install store
8//!   (`$TAMIAS_HOME/store` → XDG → `~/.local/share/tamias/store`
9//!   → `./.tamias/store`, `--store`/`store` param overrides);
10//! - the age identity keyfile (`FileIdentity` —
11//!   `~/.config/cuora/identity.txt`, `$CUORA_IDENTITY_FILE`
12//!   override, `--keyfile` override);
13//! - `encrypt`/`decrypt` over `Zeroizing` buffers — plaintext
14//!   exists only between read and write, never logged;
15//! - [`open_pack_ref`] / [`pack_declares`] — the engine's read
16//!   path: `cuora://<pack>/<secret>` resolves to a decrypted
17//!   value (or a declared-secret probe for `cuora check`). The pack
18//!   segment may carry a `@<version>`/`@<sha>` pin —
19//!   `cuora://secrets-9aia@1.0.0/registry-token` — matching the
20//!   tamias store label spelling; an unpinned name matching several
21//!   installs is an ambiguity error, never a silent pick.
22//!
23//! The argv verbs (`seal`/`open`/`materialize`/`recipients`/
24//! `packs`) live in `cuora-driver`'s `sealed` module and call
25//! into this crate; `cuora-ops` calls [`open_pack_ref`] when a
26//! vault lookup misses on a two-segment ref. Audit emission stays
27//! with the callers (each knows its `Op` and actor).
28

42/// The tamias install store root — mirrors `tamias-cli`'s
43/// `store_root()` minus the config-file location (callers pass
44/// `--store`; `TAMIAS_HOME`/XDG cover the rest).
45pub fn store_root(store: Option<&Path>) -> PathBuf {
46    if let Some(s) = store {
47        return s.to_path_buf();
48    }
49    if let Some(root) = std::env::var_os("TAMIAS_HOME") {
50        return PathBuf::from(root).join("store");
51    }
52    std::env::var_os("XDG_DATA_HOME")
53        .map(|d| PathBuf::from(d).join("tamias/store"))
54        .or_else(|| {
55            std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".local/share/tamias/store"))
56        })
57        .unwrap_or_else(|| PathBuf::from(".tamias/store"))
58}

60/// One installed store entry — dir + record (manifest loads lazily).
61#[derive(Debug)]
62pub struct Installed {
63    dir: PathBuf,
64    record: StoreRecord,
65    label: String,
66}
67
68impl Installed {
69    /// `name@version` / `@scope/name@version` — same spelling
70    /// `tamias-store-local`'s `StoreEntry::label` produces.
71    fn label_of(record: &StoreRecord) -> String {
72        let scoped = match &record.scope {
73            Some(scope) => format!("@{scope}/{}", record.name),
74            None => record.name.clone(),
75        };
76        let version = record.version.clone().unwrap_or_else(|| {
77            let sha = record
78                .resolved
79                .strip_prefix("sha256:")
80                .unwrap_or(&record.resolved);
81            sha.chars().take(12).collect()
82        });
83        format!("{scoped}@{version}")
84    }
85
86    /// The payload dir — holds `tamias.yaml` + `sealed/`.
87    pub fn dir(&self) -> &Path {
88        &self.dir
89    }
90
91    /// `name@version` label for messages.
92    pub fn label(&self) -> &str {
93        &self.label
94    }
95
96    /// Bare pack name (`record.name`).
97    pub fn name(&self) -> &str {
98        &self.record.name
99    }
100
101    /// The store record (source ref, resolved digest, …).
102    pub fn record(&self) -> &StoreRecord {
103        &self.record
104    }
105
106    pub fn manifest(&self) -> Result<Manifest, String> {
107        Manifest::load(&self.dir).map_err(|e| format!("{}: {e}", self.dir.display()))
108    }
109}

111/// Why [`find_pack`] found nothing usable.
112#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
113pub enum FindPackError {
114    /// No installed entry matches the spec.
115    #[error("sealed pack '{spec}' is not installed under {root}")]
116    NotInstalled {
117        /// The spec that was looked up.
118        spec: String,
119        /// The store root searched.
120        root: PathBuf,
121    },
122    /// Several same-name installs — pin one by label
123    /// (`<spec>@<version>`, the tamias `name@version` spelling).
124    #[error(
125        "sealed pack '{spec}' has several installs — pin one as '{spec}@<version>' ({candidates})"
126    )]
127    Ambiguous {
128        /// The spec that was looked up.
129        spec: String,
130        /// Candidate labels, comma-joined.
131        candidates: String,
132    },
133}

135/// Depth-first walk for dirs carrying `.tamias-store.json` — the
136/// same shape `tamias-store-local`'s `collect` uses (dot-dirs and
137/// symlinked records skipped). Unparsable records warn + skip.
138pub fn scan_store(root: &Path) -> Vec<Installed> {
139    let mut out = Vec::new();
140    collect_entries(root, &mut out);
141    out.sort_by(|a, b| a.label.cmp(&b.label));
142    out
143}
144
145fn collect_entries(dir: &Path, out: &mut Vec<Installed>) {
146    let Ok(read) = std::fs::read_dir(dir) else {
147        return;
148    };
149    for entry in read.flatten() {
150        let path = entry.path();
151        let Ok(ft) = entry.file_type() else { continue };
152        if !ft.is_dir() {
153            continue;
154        }
155        // `file_type` never follows symlinks — a linked dir isn't
156        // an entry (same rule as the store's own walk).
157        if entry.file_name().to_string_lossy().starts_with('.') {
158            continue;
159        }
160        let record_path = path.join(RECORD_FILE);
161        let is_file = std::fs::symlink_metadata(&record_path)
162            .map(|m| m.is_file())
163            .unwrap_or(false);
164        if is_file {
165            match std::fs::read_to_string(&record_path)
166                .ok()
167                .and_then(|t| serde_json::from_str::<StoreRecord>(&t).ok())
168            {
169                Some(record) => {
170                    let label = Installed::label_of(&record);
171                    out.push(Installed {
172                        dir: path,
173                        record,
174                        label,
175                    });
176                }
177                None => eprintln!(
178                    "cuora sealed: skipping unparsable record {}",
179                    record_path.display()
180                ),
181            }
182        } else {
183            collect_entries(&path, out);
184        }
185    }
186}

188/// Resolve a pack spec — bare name, `@scope/name`, `name@version`,
189/// or a full `label()` — to an installed entry. Several same-name
190/// installs are [`FindPackError::Ambiguous`] (mirrors
191/// `Store::find`'s ambiguity rule).
192pub fn find_pack(root: &Path, spec: &str) -> Result<Installed, FindPackError> {
193    let entries = scan_store(root);
194    // Exact label first (`@scope/name@1.0.0`, `name@1.0.0`).
195    if let Some(e) = entries.iter().find(|e| e.label == spec) {
196        return Ok(clone_installed(e));
197    }
198    let scoped = |e: &Installed| match &e.record.scope {
199        Some(s) => format!("@{s}/{}", e.record.name),
200        None => e.record.name.clone(),
201    };
202    let mut matches: Vec<&Installed> = entries
203        .iter()
204        .filter(|e| e.record.name == spec || scoped(e) == spec || e.label == spec)
205        .collect();
206    match matches.len() {
207        0 => Err(FindPackError::NotInstalled {
208            spec: spec.to_string(),
209            root: root.to_path_buf(),
210        }),
211        1 => Ok(clone_installed(matches.swap_remove(0))),
212        _ => Err(FindPackError::Ambiguous {
213            spec: spec.to_string(),
214            candidates: matches
215                .iter()
216                .map(|e| e.label.as_str())
217                .collect::<Vec<_>>()
218                .join(", "),
219        }),
220    }
221}
222
223fn clone_installed(e: &Installed) -> Installed {
224    Installed {
225        dir: e.dir.clone(),
226        record: e.record.clone(),
227        label: e.label.clone(),
228    }
229}

233/// What a verb operates on — a payload dir plus its manifest's
234/// `sealed:` block.
235#[derive(Debug)]
236pub struct Target {
237    /// Human label for messages (`name@version`, `--name`, or the
238    /// `--dir` path).
239    pub label: String,
240    /// Bare pack name for audit refs — `record.name` from the store,
241    /// `--name`, or the payload dir's file name.
242    pub pack: String,
243    /// Payload root (the dir holding `tamias.yaml` + `sealed/`).
244    pub dir: PathBuf,
245    /// The manifest's `sealed:` block.
246    pub sealed: Sealed,
247}
248
249/// Resolve a pack spec against the store (or `--dir` mode: the
250/// payload dir is handed over directly — driver dispatch). The
251/// manifest must carry a `sealed:` block.
252pub fn resolve_target(
253    dir: Option<&Path>,
254    name: Option<&str>,
255    store: Option<&Path>,
256    pack_spec: Option<&str>,
257) -> Result<Target, String> {
258    if let Some(dir) = dir {
259        let manifest = Manifest::load(dir).map_err(|e| format!("{}: {e}", dir.display()))?;
260        let sealed = manifest
261            .sealed
262            .clone()
263            .ok_or_else(|| format!("{} carries no sealed: block", dir.display()))?;
264        let label = name
265            .map(str::to_string)
266            .unwrap_or_else(|| dir.display().to_string());
267        let pack = name
268            .map(str::to_string)
269            .or_else(|| dir.file_name().map(|n| n.to_string_lossy().into_owned()))
270            .unwrap_or_else(|| "payload".into());
271        return Ok(Target {
272            label,
273            pack,
274            dir: dir.to_path_buf(),
275            sealed,
276        });
277    }
278    let spec =
279        pack_spec.ok_or_else(|| "a pack name (or --dir <payload>) is required".to_string())?;
280    let root = store_root(store);
281    let entry = find_pack(&root, spec).map_err(|e| e.to_string())?;
282    let manifest = entry.manifest()?;
283    let sealed = manifest
284        .sealed
285        .clone()
286        .ok_or_else(|| format!("pack '{spec}' carries no sealed: block"))?;
287    Ok(Target {
288        label: entry.label.clone(),
289        pack: entry.record.name.clone(),
290        dir: entry.dir.clone(),
291        sealed,
292    })
293}

295/// Parse `cuora://<pack>/<secret>` — the sealed-pack ref form. The
296/// ref grammar allows deeper paths, but a sealed logical name is a
297/// single `sealed.files` key, so exactly two segments. `<pack>` may
298/// itself be a store spec — bare name, `@scope/name`, or a
299/// `name@version`/`name@<sha>` pin (see [`find_pack`]).
300pub fn parse_pack_ref(arg: &str) -> Result<(String, String), String> {
301    let r = Ref::new(arg).map_err(|e| e.to_string())?;
302    let segs = r.segments();
303    if segs.len() != 2 {
304        return Err(format!(
305            "sealed pack ref '{arg}' must be cuora://<pack>/<secret> (got {})",
306            r.path()
307        ));
308    }
309    Ok((segs[0].clone(), segs[1].clone()))
310}

314/// The identity keyfile source — `--keyfile` or the default
315/// (`$CUORA_IDENTITY_FILE` → `~/.config/cuora/identity.txt`, mode
316/// 0600, legacy `arca` fallback — all inside `FileIdentity`).
317pub fn identity_source(keyfile: Option<&Path>) -> FileIdentity {
318    match keyfile {
319        Some(p) => FileIdentity::new(p.to_path_buf()),
320        None => FileIdentity::default(),
321    }

324pub fn load_identity(keyfile: Option<&Path>) -> Result<x25519::Identity, String> {
325    let src = identity_source(keyfile);
326    src.load()
327        .map_err(|e| format!("identity from {}: {e}", src.describe()))?
328        .ok_or_else(|| {
329            format!(
330                "no age identity at {} — run `cuora init --with-file` or set CUORA_IDENTITY_FILE",
331                src.path().display()
332            )
333        })
334}

336/// The acting identity for audit attribution — `CUORA_ACTOR`,

349/// Encrypt `plain` to every recipient — fresh file key + nonce per
350/// call (age guarantees; re-sealing never reproduces ciphertext).
351pub fn encrypt(plain: &[u8], recipients: &[x25519::Recipient]) -> Result<Vec<u8>, String> {
352    let encryptor =
353        age::Encryptor::with_recipients(recipients.iter().map(|r| r as &dyn age::Recipient))
354            .map_err(|e| format!("age encryptor: {e}"))?;
355    let mut cipher = Vec::new();
356    let mut w = encryptor
357        .wrap_output(&mut cipher)
358        .map_err(|e| format!("age wrap: {e}"))?;
359    w.write_all(plain).map_err(|e| format!("age write: {e}"))?;
360    w.finish().map_err(|e| format!("age finish: {e}"))?;
361    Ok(cipher)
362}
363
364/// Decrypt an age file with `identity` — scrypt/passphrase files
365/// are refused (sealed packs are recipient-key only, same rule as
366/// the vault).
367pub fn decrypt(identity: &x25519::Identity, cipher: &[u8]) -> Result<Zeroizing<Vec<u8>>, String> {
368    let decryptor =
369        age::Decryptor::new_buffered(cipher).map_err(|e| format!("not an age file: {e}"))?;
370    if decryptor.is_scrypt() {
371        return Err("sealed payload is passphrase-encrypted; recipient key required".into());
372    }
373    let mut reader = decryptor
374        .decrypt(std::iter::once(identity as &dyn age::Identity))
375        .map_err(|e| format!("decrypt: {e}"))?;
376    let mut buf = Zeroizing::new(Vec::new());
377    std::io::Read::read_to_end(&mut reader, &mut buf)
378        .map_err(|e| format!("read plaintext: {e}"))?;
379    Ok(buf)
380}

382/// Parse a recipients file — one `age1…` per line, `#` comments and
383/// blanks skipped (the same shape `sealed/recipients` ships).
384pub fn parse_recipient_lines(text: &str) -> Result<Vec<String>, String> {
385    let mut out = Vec::new();
386    for (n, line) in text.lines().enumerate() {
387        let line = line.trim();
388        if line.is_empty() || line.starts_with('#') {
389            continue;
390        }
391        x25519::Recipient::from_str(line).map_err(|e| format!("recipient line {}: {e}", n + 1))?;
392        out.push(line.to_string());
393    }
394    Ok(out)
395}

397/// Write `bytes` to `path` with parent dirs created; on unix the
398/// file lands `0600` at open (no world-readable window) — materialized
399/// secrets are as sensitive as the keyfile.
400pub fn write_secret_file(path: &Path, bytes: &[u8]) -> Result<(), String> {
401    if let Some(parent) = path.parent() {
402        std::fs::create_dir_all(parent).map_err(|e| format!("create {}: {e}", parent.display()))?;
403    }
404    let mut opts = std::fs::OpenOptions::new();
405    opts.write(true).create(true).truncate(true);
406    #[cfg(unix)]
407    {
408        use std::os::unix::fs::OpenOptionsExt;
409        opts.mode(0o600);
410    }
411    let mut f = opts
412        .open(path)
413        .map_err(|e| format!("write {}: {e}", path.display()))?;
414    f.write_all(bytes)
415        .map_err(|e| format!("write {}: {e}", path.display()))?;
416    f.sync_all()
417        .map_err(|e| format!("sync {}: {e}", path.display()))
418}

422/// The result of [`open_pack_ref`] — a three-way answer so the
423/// engine can tell "no pack answers this" (fall back to its usual
424/// not-stored) from "the pack answered".
425#[derive(Debug)]
426pub enum PackOpen {
427    /// No installed sealed pack named `pack` — or the match
428    /// carries no `sealed:` block. The caller reports its normal
429    /// miss.
430    Miss,
431    /// The pack is installed but declares no secret `secret`.
432    NoSecret,
433    /// Decrypted plaintext (`Zeroizing` — scrubbed on drop).
434    Hit(Zeroizing<Vec<u8>>),
435}
436
437/// Resolve `cuora://<pack>/<secret>` against the tamias install
438/// store and decrypt — the engine's pack-ref read path.
439///
440/// `store`/`keyfile` are the same overrides the driver verbs take
441/// (`--store`/`--keyfile`); `None` means the standard resolution
442/// (`TAMIAS_HOME`/XDG/`~/.local/share`, `CUORA_IDENTITY_FILE`/default
443/// keyfile).
444///
445/// `Ok(Miss)`/`Ok(NoSecret)` are *answers*, not errors: the engine
446/// maps them onto its not-stored reporting. `Err` is a real
447/// failure — ambiguous installs, unreadable manifest/store record,
448/// missing identity, decrypt failure — and should surface to the
449/// user rather than silently degrade to a vault miss.
450pub fn open_pack_ref(
451    pack: &str,
452    secret: &str,
453    store: Option<&Path>,
454    keyfile: Option<&Path>,
455) -> Result<PackOpen, String> {
456    let Some((target, sealed)) = find_sealed(pack, store)? else {
457        return Ok(PackOpen::Miss);
458    };
459    let Some(rel) = sealed.files.get(secret) else {
460        return Ok(PackOpen::NoSecret);
461    };
462    let cipher_path = target.dir.join(rel);
463    let cipher =
464        std::fs::read(&cipher_path).map_err(|e| format!("read {}: {e}", cipher_path.display()))?;
465    let identity = load_identity(keyfile)?;
466    let plain = decrypt(&identity, &cipher)?;
467    Ok(PackOpen::Hit(plain))
468}

470/// Does an installed sealed pack `pack` declare `secret`? The
471/// `cuora check` probe — no identity needed, no decrypt.
472/// `Ok(false)` covers both "no such pack" and "pack doesn't
473/// declare it"; `Err` is a real lookup failure (ambiguous install,
474/// unreadable manifest).
475pub fn pack_declares(pack: &str, secret: &str, store: Option<&Path>) -> Result<bool, String> {
476    let Some((_, sealed)) = find_sealed(pack, store)? else {
477        return Ok(false);
478    };
479    Ok(sealed.files.contains_key(secret))
480}

482/// Find the installed sealed pack `pack` spec resolves to —
483/// `Ok(None)` when nothing matches or the match isn't sealed.
484fn find_sealed(pack: &str, store: Option<&Path>) -> Result<Option<(Target, Sealed)>, String> {
485    let root = store_root(store);
486    let entry = match find_pack(&root, pack) {
487        Ok(e) => e,
488        Err(FindPackError::NotInstalled { .. }) => return Ok(None),
489        Err(e) => return Err(e.to_string()),
490    };
491    let manifest = entry.manifest()?;
492    let Some(sealed) = manifest.sealed.clone() else {
493        return Ok(None);
494    };
495    Ok(Some((
496        Target {
497            label: entry.label.clone(),
498            pack: entry.record.name.clone(),
499            dir: entry.dir.clone(),
500            sealed: sealed.clone(),
501        },
502        sealed,
503    )))
504}

601    /// Write a FileIdentity-format keyfile, returning the public key.
602    fn identity_keyfile(kf: &Path) -> String {
603        let (id, pubkey) = identity();
604        // The keyfile the caller passes must BE this identity —
605        // write it here so callers don't duplicate the format.
606        let body = {
607            use age::secrecy::ExposeSecret;
608            format!(
609                "# cuora identity\n# public: {pubkey}\n{}\n",
610                id.to_string().expose_secret()
611            )
612        };
613        std::fs::write(kf, body).unwrap();
614        #[cfg(unix)]
615        {
616            use std::os::unix::fs::PermissionsExt;
617            std::fs::set_permissions(kf, std::fs::Permissions::from_mode(0o600)).unwrap();
618        }
619        pubkey
620    }

622    /// Build a fake store entry sealed TO `pubkey` — the shared
623    /// half of [`fake_store`] for multi-install fixtures.
624    fn fake_store_pub(
625        root: &Path,
626        label: &str,
627        name: &str,
628        version: &str,
629        secret: &str,
630        value: &[u8],
631        pubkey: &str,
632    ) {
633        let recip = x25519::Recipient::from_str(pubkey).unwrap();
634        let cipher = encrypt(value, &[recip]).unwrap();
635
636        let dir = root.join(label);
637        std::fs::create_dir_all(dir.join("sealed")).unwrap();
638        std::fs::write(dir.join("sealed/recipients"), format!("{pubkey}\n")).unwrap();
639        std::fs::write(dir.join(format!("sealed/{secret}.age")), &cipher).unwrap();
640        std::fs::write(
641            dir.join("tamias.yaml"),
642            format!(
643                "name: {name}\nversion: \"{version}\"\ntype: secrets\nsealed:\n  format: age/v1\n  recipients: sealed/recipients\n  files:\n    {secret}: sealed/{secret}.age\n"
644            ),
645        )
646        .unwrap();
647        std::fs::write(
648            dir.join(RECORD_FILE),
649            format!(
650                r#"{{"name":"{name}","version":"{version}","source_ref":"x","resolved":"sha256:abc","digest":"d","installed_at":0}}"#
651            ),
652        )
653        .unwrap();
654    }

666    fn open_pack_ref_hits_misses_and_errors() {
667        let dir = tempfile::tempdir().unwrap();
668        let store = dir.path().join("store");
669        std::fs::create_dir_all(&store).unwrap();
670        let kf = dir.path().join("identity.txt");
671        fake_store(
672            &store,
673            "secrets-test@1.0.0",
674            "secrets-test",
675            "token",
676            b"s3cr3t",
677            &kf,
678        );
679
680        // Hit — decrypted value comes back zeroized.
681        match open_pack_ref("secrets-test", "token", Some(&store), Some(&kf)).unwrap() {
682            PackOpen::Hit(v) => assert_eq!(&*v, b"s3cr3t"),
683            other => panic!("expected Hit, got {}", variant(&other)),
684        }
685        // Unknown pack → Miss (engine reports its usual not-stored).
686        assert!(matches!(
687            open_pack_ref("nope", "token", Some(&store), Some(&kf)).unwrap(),
688            PackOpen::Miss
689        ));
690        // Pack installed, secret undeclared → NoSecret.
691        assert!(matches!(
692            open_pack_ref("secrets-test", "other", Some(&store), Some(&kf)).unwrap(),
693            PackOpen::NoSecret
694        ));
695        // pack_declares agrees without touching identity/decrypt.
696        assert!(pack_declares("secrets-test", "token", Some(&store)).unwrap());
697        assert!(!pack_declares("secrets-test", "other", Some(&store)).unwrap());
698        assert!(!pack_declares("nope", "token", Some(&store)).unwrap());
699    }

742    #[test]
743    fn version_pin_resolves_ambiguous_installs() {
744        let dir = tempfile::tempdir().unwrap();
745        let store = dir.path().join("store");
746        std::fs::create_dir_all(&store).unwrap();
747        let kf = dir.path().join("identity.txt");
748        let pubkey = identity_keyfile(&kf);
749        // Two installs of the same pack name, different versions —
750        // sealed to the SAME recipient so one keyfile opens both.
751        fake_store_pub(
752            &store,
753            "secrets-dup@1.0.0",
754            "secrets-dup",
755            "1.0.0",
756            "token",
757            b"value-v1",
758            &pubkey,
759        );
760        fake_store_pub(
761            &store,
762            "secrets-dup@2.0.0",
763            "secrets-dup",
764            "2.0.0",
765            "token",
766            b"value-v2",
767            &pubkey,
768        );
769
770        // Unpinned → Ambiguous, and the message teaches the pin.
771        match find_pack(&store, "secrets-dup") {
772            Err(FindPackError::Ambiguous { spec, candidates }) => {
773                assert_eq!(spec, "secrets-dup");
774                assert!(candidates.contains("secrets-dup@1.0.0"));
775                assert!(candidates.contains("secrets-dup@2.0.0"));
776            }
777            other => panic!("expected Ambiguous, got {other:?}"),
778        }
779        // The pin spelling is the tamias `name@version` label.
780        for (pin, want) in [
781            ("secrets-dup@1.0.0", b"value-v1".as_slice()),
782            ("secrets-dup@2.0.0", b"value-v2".as_slice()),
783        ] {
784            match open_pack_ref(pin, "token", Some(&store), Some(&kf)).unwrap() {
785                PackOpen::Hit(v) => assert_eq!(&*v, want),
786                other => panic!("expected Hit for {pin}, got {}", variant(&other)),
787            }
788            assert!(pack_declares(pin, "token", Some(&store)).unwrap());
789        }
790        // A pin that matches nothing is a plain miss, not ambiguity.
791        assert!(matches!(
792            open_pack_ref("secrets-dup@9.9.9", "token", Some(&store), Some(&kf)).unwrap(),
793            PackOpen::Miss
794        ));
795        // And the engine-facing error for the unpinned ref names the fix.
796        let err = open_pack_ref("secrets-dup", "token", Some(&store), Some(&kf)).unwrap_err();
797        assert!(err.contains("@<version>"), "{err}");
798    }
